    48	// Called when driver is first loaded.
    49	NTSTATUS
    50	DriverEntry(
    51	  __in PDRIVER_OBJECT   driverObject,
    52	  __in PUNICODE_STRING  registryPath)
    53	{
    54	   NTSTATUS                      retStatus = STATUS_SUCCESS;
    55	   HID_MINIDRIVER_REGISTRATION   hidMinidriverRegistration;
    56	   ULONG functionIndex = 0;
    57	
    58	   ChatpadTrace(("XBox 360 Controller Chatpad Virtual Mouse Device by GAFBlizzard.\n"));
    59	   ChatpadTrace(("Build %s %s\n", __DATE__, __TIME__));
    60	
    61	   // Initialize the dispatch table to pass through all the IRPs by default.
    62	   for(functionIndex = 0; functionIndex <= IRP_MJ_MAXIMUM_FUNCTION; functionIndex++)
    63	   {
    64	      driverObject->MajorFunction[functionIndex] = ChatpadMousePassThrough;
    65	   }
    66	
    67	   // Set up special case for IRP_MJ_POWER.
    68	   driverObject->MajorFunction[IRP_MJ_POWER] = ChatpadMousePowerPassThrough;
    69	
    70	   // Set up add device and driver unload functions.
    71	   driverObject->DriverExtension->AddDevice = ChatpadMouseAddDevice;
    72	   driverObject->DriverUnload = DriverUnload;
    73	
    74	   RtlZeroMemory(&hidMinidriverRegistration, sizeof(hidMinidriverRegistration));
    75	   hidMinidriverRegistration.Revision              = HID_REVISION;
    76	   hidMinidriverRegistration.DriverObject          = driverObject;
    77	   hidMinidriverRegistration.RegistryPath          = registryPath;
    78	   hidMinidriverRegistration.DeviceExtensionSize   = 0;
    79	   hidMinidriverRegistration.DevicesArePolled      = FALSE;
    80	
    81	   // Register with hidclass.
    82	   retStatus = HidRegisterMinidriver(&hidMinidriverRegistration);
    83	   if(STATUS_SUCCESS == retStatus)
    84	   {
    85	      ChatpadTrace(("HidRegisterMinidriver succeeded.\n"));
    86	   }
    87	   else
    88	   {
    89	      ChatpadTrace(("HidRegisterMinidriver failed:  0x%0x\n", retStatus));
    90	   }
    91	
    92	   return retStatus;
    93	} // end DriverEntry

   118	// This function passes through non-power IRPs.
   119	NTSTATUS
   120	ChatpadMousePassThrough(
   121	  __in PDEVICE_OBJECT   deviceObject,
   122	  __in PIRP             irp)
   123	{
   124	   NTSTATUS             retStatus      = STATUS_SUCCESS;
   125	   PIO_STACK_LOCATION   stackLocation  = NULL;
   126	
   127	   stackLocation = IoGetCurrentIrpStackLocation(irp);
   128	   if(NULL != stackLocation)
   129	   {
   130	//      ChatpadTrace(("Major function code %u, minor function code %u.\n",
   131	//                    stackLocation->MajorFunction,
   132	//                    stackLocation->MinorFunction));
   133	   }
   134	   else
   135	   {
   136	      ChatpadTrace(("Error, stack location pointer is NULL.\n"));
   137	   }
   138	
   139	   // Copy the stack to the next level.
   140	   IoCopyCurrentIrpStackLocationToNext(irp);
   141	
   142	   retStatus = IoCallDriver(GET_NEXT_DEVICE_OBJECT(deviceObject), irp);
   143	//   ChatpadTrace(("ChatpadMousePassThrough returning retStatus 0x%x\n", retStatus));
   144	
   145	   return retStatus;
   146	} // end ChatpadMousePassThrough
