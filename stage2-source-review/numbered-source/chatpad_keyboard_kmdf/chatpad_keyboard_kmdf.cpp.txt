   157	// Called when driver is first loaded.
   158	extern "C"
   159	NTSTATUS
   160	DriverEntry(
   161	  __in PDRIVER_OBJECT  driverObject,
   162	  __in PUNICODE_STRING registryPath)
   163	{
   164	   NTSTATUS                retStatus = STATUS_SUCCESS;
   165	   WDF_DRIVER_CONFIG       driverConfig;
   166	   WDF_OBJECT_ATTRIBUTES   objectAttributes;
   167	
   168	   ChatpadTrace(("XBox 360 Controller Chatpad Virtual Keyboard KMDF Layer by GAFBlizzard.\n"));
   169	   ChatpadTrace(("Build %s %s\n", __DATE__, __TIME__));
   170	
   171	   // This is just an information printout so I can check the size of a DEVICE_CONTEXT structure
   172	   // (only the structure itself, not any additional allocated memory).
   173	   ChatpadTrace(("sizeof(DEVICE_CONTEXT) == %u\n", sizeof(DEVICE_CONTEXT)));
   174	   ChatpadTrace(("sizeof(CHATPAD_KEYBOARD_HID_INPUT_REPORT)  == %u\n", sizeof(CHATPAD_KEYBOARD_HID_INPUT_REPORT)));
   175	// TODO NEXT REMOVE BELOW
   176	ChatpadTrace(("report descriptor size is %u\n", sizeof(chatpadKeyboardHidReportDescriptor))); 
   177	
   178	   WDF_DRIVER_CONFIG_INIT(&driverConfig,
   179	                          ChatpadKeyboardKMDFEvtDeviceAdd);
   180	
   181	   // Set up the driver unload function.
   182	   driverConfig.EvtDriverUnload = DriverUnload;
   183	
   184	   // Register a cleanup callback.
   185	   WDF_OBJECT_ATTRIBUTES_INIT(&objectAttributes);
   186	   objectAttributes.EvtCleanupCallback = ChatpadKeyboardKMDFDriverContextCleanup;
   187	
   188	   // Create framework driver object (FDO).
   189	   retStatus = WdfDriverCreate(
   190	     driverObject,
   191	     registryPath,
   192	     &objectAttributes,
   193	     &driverConfig,
   194	     WDF_NO_HANDLE);
   195	   if(STATUS_SUCCESS != retStatus)
   196	   {
   197	      ChatpadTrace(("WdfDriverCreate failed:  0x%0x\n", retStatus));
   198	   }
   199	
   200	   // Create collection for different instances of the top-level (non-control) device.
   201	   if(STATUS_SUCCESS == retStatus)
   202	   {
   203	      retStatus = WdfCollectionCreate(
   204	        WDF_NO_OBJECT_ATTRIBUTES,
   205	        &chatpadKeyboardKMDFDeviceCollection);
   206	      if(STATUS_SUCCESS != retStatus)
   207	      {
   208	         ChatpadTrace(("WdfCollectionCreate failed:  0x%x\n", retStatus));
   209	      }
   210	   }
   211	
   212	   if(STATUS_SUCCESS == retStatus)
   213	   {
   214	      retStatus = WdfWaitLockCreate(
   215	        WDF_NO_OBJECT_ATTRIBUTES,
   216	        &chatpadKeyboardKMDFDeviceCollectionLock);
   217	      if(STATUS_SUCCESS != retStatus)
   218	      {
   219	         ChatpadTrace(("WdfWaitLockCreate failed:  0x%x\n", retStatus));
   220	      }
   221	   }
   222	
   223	   return retStatus;
   224	} // end DriverEntry

   312	NTSTATUS
   313	ChatpadKeyboardKMDFEvtDeviceAdd(
   314	  IN WDFDRIVER       driver,
   315	  IN PWDFDEVICE_INIT deviceInit)
   316	{
   317	   NTSTATUS                      retStatus = STATUS_SUCCESS;
   318	   WDF_IO_QUEUE_CONFIG           ioQueueConfig;
   319	   WDF_OBJECT_ATTRIBUTES         objectAttributes;
   320	   WDFDEVICE                     wdfDevice;
   321	   PDEVICE_CONTEXT               devContext = NULL;
   322	   WDFQUEUE                      queue;
   323	   WDF_PNPPOWER_EVENT_CALLBACKS  powerCallbacks;
   324	   UCHAR                         minorFunction;
   325	
   326	   if(STATUS_SUCCESS == retStatus)
   327	   {
   328	      // Mark ourselves as a filter, causing requests we don't care about to be
   329	      // passed along etc.
   330	      WdfFdoInitSetFilter(deviceInit);
   331	
   332	      // Mark our device type as unknown.
   333	      // TODO this was from another example, I think, but the hidusbfx2 example does not have it.
   334	      //   Is there something I should understand about what it does and whether I need it?
   335	      // TODO should this be set eventually since this driver is not generic, or is the
   336	      //      automatic stuff fine?
   337	//      WdfDeviceInitSetDeviceType(deviceInit, FILE_DEVICE_UNKNOWN);
   338	
   339	      // Override IRP_MN_QUERY_ID, like vmulti does.
   340	      // TODO is overriding this IRP necessary?  It may only be necessary for multifunction drivers.
   341	      minorFunction = IRP_MN_QUERY_ID;
   342	      retStatus = WdfDeviceInitAssignWdmIrpPreprocessCallback(
   343	        deviceInit,
   344	        ChatpadKeyboardKMDFEvtWdmPreprocessMnQueryId,
   345	        IRP_MJ_PNP,        // Major function.
   346	        &minorFunction,    // Array of minor functions.
   347	        1);                // Number of minor functions.
   348	      if(STATUS_SUCCESS != retStatus)
   349	      {
   350	         ChatpadTrace(("WdfDeviceInitAssignWdmIrpPreprocessCallback failed:  0x%x\n", retStatus));
   351	      }
   352	   }
   353	
   354	   if(STATUS_SUCCESS == retStatus)
   355	   {
   356	      // Register a prepare hardware callback to create a WDF USB device object.
   357	      WDF_PNPPOWER_EVENT_CALLBACKS_INIT(&powerCallbacks);
   358	      powerCallbacks.EvtDevicePrepareHardware = ChatpadKeyboardKMDFEvtPrepareHardware;
   359	      powerCallbacks.EvtDeviceD0Entry = ChatpadKeyboardKMDFD0Entry;
   360	      powerCallbacks.EvtDeviceD0Exit = ChatpadKeyboardKMDFD0Exit;
   361	
   362	      WdfDeviceInitSetPnpPowerEventCallbacks(deviceInit, &powerCallbacks);
   363	
   364	      WDF_OBJECT_ATTRIBUTES_INIT_CONTEXT_TYPE(&objectAttributes, DEVICE_CONTEXT);
   365	
   366	      // Register for cleanup notification.
   367	      objectAttributes.EvtCleanupCallback = ChatpadKeyboardKMDFContextCleanup;
   368	
   369	      // Create framework device object, attach to the lower stack etc.
   370	      retStatus = WdfDeviceCreate(&deviceInit, &objectAttributes, &wdfDevice);
   371	      if(STATUS_SUCCESS != retStatus)
   372	      {
   373	         ChatpadTrace(("WdfDeviceCreate failed:  0x%x\n", retStatus));
   374	      }
   375	   } // end if(STATUS_SUCCESS == retStatus)
   376	
   377	   if(STATUS_SUCCESS == retStatus)
   378	   {
   379	      devContext = ChatpadKeyboardKMDFGetDeviceContext(wdfDevice);
   380	      if(NULL != devContext)
   381	      {
   382	         // Initialize DEVICE_CONTEXT values.
   383	         ChatpadTrace(("Initializing device context.\n"));
   384	
   385	         // TODO is it possible for a driver to read from the registry and detect whether to start up enabled?
   386	         //   (apparently so, see http://flylib.com/books/en/3.141.1.102/1/)
   387	         devContext->keyboardEnabled = FALSE;
   388	
   389	         if(STATUS_SUCCESS == retStatus)
   390	         {
   391	            // Initialize default queue config structure.
   392	            WDF_IO_QUEUE_CONFIG_INIT_DEFAULT_QUEUE(&ioQueueConfig, WdfIoQueueDispatchParallel);
   393	
   394	            // Select what we want from the queue.
   395	            // This should catch IOCTL_INTERNAL_USB_SUBMIT_URB.
   396	            ioQueueConfig.EvtIoInternalDeviceControl =
   397	              ChatpadKeyboardKMDFInternalIOCTLHandler;
   398	
   399	            // Create the default queue.
   400	            retStatus = WdfIoQueueCreate(
   401	              wdfDevice,
   402	              &ioQueueConfig,
   403	              WDF_NO_OBJECT_ATTRIBUTES,
   404	              NULL);
   405	            if(STATUS_SUCCESS == retStatus)
   406	            {
   407	               ChatpadTrace(("Created chatpad keyboard KMDF layer default queue successfully.\n"));
   408	            }
   409	            else
   410	            {
   411	               ChatpadTrace(("WdfIoQueueCreate failed:  0x%x\n", retStatus));
   412	            }
   413	         } // end if(STATUS_SUCCESS == retStatus)
   414	
   415	         if(STATUS_SUCCESS == retStatus)
   416	         {
   417	            // Create manual I/O queue for handling HID read requests from
   418	            // above us.
   419	            WDF_IO_QUEUE_CONFIG hidReadRequestQueueConfig;
   420	
   421	            WDF_IO_QUEUE_CONFIG_INIT(&hidReadRequestQueueConfig, WdfIoQueueDispatchManual);
   422	            hidReadRequestQueueConfig.PowerManaged = WdfFalse;
   423	
   424	            retStatus =
   425	              WdfIoQueueCreate(
   426	                wdfDevice,
   427	                &hidReadRequestQueueConfig,
   428	                WDF_NO_OBJECT_ATTRIBUTES,
   429	                &(devContext->hidReadRequestQueue));
   430	            if(STATUS_SUCCESS == retStatus)
   431	            {
   432	               ChatpadTrace(("Created HID read request queue successfully.\n"));
   433	            }
   434	            else
   435	            {
   436	               ChatpadTrace(("WdfIoQueueCreate failed:  0x%x\n", retStatus));
   437	            }
   438	         } // end if(STATUS_SUCCESS == retStatus)
   439	      } // end if(NULL != devContext)
   440	      else
   441	      {
   442	         retStatus = STATUS_INTERNAL_ERROR;
   443	      }
   444	   } // end if(STATUS_SUCCESS == retStatus)
   445	
   446	   if(STATUS_SUCCESS == retStatus)
   447	   {
   448	      // Add the device to the collection.
   449	      WdfWaitLockAcquire(chatpadKeyboardKMDFDeviceCollectionLock, NULL);
   450	
   451	      retStatus = WdfCollectionAdd(
   452	        chatpadKeyboardKMDFDeviceCollection,
   453	        wdfDevice);
   454	      if(STATUS_SUCCESS != retStatus)
   455	      {
   456	         ChatpadTrace(("WdfCollectionAdd failed:  0x%x\n", retStatus));
   457	      }
   458	
   459	      WdfWaitLockRelease(chatpadKeyboardKMDFDeviceCollectionLock);
   460	   }
   461	
   462	   // Create the control device.
   463	   if(STATUS_SUCCESS == retStatus)
   464	   {
   465	      retStatus = ChatpadKeyboardKMDFCreateControlDevice(wdfDevice);
   466	      // ChatpadKeyboardKMDFCreateControlDevice should print an error if it fails.
   467	   }
   468	
   469	   return retStatus;
   470	} // end ChatpadKeyboardKMDFEvtDeviceAdd

   485	// This code is hand-written but is patterned after VMultiEvtWdmPreprocessMnQueryID
   486	// from vmulti (http://code.google.com/p/vmulti/).
   487	NTSTATUS
   488	ChatpadKeyboardKMDFEvtWdmPreprocessMnQueryId(
   489	  WDFDEVICE       device,
   490	  PIRP            irp)
   491	{
   492	   NTSTATUS             retStatus               = STATUS_SUCCESS;
   493	   PIO_STACK_LOCATION   currentStackLocation    = NULL;
   494	   PIO_STACK_LOCATION   previousStackLocation   = NULL;
   495	   PDEVICE_OBJECT       deviceObject            = NULL;
   496	   PWCHAR               buffer                  = NULL;
   497	
   498	   currentStackLocation = IoGetCurrentIrpStackLocation(irp);
   499	   deviceObject = WdfDeviceWdmGetDeviceObject(device);
   500	
   501	   // According to vmulti, this check is required to filter out QUERY_IDs
   502	   // forwarded by the HIDCLASS for the parent framework device object.
   503	   // These IDs, says vmulti, are sent by the PNP manager for the parent
   504	   // framework device object if you root-enumerate this driver.
   505	   // TODO study the documentation to fully understand how all this works!
   506	   // TODO also study how stack locations work to fully understand the pointer arithmetic.
   507	   previousStackLocation =
   508	     (reinterpret_cast<PIO_STACK_LOCATION>(
   509	        reinterpret_cast<PUCHAR>(currentStackLocation) +
   510	        sizeof(IO_STACK_LOCATION)));
   511	   if(NULL == previousStackLocation)
   512	   {
   513	      ChatpadTrace(("Error trying to get previous stack pointer.\n"));
   514	      retStatus = STATUS_INTERNAL_ERROR;
   515	   }
   516	
   517	   if(STATUS_SUCCESS == retStatus)
   518	   {
   519	      if(previousStackLocation->DeviceObject == deviceObject)
   520	      {
   521	         // The vmulti code says that filtering out this basically prevents the
   522	         // found new hardware popup for a root-enumerated device (on reboot
   523	         // only?).
   524	         // TODO study how this works
   525	         retStatus = irp->IoStatus.Status;
   526	      }
   527	      else
   528	      {
   529	         if((BusQueryDeviceID == currentStackLocation->Parameters.QueryId.IdType) ||
   530	            (BusQueryHardwareIDs == currentStackLocation->Parameters.QueryId.IdType))
   531	         {
   532	            // In this case, according to vmulti, HIDClass is asking for a child
   533	            // device ID and hardware IDs.
   534	            // TODO study how this works regarding the device/hardware IDs
   535	
   536	            // TODO NEXT do we ever need to free this, or is that handled
   537	            //   automatically by the system, or when the driver is unloaded?
   538	            buffer = static_cast<PWCHAR>(ExAllocatePoolWithTagWrapper(
   539	              NonPagedPool,
   540	              CHATPADKEYBOARDKMDF_HARDWARE_IDS_LENGTH,
   541	              CHATPADKEYBOARDKMDF_POOL_TAG));
   542	
   543	            if(NULL != buffer)
   544	            {
   545	               RtlCopyMemory(
   546	                 buffer,
   547	                 CHATPADKEYBOARDKMDF_HARDWARE_IDS,
   548	                 CHATPADKEYBOARDKMDF_HARDWARE_IDS_LENGTH);
   549	               // TODO why do we have to cast to a ULONG_PTR?  And why ULONG_PTR, not PULONG?
   550	               irp->IoStatus.Information = reinterpret_cast<ULONG_PTR>(buffer);
   551	            }
   552	            else
   553	            {
   554	               ChatpadTrace(("Failed to allocate memory for hardware IDs string.\n"));
   555	               retStatus = STATUS_INSUFFICIENT_RESOURCES;
   556	            }
   557	            irp->IoStatus.Status = retStatus;
   558	         }
   559	         else
   560	         {
   561	            retStatus = irp->IoStatus.Status;
   562	         }
   563	      } // end case where previous stack location's device object is not the current device object
   564	
   565	      // TODO NEXT WIP is this invalid?  See http://msdn.microsoft.com/en-us/library/ff560205%28v=VS.85%29.aspx, I/O error 0x22E
   566	      IoCompleteRequest(irp, IO_NO_INCREMENT);
   567	   } // end if(STATUS_SUCCESS == retStatus)
   568	
   569	   ChatpadTrace(("ChatpadKeyboardKMDFEvtWdmPreprocessMnQueryId returning status 0x%x\n", retStatus));
   570	
   571	   return retStatus;
   572	} // end ChatpadKeyboardKMDFEvtWdmPreprocessMnQueryId

   575	ChatpadKeyboardKMDFControlDeviceCreateHandler(
   576	  WDFDEVICE       device,
   577	  WDFREQUEST      request,
   578	  WDFFILEOBJECT   fileObject)
   579	{
   580	   NTSTATUS apiStatus = STATUS_SUCCESS;
   581	
   582	   // Increment the open count.
   583	   WdfWaitLockAcquire(chatpadKeyboardKMDFDeviceCollectionLock, NULL);
   584	
   585	   // Since I have a wait lock, can I safely just modify it like this?
   586	   if(numOpenHandles < 1)

   590	   else

   596	   ChatpadTrace(("There are now %lu open handles to the chatpad keyboard KMDF control device.\n", numOpenHandles));
   597	
   598	   if(STATUS_SUCCESS == apiStatus)
   599	   {
   600	      // TODO there is probably a cleaner way to do this than having one extension
   601	      //   have a pointer to the other one etc...maybe?
   602	      PCHATPAD_KEYBOARD_KMDF_CONTROL_DEVICE_EXTENSION controlExtension  = NULL;
   603	      PDEVICE_CONTEXT                                 devContext        = NULL;
   604	
   605	      if(STATUS_SUCCESS == apiStatus)
   606	      {
   607	         controlExtension = ChatpadKeyboardKMDFControlGetData(chatpadKeyboardKMDFControlDevice);
   608	
   609	         if(NULL == controlExtension)
   610	         {
   611	            ChatpadTrace(("Failed to get control device extension in open handler.\n"));
   612	            apiStatus = STATUS_INTERNAL_ERROR;
   613	         }
   614	      }
   615	
   616	      if(STATUS_SUCCESS == apiStatus)
   617	      {
   618	         devContext = controlExtension->keyboardDeviceContext;
   619	         if(NULL == devContext)
   620	         {
   621	            ChatpadTrace(("Failed to get device context in open handler.\n"));

   628	         // Mark the virtual keyboard as enabled when the application connects.
   629	         // TODO could messing with the device context here cause synchronization issues?
   630	         devContext->keyboardEnabled = TRUE;
   631	         ChatpadTrace(("Virtual keyboard enabled on application connect.\n"));
   632	      }
   633	   }
   634	
   635	   WdfRequestComplete(request, apiStatus);
   636	
   637	   WdfWaitLockRelease(chatpadKeyboardKMDFDeviceCollectionLock);

   640	VOID
   641	ChatpadKeyboardKMDFControlDeviceCloseHandler(
   642	  WDFFILEOBJECT   fileObject)
   643	{
   644	   NTSTATUS apiStatus = STATUS_SUCCESS;
   645	
   646	   ChatpadTrace(("The IRQL in the keyboard close handler is %u.\n", KeGetCurrentIrql()));
   647	
   648	   // Decrement the open count.
   649	   WdfWaitLockAcquire(chatpadKeyboardKMDFDeviceCollectionLock, NULL);
   650	
   651	   // Since I have a wait lock, can I safely just modify it like this?
   652	   if(numOpenHandles > 0)
   653	   {
   654	      numOpenHandles--;
   655	   }
   656	   else
   657	   {
   658	      ChatpadTrace(("Error, there are supposedly 0 open handles but the close handler was called...\n"));
   659	   }
   660	
   661	   ChatpadTrace(("There are now %lu open handles to the chatpad keyboard KMDF control device.\n", numOpenHandles));
   662	
   663	   if(0 == numOpenHandles)
   664	   {
   665	      // TODO there is probably a cleaner way to do this than having one extension
   666	      //   have a pointer to the other one etc...maybe?
   667	      //   I want to make sure the no-keys-pressed HID response gets sent, though.
   668	      PCHATPAD_KEYBOARD_KMDF_CONTROL_DEVICE_EXTENSION controlExtension  = NULL;
   669	      PDEVICE_CONTEXT                                 devContext        = NULL;
   670	      WDFREQUEST                                      requestFromAbove;
   671	
   672	      if(NULL == chatpadKeyboardKMDFControlDevice)
   673	      {
   674	         ChatpadTrace(("Error, control device handle is NULL.\n"));
   675	         apiStatus = STATUS_INTERNAL_ERROR;
   676	      }
   677	
   678	      if(STATUS_SUCCESS == apiStatus)
   679	      {
   680	         controlExtension = ChatpadKeyboardKMDFControlGetData(chatpadKeyboardKMDFControlDevice);
   681	
   682	         if(NULL == controlExtension)
   683	         {
   684	            ChatpadTrace(("Failed to get control device extension in close handler.\n"));
   685	            apiStatus = STATUS_INTERNAL_ERROR;
   686	         }
   687	      }
   688	
   689	      if(STATUS_SUCCESS == apiStatus)
   690	      {
   691	         devContext = controlExtension->keyboardDeviceContext;
   692	         if(NULL == devContext)
   693	         {
   694	            ChatpadTrace(("Failed to get device context in close handler.\n"));
   695	            apiStatus = STATUS_INTERNAL_ERROR;
   696	         }
   697	      }
   698	
   699	      if(STATUS_SUCCESS == apiStatus)
   700	      {
   701	         // Mark the virtual keyboard as disabled when the application disconnects.
   702	         // TODO could messing with the device context here cause synchronization issues?
   703	         devContext->keyboardEnabled = FALSE;
   704	         ChatpadTrace(("Virtual keyboard disabled on application disconnect.\n"));
   705	
   706	         apiStatus = WdfIoQueueRetrieveNextRequest(
   707	           devContext->hidReadRequestQueue,
   708	           &requestFromAbove);
   709	         if(STATUS_SUCCESS != apiStatus)
   710	         {
   711	            ChatpadTrace(("No top-level HID read report request available.\n"));
   712	            apiStatus = STATUS_INTERNAL_ERROR;
   713	         }
   714	      }
   715	
   716	      if(STATUS_SUCCESS == apiStatus)
   717	      {
   718	         PCHATPAD_KEYBOARD_HID_INPUT_REPORT  inputReport                   = NULL;
   719	         ULONG                               bytesTransferred              = 0;
   720	         size_t                              bytesToCopy                   = 0;
   721	         size_t                              requestFromAboveBufferLength  = 0;
   722	
   723	         bytesToCopy = sizeof(CHATPAD_KEYBOARD_HID_INPUT_REPORT);
   724	         apiStatus = WdfRequestRetrieveOutputBuffer(
   725	           requestFromAbove,
   726	           bytesToCopy,
   727	           reinterpret_cast<PVOID*>(&inputReport),
   728	           &requestFromAboveBufferLength);
   729	         if((STATUS_SUCCESS == apiStatus)                   &&
   730	            (NULL != inputReport)                           &&
   731	            (requestFromAboveBufferLength >= bytesToCopy))
   732	         {
   733	            // Fill in the report fields to indicate that no keys are pressed, so the virtual keyboard
   734	            // does not end up in a state where keys are repeating when the control application disconnects.
   735	            inputReport->modifierByte  = 0x00;
   736	            inputReport->reservedByte  = 0x00;
   737	            inputReport->keyCodes[0]   = 0x00;
   738	            inputReport->keyCodes[1]   = 0x00;
   739	            inputReport->keyCodes[2]   = 0x00;
   740	            inputReport->keyCodes[3]   = 0x00;
   741	            inputReport->keyCodes[4]   = 0x00;
   742	            inputReport->keyCodes[5]   = 0x00;
   743	
   744	            // Indicate to the requester how many bytes were transferred.
   745	            bytesTransferred = bytesToCopy;
   746	         }
   747	         else
   748	         {
   749	            ChatpadTrace(("Unable to get valid buffer from top-level HID read report request:  0x%x\n",
   750	                          apiStatus));
   751	            apiStatus = STATUS_INTERNAL_ERROR;
   752	         }
   753	
   754	         // Return the report to the requester.
   755	         WdfRequestCompleteWithInformation(requestFromAbove, apiStatus, bytesTransferred);
   756	
   757	/*
   758	         if(STATUS_SUCCESS == apiStatus)
   759	         {
   760	            ChatpadTrace(("Successfully sent HID response indicating all keys are released.\n"));
   761	         }
   762	*/
   763	      } // end if(STATUS_SUCCESS == apiStatus)
   764	   } // end if(0 == numOpenHandles)
   765	
   766	   WdfWaitLockRelease(chatpadKeyboardKMDFDeviceCollectionLock);
   767	} // end ChatpadKeyboardKMDFControlDeviceCloseHandler

   779	VOID
   780	ChatpadKeyboardKMDFContextCleanup(
   781	  IN WDFOBJECT       object)
   782	{
   783	   NTSTATUS apiStatus = STATUS_SUCCESS;
   784	   // TODO is static_cast or reinterpret_cast preferred here?  I'm doing an
   785	   //      oldstyle cast because that's what the suggestion on osronline
   786	   //      said to do.  :)
   787	   //      Normally, for void* to something* I'd do static_cast, and from
   788	   //      thingA* to thingB* I'd do reinterpret_cast.
   789	   WDFDEVICE device = (WDFDEVICE) object;
   790	   PDEVICE_CONTEXT devContext = NULL;
   791	   ULONG deviceCount;
   792	
   793	   ChatpadTrace(("Normal context cleanup handler called.\n"));
   794	
   795	   devContext = ChatpadKeyboardKMDFGetDeviceContext(device);
   796	   if((NULL != device) && (NULL != devContext))
   797	   {
   798	      WdfWaitLockAcquire(chatpadKeyboardKMDFDeviceCollectionLock, NULL);
   799	
   800	      // Clean up device context resources.
   801	
   802	      deviceCount = WdfCollectionGetCount(chatpadKeyboardKMDFDeviceCollection);
   803	      if(deviceCount == 1)
   804	      {
   805	         // If we are the last instance, then delete the control device so that
   806	         // the driver can unload.  Apparently there was a bug in WDF 1.0, but 
   807	         // apparently this sort of thing is fine in WDF 1.9 (and even earlier?).
   808	         //
   809	         // According to the toaster KMDF sideband filter example, the deletion
   810	         // below has to be done while the lock on the collection is acquired
   811	         // since it protects the global device variable, and another thread
   812	         // cannot try to create while we are deleting the device.
   813	
   814	         if(0 == numOpenHandles)
   815	         {
   816	            ChatpadKeyboardKMDFDeleteControlDevice();
   817	         }
   818	         else
   819	         {
   820	            ChatpadTrace(("The device was not deleted since there are still %lu open handles to it.\n",
   821	                          numOpenHandles));
   822	         }
   823	      } // end if(deviceCount == 1)
   824	
   825	      // Even if the control device was not deleted, however, we mark it as unavailable.
   826	      chatpadKeyboardKMDFControlDeviceAvailable = FALSE;
   827	
   828	      WdfCollectionRemove(chatpadKeyboardKMDFDeviceCollection, device);
   829	
   830	      WdfWaitLockRelease(chatpadKeyboardKMDFDeviceCollectionLock);
   831	   } // end if((NULL != device) && (NULL != devContext))
   832	   else
   833	   {
   834	      // This should never happen.
   835	      ChatpadTrace(("Error, device and/or devContext are NULL while cleaning up.\n"));
   836	   }
   837	}

   839	// Creates a control device object so that an application can talk to the filter driver.
   840	NTSTATUS
   841	ChatpadKeyboardKMDFCreateControlDevice(
   842	  WDFDEVICE device)
   843	{
   844	   NTSTATUS                retStatus = STATUS_SUCCESS;
   845	   BOOLEAN                 createDevice = FALSE;
   846	   PWDFDEVICE_INIT         deviceInit = NULL;
   847	   WDF_FILEOBJECT_CONFIG   fileObjectConfig;
   848	   WDFDEVICE               newChatpadKeyboardKMDFControlDevice = NULL;
   849	   WDF_OBJECT_ATTRIBUTES   controlAttributes;
   850	   DECLARE_CONST_UNICODE_STRING(ntDeviceName, CHATPAD_KEYBOARD_KMDF_CONTROL_NAME);
   851	   DECLARE_CONST_UNICODE_STRING(symbolicLinkName, CHATPAD_KEYBOARD_KMDF_CONTROL_LINK);
   852	   WDFQUEUE                newQueue;
   853	   WDF_IO_QUEUE_CONFIG     ioQueueConfig;
   854	
   855	   // Determine whether we should create the device, or if it already exists.
   856	   WdfWaitLockAcquire(chatpadKeyboardKMDFDeviceCollectionLock, NULL);
   857	
   858	   // TODO REMOVE after testing
   859	   ChatpadTrace(("Collection device count is %d.\n", WdfCollectionGetCount(chatpadKeyboardKMDFDeviceCollection)));
   860	   // TODO is this a safe way to synchronize things?
   861	   if(NULL == chatpadKeyboardKMDFControlDevice)
   862	   {
   863	      createDevice = TRUE;
   864	   }
   865	
   866	   WdfWaitLockRelease(chatpadKeyboardKMDFDeviceCollectionLock);
   867	
   868	   if(!createDevice)
   869	   {
   870	      ChatpadTrace(("Chatpad keyboard KMDF control device was already created.\n"));
   871	   }
   872	   else
   873	   {
   874	      ChatpadTrace(("Creating chatpad keyboard KMDF control device.\n"));
   875	
   876	      // Allocate and initialize a WDFDEVICE_INIT structure for the control device.
   877	      deviceInit = WdfControlDeviceInitAllocate(
   878	        WdfDeviceGetDriver(device),
   879	        // Permissions:  Allow kernel, system, and admin control over the device,
   880	        //   but do not allow any other users to access it.
   881	        // TODO does this work with XP?
   882	        &SDDL_DEVOBJ_SYS_ALL_ADM_ALL);
   883	      if(NULL == deviceInit)
   884	      {
   885	         retStatus = STATUS_INSUFFICIENT_RESOURCES;
   886	      }
   887	
   888	      if(STATUS_SUCCESS == retStatus)
   889	      {
   890	         // Make the control device exclusive.  I only want one application talking to it at a time.
   891	         WdfDeviceInitSetExclusive(deviceInit, TRUE);
   892	         // Get device name.
   893	         retStatus = WdfDeviceInitAssignName(deviceInit, &ntDeviceName);
   894	      }
   895	
   896	      // Set up callbacks so we can keep track of whether user mode has the
   897	      // control device open.
   898	      WDF_FILEOBJECT_CONFIG_INIT(
   899	        &fileObjectConfig,
   900	        ChatpadKeyboardKMDFControlDeviceCreateHandler,
   901	        ChatpadKeyboardKMDFControlDeviceCloseHandler,
   902	        WDF_NO_EVENT_CALLBACK);
   903	
   904	      WdfDeviceInitSetFileObjectConfig(
   905	        deviceInit,
   906	        &fileObjectConfig,
   907	        WDF_NO_OBJECT_ATTRIBUTES);
   908	
   909	      if(STATUS_SUCCESS == retStatus)
   910	      {
   911	         WDF_OBJECT_ATTRIBUTES_INIT_CONTEXT_TYPE(
   912	           &controlAttributes,
   913	           CHATPAD_KEYBOARD_KMDF_CONTROL_DEVICE_EXTENSION);
   914	         // Create the control device.
   915	         retStatus = WdfDeviceCreate(
   916	           &deviceInit,
   917	           &controlAttributes,
   918	           &newChatpadKeyboardKMDFControlDevice);
   919	      }
   920	
   921	      // Initialize control extension.
   922	      PDEVICE_CONTEXT devContext = ChatpadKeyboardKMDFGetDeviceContext(device);
   923	      if(NULL != devContext)
   924	      {
   925	         PCHATPAD_KEYBOARD_KMDF_CONTROL_DEVICE_EXTENSION controlExtension =
   926	           ChatpadKeyboardKMDFControlGetData(newChatpadKeyboardKMDFControlDevice);
   927	
   928	         if(NULL != controlExtension)
   929	         {
   930	            ChatpadTrace(("Initializing control device extension.\n"));
   931	            controlExtension->keyboardDeviceContext = devContext;
   932	         }
   933	         else
   934	         {
   935	            ChatpadTrace(("Error, could not initialize control device extension.\n"));
   936	            retStatus = STATUS_INTERNAL_ERROR;
   937	         }
   938	      }
   939	      else
   940	      {
   941	         retStatus = STATUS_INTERNAL_ERROR;
   942	      }
   943	
   944	      if(STATUS_SUCCESS == retStatus)
   945	      {
   946	         // Create symbolic link so user mode can open the device.
   947	         retStatus = WdfDeviceCreateSymbolicLink(
   948	           newChatpadKeyboardKMDFControlDevice,
   949	           &symbolicLinkName);
   950	      }
   951	
   952	      if(STATUS_SUCCESS == retStatus)
   953	      {
   954	         // Configure default queue so that requests to the control device will be parallel.
   955	         WDF_IO_QUEUE_CONFIG_INIT_DEFAULT_QUEUE(
   956	           &ioQueueConfig,
   957	//           WdfIoQueueDispatchSequential);
   958	           WdfIoQueueDispatchParallel);
   959	
   960	         // Select what we want from the queue.
   961	
   962	         // This should catch application-command IOCTLs.
   963	         ioQueueConfig.EvtIoDeviceControl =
   964	           ChatpadKeyboardKMDFIOCTLHandler;
   965	
   966	         // Create the queue.
   967	         retStatus = WdfIoQueueCreate(
   968	           newChatpadKeyboardKMDFControlDevice,
   969	           &ioQueueConfig,
   970	           WDF_NO_OBJECT_ATTRIBUTES, // TODO what does this parameter do?
   971	           &newQueue);
   972	      } // end if(STATUS_SUCCESS == retStatus)
   973	
   974	      if(STATUS_SUCCESS == retStatus)
   975	      {
   976	         WdfControlFinishInitializing(newChatpadKeyboardKMDFControlDevice);
   977	         WdfWaitLockAcquire(chatpadKeyboardKMDFDeviceCollectionLock, NULL);
   978	         chatpadKeyboardKMDFControlDevice = newChatpadKeyboardKMDFControlDevice;
   979	         WdfWaitLockRelease(chatpadKeyboardKMDFDeviceCollectionLock);
   980	
   981	         ChatpadTrace(("Chatpad keyboard KMDF control device created successfully.\n"));
   982	      }
   983	      else
   984	      {
   985	         ChatpadTrace(("WdfIoQueueCreate failed:  0x%x\n", retStatus));
   986	      }
   987	   } // end if creating the control device
   988	
   989	   // Handle error cases.
   990	   if(STATUS_SUCCESS != retStatus)
   991	   {
   992	      ChatpadTrace(("Failed to create chatpad keyboard KMDF control device:  0x%x\n", retStatus));
   993	
   994	      if(NULL != deviceInit)
   995	      {
   996	         WdfDeviceInitFree(deviceInit);
   997	         deviceInit = NULL;
   998	      }
   999	
  1000	      if(NULL != newChatpadKeyboardKMDFControlDevice)
  1001	      {
  1002	         // Release the newly created, but uninitialized, object.
  1003	         WdfObjectDelete(newChatpadKeyboardKMDFControlDevice);
  1004	         newChatpadKeyboardKMDFControlDevice = NULL;
  1005	      }
  1006	   }
  1007	
  1008	   return retStatus;
  1009	} // end ChatpadKeyboardKMDFCreateControlDevice

  1027	VOID
  1028	ChatpadKeyboardKMDFInternalIOCTLHandler(
  1029	  IN WDFQUEUE     queue,
  1030	  IN WDFREQUEST   request,
  1031	  IN size_t       outputBufferLength,
  1032	  IN size_t       inputBufferLength,
  1033	  IN ULONG        ioControlCode)
  1034	{
  1035	   NTSTATUS                   apiStatus               = STATUS_SUCCESS;
  1036	   WDFDEVICE                  wdfDevice;
  1037	   PDEVICE_CONTEXT            devContext              = NULL;
  1038	   // Whether this handler should complete the request.
  1039	   BOOLEAN                    completeRequestNow      = TRUE;
  1040	
  1041	//   ChatpadTrace(("Chatpad keyboard KMDF layer internal IOCTL handler called with ioControlCode == %x\n",
  1042	//                 ioControlCode));
  1043	
  1044	   wdfDevice = WdfIoQueueGetDevice(queue);
  1045	
  1046	   // Get device context so we know where to forward the request.
  1047	   devContext = ChatpadKeyboardKMDFGetDeviceContext(wdfDevice);
  1048	
  1049	// TODO what control codes do I need to support for a keyboard?
  1050	   if(IOCTL_HID_GET_DEVICE_DESCRIPTOR == ioControlCode)
  1051	   {
  1052	      apiStatus = ChatpadKeyboardKMDFGetHidDescriptor(wdfDevice, request);
  1053	   }
  1054	   else if(IOCTL_HID_GET_DEVICE_ATTRIBUTES == ioControlCode)
  1055	   {
  1056	// TODO NEXT do I need to support this?  If so I guess I have to provide placeholder vendor/product IDs.
  1057	      apiStatus = ChatpadKeyboardKMDFGetHidDeviceAttributes(request);
  1058	   }
  1059	   else if(IOCTL_HID_GET_REPORT_DESCRIPTOR == ioControlCode)
  1060	   {
  1061	      apiStatus = ChatpadKeyboardKMDFGetHidReportDescriptor(wdfDevice, request);
  1062	   }
  1063	   else if(IOCTL_HID_GET_STRING == ioControlCode)
  1064	   {
  1065	      apiStatus = ChatpadKeyboardKMDFGetHidString(request);
  1066	   }
  1067	   else if(IOCTL_HID_SET_FEATURE == ioControlCode)
  1068	   {
  1069	ChatpadTrace(("Received HID set feature request.\n"));
  1070	// TODO NEXT do I need to support this request?
  1071	apiStatus = STATUS_NOT_SUPPORTED; // TODO NEXT REMOVE
  1072	   }
  1073	   else if(IOCTL_HID_GET_FEATURE == ioControlCode)
  1074	   {
  1075	ChatpadTrace(("Received HID get feature request.\n"));
  1076	// TODO NEXT do I need to support this request?
  1077	apiStatus = STATUS_NOT_SUPPORTED; // TODO NEXT REMOVE
  1078	   }
  1079	   else if(IOCTL_HID_WRITE_REPORT == ioControlCode)
  1080	   {
  1081	ChatpadTrace(("Received HID write report request.\n"));
  1082	// TODO NEXT do I need to support this request?  Apparently so, if I want to track the system caps lock state.
  1083	apiStatus = STATUS_NOT_SUPPORTED; // TODO NEXT REMOVE
  1084	   }
  1085	   else if(IOCTL_HID_READ_REPORT == ioControlCode)
  1086	   {
  1087	      // Queue the request to a manual queue so that it can be handled when we
  1088	      // receive data from the control device.
  1089	      // TODO NEXT do I need to handle IOCTL_HID_GET_INPUT_REPORT as well?
  1090	      //   If only handling IOCTL_HID_READ_REPORT works on XP, Vista, and Windows 7, then apparently I don't.

  1282	// This function retrieves a keyboard HID string.

  1287	{
  1288	   NTSTATUS                retStatus                  = STATUS_SUCCESS;
  1289	   PWSTR                   hidString                  = NULL;

  1293	   WDF_REQUEST_PARAMETERS  requestParameters;

  1376	// This function runs at PASSIVE IRQL.
  1377	VOID
  1378	ChatpadKeyboardKMDFIOCTLHandler(
  1379	  IN WDFQUEUE     queue,
  1380	  IN WDFREQUEST   request,
  1381	  IN size_t       outputBufferLength,
  1382	  IN size_t       inputBufferLength,
  1383	  IN ULONG        ioControlCode)
  1384	{
  1385	   NTSTATUS                   apiStatus                           = STATUS_SUCCESS;
  1386	   ULONG                      numDriverInstances                  = 0;
  1387	   // Device context for a particular filter driver instance.
  1388	   PDEVICE_CONTEXT            devContext                          = NULL;
  1389	   PVOID                      dataFromUserAsVoid                  = NULL;
  1390	   PUCHAR                     dataFromUser                        = NULL;
  1391	   // The following pointer can point at either a receive buffer or a send buffer.
  1392	   ULONG                      bytesToCopy                         = 0;
  1393	   ULONG                      bytesTransferred                    = 0;
  1394	   BOOLEAN                    immediatelyFailRequest              = FALSE;
  1395	   WDFREQUEST                 requestFromAbove;
  1396	
  1397	//   ChatpadTrace(("The IRQL in the chatpad keyboard KMDF IOCTL handler is %u.\n", KeGetCurrentIrql()));
  1398	
  1399	   WdfWaitLockAcquire(chatpadKeyboardKMDFDeviceCollectionLock, NULL);
  1400	
  1401	   if(TRUE == chatpadKeyboardKMDFControlDeviceAvailable)
  1402	   {
  1403	      // Get the number of filter driver instances.
  1404	      numDriverInstances = WdfCollectionGetCount(chatpadKeyboardKMDFDeviceCollection);
  1405	      // Get device context.
  1406	      if(numDriverInstances >= 1)
  1407	      {
  1408	         devContext = ChatpadKeyboardKMDFGetDeviceContext(WdfCollectionGetItem(chatpadKeyboardKMDFDeviceCollection, 0));
  1409	      }
  1410	   }
  1411	   else
  1412	   {
  1413	      // If there is no control device available, then we want to return immediately to avoid
  1414	      // crashing.  This situation can happen if a user program has the driver open
  1415	      // and the device is suddenly unplugged.
  1416	      immediatelyFailRequest = TRUE;
  1417	   }
  1418	
  1419	   WdfWaitLockRelease(chatpadKeyboardKMDFDeviceCollectionLock);
  1420	
  1421	   if(TRUE == immediatelyFailRequest)
  1422	   {
  1423	      ChatpadTrace(("Ignoring control device IOCTL since the device is unavailable.\n"));
  1424	      WdfRequestCompleteWithInformation(request, STATUS_INTERNAL_ERROR, 0);
  1425	      return;
  1426	   }
  1427	
  1428	   if(IOCTL_CHATPAD_KEYBOARD_KMDF_SET_ENABLED == ioControlCode)
  1429	   {
  1430	      // TODO add constant for how long the buffer should be
  1431	      if(inputBufferLength != 1)
  1432	      {
  1433	         ChatpadTrace(("Error, input buffer length is %u.\n", inputBufferLength));
  1434	         apiStatus = STATUS_INTERNAL_ERROR;
  1435	      }
  1436	
  1437	      if(STATUS_SUCCESS == apiStatus)
  1438	      {
  1439	         // Get input buffer data.
  1440	         apiStatus = WdfRequestRetrieveInputBuffer(request, inputBufferLength, &dataFromUserAsVoid, NULL);
  1441	         // This ugly cast is probably avoidable, but it works for now.
  1442	         dataFromUser = static_cast<PUCHAR>(dataFromUserAsVoid);
  1443	         if((STATUS_SUCCESS != apiStatus) || (NULL == dataFromUser))
  1444	         {
  1445	            ChatpadTrace(("WdfRequestRetrieveInputBuffer failed:  0x%x\n", apiStatus));
  1446	         }
  1447	      }
  1448	
  1449	      if(STATUS_SUCCESS == apiStatus)
  1450	      {
  1451	         if(CHATPAD_KEYBOARD_KMDF_ENABLE == dataFromUser[0])
  1452	         {
  1453	            devContext->keyboardEnabled = TRUE;
  1454	            ChatpadTrace(("Virtual keyboard enabled due to IOCTL command.\n"));
  1455	         }
  1456	         else if(CHATPAD_KEYBOARD_KMDF_DISABLE == dataFromUser[0])
  1457	         {
  1458	            devContext->keyboardEnabled = FALSE;
  1459	            ChatpadTrace(("Virtual keyboard disabled due to IOCTL command.\n"));
  1460	         }
  1461	         else
  1462	         {
  1463	            ChatpadTrace(("Unknown keyboard enable code %d.\n", dataFromUser[0]));
  1464	         }
  1465	      } // end if(STATUS_SUCCESS == apiStatus)
  1466	   } // end if(IOCTL_CHATPAD_KEYBOARD_KMDF_SET_ENABLED == ioControlCode)
  1467	   else if(IOCTL_CHATPAD_KEYBOARD_KMDF_SEND_DATA == ioControlCode)
  1468	   {
  1469	      // TODO add constant for how long the buffer should be at minimum
  1470	      if(inputBufferLength < 7)
  1471	      {
  1472	         ChatpadTrace(("Error, input buffer length is %u.\n", inputBufferLength));
  1473	         apiStatus = STATUS_INTERNAL_ERROR;
  1474	      }
  1475	
  1476	      if(STATUS_SUCCESS == apiStatus)
  1477	      {
  1478	         // Get input buffer data.
  1479	         apiStatus = WdfRequestRetrieveInputBuffer(request, inputBufferLength, &dataFromUserAsVoid, NULL);
  1480	         // This ugly cast is probably avoidable, but it works for now.
  1481	         dataFromUser = static_cast<PUCHAR>(dataFromUserAsVoid);
  1482	         if((STATUS_SUCCESS != apiStatus) || (NULL == dataFromUser))
  1483	         {
  1484	            ChatpadTrace(("WdfRequestRetrieveInputBuffer failed:  0x%x\n", apiStatus));
  1485	         }
  1486	      }
  1487	
  1488	      // Do not send data out from the virtual keyboard device if the device is disabled.
  1489	      if((STATUS_SUCCESS == apiStatus)             &&
  1490	         (TRUE == devContext->keyboardEnabled))
  1491	      {
  1492	/*
  1493	         ChatpadTrace((
  1494	           "Received bytes 0x%02x, 0x%02x, 0x%02x, 0x%02x, 0x%02x, 0x%02x, 0x%02x\n",
  1495	           dataFromUser[0],
  1496	           dataFromUser[1],
  1497	           dataFromUser[2],
  1498	           dataFromUser[3],
  1499	           dataFromUser[4],
  1500	           dataFromUser[5],
  1501	           dataFromUser[6]));
  1502	*/
  1503	
  1504	         apiStatus = WdfIoQueueRetrieveNextRequest(
  1505	           devContext->hidReadRequestQueue,
  1506	           &requestFromAbove);
  1507	
  1508	         if(STATUS_SUCCESS == apiStatus)
  1509	         {
  1510	            PCHATPAD_KEYBOARD_HID_INPUT_REPORT  inputReport                   = NULL;
  1511	            size_t                              requestFromAboveBufferLength  = 0;
  1512	
  1513	            bytesToCopy = sizeof(CHATPAD_KEYBOARD_HID_INPUT_REPORT);
  1514	            apiStatus = WdfRequestRetrieveOutputBuffer(
  1515	              requestFromAbove,
  1516	              bytesToCopy,
  1517	              reinterpret_cast<PVOID*>(&inputReport),
  1518	              &requestFromAboveBufferLength);
  1519	            if((STATUS_SUCCESS == apiStatus)                   &&
  1520	               (NULL != inputReport)                           &&
  1521	               (requestFromAboveBufferLength >= bytesToCopy))
  1522	            {
  1523	               // Fill in the report fields.
  1524	               inputReport->modifierByte  = dataFromUser[0];
  1525	               inputReport->reservedByte  = 0x00;
  1526	               // See section 10 in the HID usage table spec for the USB key code reference.
  1527	               inputReport->keyCodes[0]   = dataFromUser[1];
  1528	               inputReport->keyCodes[1]   = dataFromUser[2];
  1529	               inputReport->keyCodes[2]   = dataFromUser[3];
  1530	               inputReport->keyCodes[3]   = dataFromUser[4];
  1531	               inputReport->keyCodes[4]   = dataFromUser[5];
  1532	               inputReport->keyCodes[5]   = dataFromUser[6];
  1533	
  1534	               // Indicate to the requester how many bytes were transferred.
  1535	               bytesTransferred = bytesToCopy;
  1536	            }
  1537	            else
  1538	            {
  1539	               ChatpadTrace(("Unable to get valid buffer from top-level HID read report request:  0x%x\n",
  1540	                             apiStatus));
  1541	               apiStatus = STATUS_INTERNAL_ERROR;
  1542	            }
  1543	
  1544	            // Return the report to the requester.
  1545	            WdfRequestCompleteWithInformation(requestFromAbove, apiStatus, bytesTransferred);
  1546	         } // end if(STATUS_SUCCESS == apiStatus)
  1547	         else
  1548	         {
  1549	            ChatpadTrace(("No top-level HID read report request available.\n"));
  1550	            apiStatus = STATUS_INTERNAL_ERROR;
  1551	         }
  1552	      } // end if successful and the virtual keyboard device is enabled
  1553	      else if((STATUS_SUCCESS == apiStatus) &&
  1554	              (FALSE == devContext->keyboardEnabled))
  1555	      {
  1556	         ChatpadTrace(("Ignoring keyboard data since virtual keyboard device is disabled.\n"));
  1557	      }
  1558	   } // end if(IOCTL_CHATPAD_KEYBOARD_KMDF_SEND_DATA == ioControlCode)
  1559	   else
  1560	   {
  1561	      ChatpadTrace(("Unknown ioctl control code 0x%x.\n", ioControlCode));
  1562	      // TODO does this status make the driver/device die, or just return an error
  1563	      //   to DeviceIoControl?
  1564	      apiStatus = STATUS_INTERNAL_ERROR;
  1565	   }
  1566	
  1567	//   ChatpadTrace(("Completing ChatpadKeyboardKMDFIOCTLHandler with bytesTransferred == %lu\n", bytesTransferred));
  1568	
  1569	   // Completing with the length as the information here seems to work, but I cannot seem to find
  1570	   // in the API or an example where that is specified as actually always happening for IOCTLs.
  1571	   // TODO find that information at some point?
  1572	   WdfRequestCompleteWithInformation(request, apiStatus, bytesTransferred);
  1573	
  1574	   return;
  1575	} // end ChatpadKeyboardKMDFIOCTLHandler
