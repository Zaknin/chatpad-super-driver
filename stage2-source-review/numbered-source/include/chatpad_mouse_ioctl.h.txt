     9	#ifndef CHATPAD_MOUSE_IOCTL_H
    10	#define CHATPAD_MOUSE_IOCTL_H
    11	
    12	
    13	// Constants that can be used with IOCTL_CHATPAD_MOUSE_KMDF_SET_ENABLED.
    14	#define CHATPAD_MOUSE_KMDF_DISABLE        0
    15	#define CHATPAD_MOUSE_KMDF_ENABLE         1
    16	
    17	
    18	// Arbitrary function code that should be okay for non-Microsoft use.
    19	// (see http://msdn.microsoft.com/en-us/library/ms902086.aspx).
    20	#define FILE_DEVICE_CHATPAD_MOUSE_FILTER  41001
    21	
    22	// IOCTL codes.  The range can be from 2048-4095 inclusive.
    23	
    24	// This IOCTL enables or disables the virtual mouse device.
    25	#define IOCTL_CHATPAD_MOUSE_KMDF_SET_ENABLED    CTL_CODE(FILE_DEVICE_CHATPAD_MOUSE_FILTER,   \
    26	                                                            3000,                         \
    27	                                                            METHOD_BUFFERED,              \
    28	                                                            FILE_READ_DATA | FILE_WRITE_DATA)
    29	
    30	// This IOCTL sends data to the virtual mouse device.  This can cause a mouse click to be simulated, for instance.
    31	#define IOCTL_CHATPAD_MOUSE_KMDF_SEND_DATA      CTL_CODE(FILE_DEVICE_CHATPAD_MOUSE_FILTER,   \
    32	                                                            3001,                         \
    33	                                                            METHOD_BUFFERED,              \
    34	                                                            FILE_READ_DATA | FILE_WRITE_DATA)
    35	
    36	#endif // CHATPAD_MOUSE_IOCTL_H
