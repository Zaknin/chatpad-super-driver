    42	// Structure used for a global string table to map strings like
    43	// "CHATPAD_KEY_A" to the corresponding numerical constants.
    44	typedef struct ChatpadDriverConfigStringMapping_t
    45	{
    46	   // Null-terminated string.
    47	   char*             pString;
    48	   // Corresponding numerical constant.
    49	   unsigned int      uConstant;
    50	   // Config string type, i.e. chatpad key, simulated mouse click, etc.
    51	   ConfigStringType  stringType;
    52	} ChatpadDriverConfigStringMapping;

    57	
    58	// Structure to specify a bound action, such as a simulated key press.
    59	typedef struct ChatpadDriverActionBinding_t
    60	{
    61	   // Modifiers to the source action (i.e. shift, or orange).  If there is more
    62	   // than one modifier, they are combined with binary OR.
    63	   unsigned int   uSourceActionModifiers;
    64	
    65	   // Type of binding (simulated key, simulated mouse click, etc.).
    66	   BindingType    bindingType;
    67	   // Modifiers to the bound action (i.e. shift, or alt).  If there is more
    68	   // than one modifier, they are combined with binary OR.
    69	   unsigned int   uActionModifiers;
    70	   // Numerical action code, i.e. KEYBOARD_KEY_B.
    71	   unsigned int   uActionCode;
    72	} ChatpadDriverActionBinding;
    73	
    74	// Structure to contain information on the mapping(s) for a particular chatpad key.
    75	typedef struct ChatpadKeyMapping_t
    76	{
    77	   // Number of bindings that are applied to this key.  If nNumBindings is 0,
    78	   // then pressing this key will have no effect.
    79	   // TODO NEXT test to make sure keys have no effect if not bound
    80	   int                           nNumBindings;
    81	   // Dynamically allocated array of length nNumBindings containing the action
    82	   // bindings for the key.  This can be NULL if no bindings exist for the key.
    83	   ChatpadDriverActionBinding*   pBindings;
    84	} ChatpadKeyMapping;
    85	
    86	// Structure to contain information on the mapping(s) for a particular controller button.
    87	typedef struct ControllerButtonMapping_t
    88	{
    89	   // Number of bindings that are applied to this button.  If nNumBindings is 0,
    90	   // then pressing this button will have no effect.
    91	   // TODO NEXT test to make sure it has no effect if not bound...may want to reconsider
    92	   int                           nNumBindings;
    93	   // Dynamically allocated array of length nNumBindings containing the action
    94	   // bindings for the button.  This can be NULL if no bindings exist for the button.
    95	   ChatpadDriverActionBinding*   pBindings;
    96	} ControllerButtonMapping;
    97	
    98	// Structure to contain driver config data such as key mappings.
    99	typedef struct ChatpadDriverConfigStructure_t
   100	{
   101	   // Whether Windows Mode is enabled in the current configuration.
   102	   // If it is not enabled, then it cannot be activated.
   103	   bool                       bEnableWindowsMode;
   104	
   105	   // Whether the "people" chatpad key light is used to indicate that Windows
   106	   // Mode is active.
   107	   bool                       bEnableWindowsModePeopleKeyIndicator;
   108	
   109	   // Whether the caps lock state is used to control the chatpad shift
   110	   // key indicator light (if true, the left shift latched special action
   111	   // will NOT control the same light).
   112	   bool                       bCapsLockControlsShiftIndicator;
   113	
   114	   // Whether mappings of controller buttons to other controller buttons
   115	   // are allowed.  If they are allowed, the remapping will only apply
   116	   // when Windows Mode is not active.
   117	   bool                       bEnableIngameRemappedButtons;
   118	
   119	   // Virtual mouse sensitivity.  The higher this number is, the faster the
   120	   // mouse will move when the thumbsticks are moved in Windows mode.
   121	   int                        nVirtualMouseSensitivity;
   122	
   123	   // Mapping structures for each chatpad key and controller button.
   124	   // The keyMappings array is indexed by CHATPAD_KEY_ constants, and
   125	   // the buttonMappings array is indexed by the CONTROLLER_BUTTON_
   126	   // constants.
   127	   ChatpadKeyMapping          keyMappings[CHATPAD_NUM_KEYS];
   128	   ControllerButtonMapping    buttonMappings[CONTROLLER_NUM_BUTTONS];
   129	} ChatpadDriverConfigStructure;

   139	// Loads data from the configuration file with the given filename.
   140	//
   141	// This function opens the file, initializes the structure pointed to by
   142	// pConfigStructure to a default state, parses the configuration data from the
   143	// config file, writes it into the structure pointed to by pConfigStructure,
   144	// and closes the file.  If pFilename is NULL, pConfigStructure is NULL, a
   145	// syntax error is encountered, or another problem occurs, then this function
   146	// fails.
   147	//
   148	// Return SUCCESS on success and FAILURE on failure.
   149	int LoadConfigFile(
   150	  char*                          pFilename,
   151	  ChatpadDriverConfigStructure*  pConfigStructure);
