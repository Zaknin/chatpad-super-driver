   445	// Logs a message.  This could be made configurable and made to log messages
   446	// to something other than console output.
   447	void LogMessageAdvanced(bool bAddNewline, char *pMessageString, ...)
   448	{
   449	   va_list formatList;
   450	
   451	   va_start(formatList, pMessageString);
   452	   vfprintf(stdout, pMessageString, formatList);
   453	   if(true == bAddNewline)
   454	   {
   455	      fprintf(stdout, "\n");
   456	   }
   457	   va_end(formatList);
   458	   fflush(stdout);
   459	} // end LogMessageAdvanced

   461	// Opens the controller device.  For now, this will probably only work with a single chatpad filter driver on a
   462	// system, i.e. it will probably always try to open the first one.
   463	int OpenController(PHANDLE pChatpadHandle)
   464	{
   465	   int nRetValue = SUCCESS;
   466	   BOOL bAPIResult = TRUE;
   467	
   468	   if((SUCCESS == nRetValue) &&
   469	      (NULL == pChatpadHandle))
   470	   {
   471	      LogMessage("Invalid pointer provided to OpenController.\n");
   472	      nRetValue = FAILURE;
   473	   }
   474	
   475	   if(SUCCESS == nRetValue)
   476	   {
   477	      LogMessage("Opening chatpad filter driver.\n");
   478	      // Open the filter device, using the path \\.\ChatpadFilter
   479	      // TODO do a little research into how such paths work, and/or look a little more closely at the CreateFile API.
   480	      (*pChatpadHandle) = CreateFile(
   481	        "\\\\.\\ChatpadFilter",
   482	        GENERIC_READ | GENERIC_WRITE,
   483	        0,
   484	        NULL,
   485	        OPEN_EXISTING,
   486	        FILE_FLAG_OVERLAPPED,
   487	        NULL);
   488	      if(INVALID_HANDLE_VALUE != (*pChatpadHandle))
   489	      {
   490	         LogMessage("Successfully opened chatpad filter driver.\n");
   491	      }
   492	      else
   493	      {
   494	         LogMessage("Couldn't open chatpad filter:  %lu\n", GetLastError());
   495	         nRetValue = FAILURE;
   496	      }
   497	   } // end if(SUCCESS == nRetValue)
   498	
   499	   return nRetValue;
   500	} // end OpenController

   584	// Executes a control IOCTL, using hControlRequestMutex for synchronization and hControlRequestEvent 
   585	// to detect when the IOCTL has finished executing.  This function returns when the IOCTL is completed.
   586	// Due to the hControlRequestMutex synchronization, it should be safe to call this function from
   587	// multiple threads.
   588	//
   589	// This function returns SUCCESS on success and FAILURE on failure.
   590	int ExecuteControlIOCTL(
   591	  HANDLE    hChatpadDevice,
   592	  DWORD     ioctlCode,
   593	  PVOID     dataToDevice,
   594	  DWORD     bytesToSend,
   595	  PVOID     pReceiveBuffer,
   596	  DWORD     receiveBufferLength,
   597	  PDWORD    pBytesTransferred)
   598	{
   599	   int         nRetValue            = SUCCESS;
   600	   DWORD       waitResult           = 0;
   601	   OVERLAPPED  overlappedStructure;
   602	
   603	   // Get the control request mutex to ensure that only one control request is ongoing at a time,
   604	   // and that two threads do not try to use the event at the same time.
   605	   waitResult = WaitForSingleObject(hControlRequestMutex, INFINITE);
   606	   // Make sure the wait actually succeeded.
   607	   if(WAIT_OBJECT_0 != waitResult)
   608	   {
   609	      LogMessage("Wait for mutex failed:  %u\n", waitResult);
   610	      nRetValue = FAILURE;
   611	   }
   612	
   613	   if((SUCCESS == nRetValue) &&
   614	      (INVALID_HANDLE_VALUE == hChatpadDevice))
   615	   {
   616	      LogMessage("Invalid handle provided to ExecuteControlIOCTL.\n");
   617	      nRetValue = FAILURE;
   618	   }
   619	
   620	   if((SUCCESS == nRetValue) && 
   621	      (NULL == pBytesTransferred))
   622	   {
   623	      LogMessage("Invalid bytes transferred pointer provided to ExecuteControlIOCTL.\n");
   624	      nRetValue = FAILURE;
   625	   }
   626	
   627	   if((SUCCESS == nRetValue) &&
   628	      (NULL == hControlRequestEvent))
   629	   {
   630	      LogMessage("Error, control request event handle is NULL.\n");
   631	      nRetValue = FAILURE;
   632	   }
   633	
   634	   if(SUCCESS == nRetValue)
   635	   {
   636	      DWORD lastError            = 0;
   637	      BOOL  operationComplete    = FALSE;
   638	
   639	      // Initialize the OVERLAPPED structure.
   640	      memset(&overlappedStructure, 0x00, sizeof(OVERLAPPED));
   641	      overlappedStructure.hEvent = hControlRequestEvent;
   642	
   643	      // Call the IOCTL to submit the control transfer through the filter driver.
   644	      operationComplete = DeviceIoControl(
   645	        hChatpadDevice,
   646	        ioctlCode,
   647	        dataToDevice,
   648	        bytesToSend,
   649	        pReceiveBuffer,
   650	        receiveBufferLength,
   651	        pBytesTransferred,
   652	        &overlappedStructure);
   653	
   654	      if(FALSE == operationComplete)
   655	      {
   656	         // If the request failed, it is hopefully pending.
   657	         lastError = GetLastError();
   658	
   659	         if(ERROR_IO_PENDING == lastError)
   660	         {
   661	            // Wait for the request to finish or the thread shutdown signal to be sent.
   662	            DWORD ioctlWaitResult = 0;
   663	            HANDLE waitHandles[2];
   664	
   665	            waitHandles[0] = hControlRequestEvent;
   666	            waitHandles[1] = hWorkerThreadShutdownEvent;
   667	            ioctlWaitResult = WaitForMultipleObjects(
   668	              2,
   669	              waitHandles,
   670	              FALSE,       // Stop waiting after any of the specified events finishes.
   671	              INFINITE);   // No timeout.
   672	            if(WAIT_OBJECT_0 == ioctlWaitResult)
   673	            {
   674	               // Make sure that the IOCTL actually completed successfully.
   675	               operationComplete = GetOverlappedResult(
   676	                 hChatpadDevice,
   677	                 &overlappedStructure,
   678	                 pBytesTransferred,
   679	                 FALSE);
   680	               if(FALSE == operationComplete)
   681	               {
   682	                  lastError = GetLastError();
   683	               }
   684	            }
   685	            else if((WAIT_OBJECT_0 + 1) == ioctlWaitResult)
   686	            {
   687	               // The worker thread shutdown event was signaled.
   688	               nRetValue      = FAILURE;
   689	               lastError      = 0;
   690	               LogMessage("Thread shutdown detected while in ExecuteControlIOCTL.\n");
   691	            }
   692	            else
   693	            {
   694	               LogMessage("WaitForMultipleObjects failed with return value %u\n", ioctlWaitResult);
   695	               nRetValue      = FAILURE;
   696	               lastError      = GetLastError();
   697	            }
   698	         } // end if(ERROR_IO_PENDING == lastError)
   699	
   700	         //***
   701	         // Note that if lastError is not ERROR_IO_PENDING and the operation
   702	         // does not subsequently complete successfully, then it will be caught
   703	         // in code below.
   704	         //***
   705	
   706	      } // end if(FALSE == operationComplete)
   707	
   708	      // Note that even if operationComplete is FALSE above, it should be set to TRUE later in
   709	      // the above code after the operation actually finishes.
   710	      if(TRUE == operationComplete)
   711	      {
   712	         // For now, I am only printing this if a nonzero number of bytes were
   713	         // transferred.  Otherwise, a line would be printed for every 0x001F
   714	         // and 0x001E chatpad keep-alive control request.
   715	         if((*pBytesTransferred) > 0)
   716	         {
   717	            LogMessage("Successfully sent IOCTL.  Bytes transferred:  %lu\n", *pBytesTransferred);
   718	         }
   719	      }
   720	      else
   721	      {
   722	         LogMessage("IOCTL failed, lastError == %lu\n", lastError);
   723	         nRetValue = FAILURE;
   724	      }
   725	
   726	      // Reset the event in case it was signaled.  If it was not signaled, this
   727	      // call should simply have no effect.
   728	      ResetEvent(hControlRequestEvent);
   729	   } // end if(SUCCESS == nRetValue)
   730	
   731	   // Release the mutex if we obtained it.
   732	   if(WAIT_OBJECT_0 == waitResult)
   733	   {
   734	      if(0 == ReleaseMutex(hControlRequestMutex))
   735	      {
   736	         LogMessage("Control IOCTL mutex release failed:  %lu\n", GetLastError());
   737	         nRetValue = FAILURE;
   738	      }
   739	   }
   740	
   741	   return nRetValue;
   742	} // end ExecuteControlIOCTL

   744	// Sends a control request to the chatpad.
   745	//
   746	// To send additional data, include it in pExtraSendBuffer and specify the length of the data.
   747	// To receive data, include it in pReceiveBuffer and specify the length of the buffer.
   748	int SendControlRequest(
   749	  HANDLE hChatpadDevice,
   750	  XBOX_CONTROLLER_INTERFACE whichInterface,
   751	  unsigned char requestType,
   752	  unsigned char request,
   753	  unsigned int value,
   754	  unsigned int index,
   755	  unsigned int length,
   756	  unsigned char* pExtraSendBuffer = NULL,
   757	  unsigned int extraSendBufferLength = 0,
   758	  unsigned char* pReceiveBuffer = NULL,
   759	  unsigned int receiveBufferLength = 0)
   760	{
   761	   int         nRetValue            = SUCCESS;
   762	   DWORD       bytesToSend          = 0;
   763	   DWORD       bytesTransferred     = 0;
   764	   DWORD       dataToDeviceLength   = 0;
   765	   PUCHAR      dataToDevice         = NULL;
   766	
   767	   if((SUCCESS == nRetValue) &&
   768	      (INVALID_HANDLE_VALUE == hChatpadDevice))
   769	   {
   770	      LogMessage("Invalid handle provided to SendControlRequest.\n");
   771	      nRetValue = FAILURE;
   772	   }
   773	
   774	   if(SUCCESS == nRetValue)
   775	   {
   776	      // Allocate enough space for an interface number, the control transfer
   777	      // parameters (more space than we need), and any extra data that is
   778	      // to be sent.
   779	      dataToDeviceLength = 1 + 32 + extraSendBufferLength;
   780	      dataToDevice = static_cast<PUCHAR>(malloc(dataToDeviceLength));
   781	      if(NULL == dataToDevice)
   782	      {
   783	         LogMessage("Memory allocation failure in SendControlRequest.\n");
   784	         nRetValue = FAILURE;
   785	      }
   786	   }
   787	
   788	   if(SUCCESS == nRetValue)
   789	   {
   790	      memset(dataToDevice, 0x00, dataToDeviceLength);
   791	      if(NULL != pReceiveBuffer)
   792	      {
   793	         memset(pReceiveBuffer, 0x00, receiveBufferLength);
   794	      }
   795	
   796	      // The first byte specifies which interface to use.
   797	      // Technically, everything is being sent to the first interface right now, as
   798	      // far as I know, but it seems to work so I'm leaving it like that.
   799	      dataToDevice[bytesToSend++] = whichInterface;
   800	
   801	      // Set up the USB control transfer.
   802	      // See http://www.beyondlogic.org/usbnutshell/usb6.shtml for setup packet details.
   803	      dataToDevice[bytesToSend++] = requestType;
   804	      dataToDevice[bytesToSend++] = request;
   805	      dataToDevice[bytesToSend++] = value & 0x000000FF;          // Value low.
   806	      dataToDevice[bytesToSend++] = (value & 0x0000FF00) >> 8;   // Value high.
   807	      dataToDevice[bytesToSend++] = index & 0x000000FF;          // Index low.
   808	      dataToDevice[bytesToSend++] = (index & 0x0000FF00) >> 8;   // Index high.
   809	      dataToDevice[bytesToSend++] = length & 0x000000FF;         // Length low.
   810	      dataToDevice[bytesToSend++] = (length & 0x0000FF00) >> 8;  // Length high.
   811	
   812	      // Additional bytes contain additional data to be sent to the device.
   813	      for(unsigned int i = 0; i < extraSendBufferLength; i++)
   814	      {
   815	         dataToDevice[bytesToSend++] = pExtraSendBuffer[i];
   816	      }
   817	
   818	      // Execute the IOCTL.
   819	      nRetValue = ExecuteControlIOCTL(
   820	        hChatpadDevice,
   821	        IOCTL_CHATPAD_SEND_CONTROL_TRANSFER,
   822	        dataToDevice,
   823	        bytesToSend,
   824	        pReceiveBuffer,
   825	        receiveBufferLength,
   826	        &bytesTransferred);
   827	   } // end if(SUCCESS == nRetValue)
   828	
   829	   if(NULL != dataToDevice)
   830	   {
   831	      free(dataToDevice);
   832	      dataToDevice = NULL;
   833	   }
   834	
   835	   // Sleep at least 12 milliseconds after each request, since sending them too close together
   836	   // seems to potentially cause problems, at least for turning on individual key lights like
   837	   // the green and the orange inidicators.
   838	   Sleep(12);
   839	
   840	   return nRetValue;
   841	} // end SendControlRequest

   944	// Executes an IOCTL to send keyboard data to the virtual keyboard device.
   945	//
   946	// IMPORTANT NOTE:  This function attempts to synchronize via hStatesMutex,
   947	// so the calling thread should not hold that mutex (though it actually
   948	// might work properly to acquire it twice).
   949	//
   950	// The enableValue parameter must be CHATPAD_KEYBOARD_KMDF_ENABLE or
   951	// CHATPAD_KEYBOARD_KMDF_DISABLE.
   952	//
   953	// This function returns when the IOCTL is completed.
   954	//
   955	// This function returns SUCCESS on success and FAILURE on failure.
   956	int SetVirtualKeyboardDeviceEnabled(UCHAR enableValue)
   957	{
   958	   int         nRetValue            = SUCCESS;
   959	   DWORD       waitResult           = 0;
   960	   OVERLAPPED  overlappedStructure;
   961	
   962	   // Get the states mutex for synchronization.
   963	   waitResult = WaitForSingleObject(hStatesMutex, INFINITE);
   964	   // Make sure the wait actually succeeded.
   965	   if(WAIT_OBJECT_0 != waitResult)
   966	   {
   967	      LogMessage("Wait for mutex failed:  %u\n", waitResult);
   968	      nRetValue = FAILURE;
   969	   }
   970	
   971	   if((SUCCESS == nRetValue) &&
   972	      (INVALID_HANDLE_VALUE == hVirtualKeyboardDevice))
   973	   {
   974	      LogMessage("Invalid virtual keyboard device handle in SetVirtualKeyboardDeviceEnabled.\n");
   975	      nRetValue = FAILURE;
   976	   }
   977	
   978	   if((SUCCESS == nRetValue)                          && 
   979	      (CHATPAD_KEYBOARD_KMDF_ENABLE != enableValue)   &&
   980	      (CHATPAD_KEYBOARD_KMDF_DISABLE != enableValue))
   981	   {
   982	      LogMessage("Invalid parameter provided to SetVirtualKeyboardDeviceEnabled.\n");
   983	      nRetValue = FAILURE;
   984	   }
   985	
   986	   if((SUCCESS == nRetValue) &&
   987	      (NULL == hVirtualKeyboardSendEvent))
   988	   {
   989	      LogMessage("Error, virtual keyboard send event handle is NULL.\n");
   990	      nRetValue = FAILURE;
   991	   }
   992	
   993	   if(SUCCESS == nRetValue)
   994	   {
   995	      DWORD    lastError            = 0;
   996	      DWORD    bytesReturned        = 0;
   997	      BOOL     operationComplete    = FALSE;
   998	      UCHAR    sendBuffer[1];
   999	      DWORD    sendBufferLength     = 0;
  1000	
  1001	      sendBuffer[0]                 = enableValue;
  1002	      sendBufferLength = 1;
  1003	
  1004	      // Initialize the OVERLAPPED structure.
  1005	      memset(&overlappedStructure, 0x00, sizeof(OVERLAPPED));
  1006	      overlappedStructure.hEvent = hVirtualKeyboardSendEvent;
  1007	
  1008	      // Call the IOCTL to send data to the virtual keyboard control device.
  1009	      operationComplete = DeviceIoControl(
  1010	        hVirtualKeyboardDevice,
  1011	        IOCTL_CHATPAD_KEYBOARD_KMDF_SET_ENABLED,
  1012	        sendBuffer,
  1013	        sendBufferLength,
  1014	        NULL,
  1015	        0,
  1016	        &bytesReturned,
  1017	        &overlappedStructure);
  1018	
  1019	      if(FALSE == operationComplete)
  1020	      {
  1021	         // If the request failed, it is hopefully pending.
  1022	         lastError = GetLastError();
  1023	
  1024	         if(ERROR_IO_PENDING == lastError)
  1025	         {
  1026	            // Wait for the request to finish or the thread shutdown signal to be sent.
  1027	            DWORD ioctlWaitResult = 0;
  1028	            HANDLE waitHandles[2];
  1029	
  1030	            waitHandles[0] = hVirtualKeyboardSendEvent;
  1031	            waitHandles[1] = hWorkerThreadShutdownEvent;
  1032	            ioctlWaitResult = WaitForMultipleObjects(
  1033	              2,
  1034	              waitHandles,
  1035	              FALSE,       // Stop waiting after any of the specified events finishes.
  1036	              INFINITE);   // No timeout.
  1037	            if(WAIT_OBJECT_0 == ioctlWaitResult)
  1038	            {
  1039	               // Make sure that the IOCTL actually completed successfully.
  1040	               operationComplete = GetOverlappedResult(
  1041	                 hVirtualKeyboardDevice,
  1042	                 &overlappedStructure,
  1043	                 &bytesReturned,
  1044	                 FALSE);
  1045	               if(FALSE == operationComplete)
  1046	               {
  1047	                  lastError = GetLastError();
  1048	               }
  1049	            }
  1050	            else if((WAIT_OBJECT_0 + 1) == ioctlWaitResult)
  1051	            {
  1052	               // The worker thread shutdown event was signaled.
  1053	               nRetValue      = FAILURE;
  1054	               lastError      = 0;
  1055	               LogMessage("Thread shutdown detected while in SetVirtualKeyboardDeviceEnabled.\n");
  1056	            }
  1057	            else
  1058	            {
  1059	               LogMessage("WaitForMultipleObjects failed with return value %u\n", ioctlWaitResult);
  1060	               nRetValue      = FAILURE;
  1061	               lastError      = GetLastError();
  1062	            }
  1063	         } // end if(ERROR_IO_PENDING == lastError)
  1064	
  1065	         //***
  1066	         // Note that if lastError is not ERROR_IO_PENDING and the operation
  1067	         // does not subsequently complete successfully, then it will be caught
  1068	         // in code below.
  1069	         //***
  1070	
  1071	      } // end if(FALSE == operationComplete)
  1072	
  1073	      // Note that even if operationComplete is FALSE above, it should be set to TRUE later in
  1074	      // the above code after the operation actually finishes.
  1075	      if(TRUE != operationComplete)
  1076	      {
  1077	         LogMessage("Enable/disable virtual keyboard device IOCTL failed, lastError == %lu\n", lastError);
  1078	         nRetValue = FAILURE;
  1079	      }
  1080	
  1081	      // Reset the event in case it was signaled.  If it was not signaled, this
  1082	      // call should simply have no effect.
  1083	      ResetEvent(hVirtualKeyboardSendEvent);
  1084	   } // end if(SUCCESS == nRetValue)
  1085	
  1086	   // Release the mutex if we obtained it.
  1087	   if(WAIT_OBJECT_0 == waitResult)
  1088	   {
  1089	      if(0 == ReleaseMutex(hStatesMutex))
  1090	      {
  1091	         LogMessage("States mutex release failed:  %lu\n", GetLastError());
  1092	         nRetValue = FAILURE;
  1093	      }
  1094	   }
  1095	
  1096	   return nRetValue;
  1097	} // end SetVirtualKeyboardDeviceEnabled

  1099	// Executes an IOCTL to send keyboard data to the virtual keyboard device.
  1100	//
  1101	// IMPORTANT NOTE:  This function must only be called if hStatesMutex
  1102	// is held by the calling thread!  This function does not attempt
  1103	// to synchronize on its own!
  1104	//
  1105	// The sendBuffer pointer must point to a buffer that has exactly the right
  1106	// amount of data to work with the IOCTL_CHATPAD_KEYBOARD_KMDF_SEND_DATA IOCTL,
  1107	// and sendBufferLength must match this length.
  1108	//
  1109	// This function returns when the IOCTL is completed.
  1110	//
  1111	// This function returns SUCCESS on success and FAILURE on failure.
  1112	int SendVirtualKeyboardData(
  1113	  PUCHAR sendBuffer,
  1114	  DWORD  sendBufferLength)
  1115	{
  1116	   int         nRetValue            = SUCCESS;
  1117	   OVERLAPPED  overlappedStructure;
  1118	
  1119	   if((SUCCESS == nRetValue) &&
  1120	      (INVALID_HANDLE_VALUE == hVirtualKeyboardDevice))
  1121	   {
  1122	      LogMessage("Invalid virtual keyboard device handle in SendVirtualKeyboardData.\n");
  1123	      nRetValue = FAILURE;
  1124	   }
  1125	
  1126	   // TODO make 7 a constant somewhere instead of hardcoded here...maybe add it to the IOCTL header?
  1127	   if((SUCCESS == nRetValue) && 
  1128	      ((NULL == sendBuffer)         ||
  1129	       (7 != sendBufferLength)))
  1130	   {
  1131	      LogMessage("Invalid parameters provided to SendVirtualKeyboardData.\n");
  1132	      nRetValue = FAILURE;
  1133	   }
  1134	
  1135	   if((SUCCESS == nRetValue) &&
  1136	      (NULL == hVirtualKeyboardSendEvent))
  1137	   {
  1138	      LogMessage("Error, virtual keyboard send event handle is NULL.\n");
  1139	      nRetValue = FAILURE;
  1140	   }
  1141	
  1142	   if(SUCCESS == nRetValue)
  1143	   {
  1144	      DWORD lastError            = 0;
  1145	      DWORD bytesReturned        = 0;
  1146	      BOOL  operationComplete    = FALSE;
  1147	
  1148	      // Save these keys as the previously held-down keys.
  1149	      memcpy(previousOutputKeysHeldDown, &(sendBuffer[1]), MAX_SIMULTANEOUS_KEYS);
  1150	
  1151	      // Initialize the OVERLAPPED structure.
  1152	      memset(&overlappedStructure, 0x00, sizeof(OVERLAPPED));
  1153	      overlappedStructure.hEvent = hVirtualKeyboardSendEvent;
  1154	
  1155	/*
  1156	      LogMessage(

  1687	                  pChatpadStateData->rightControlLatchActive = false;
  1688	               }

  1691	               {
  1692	                  keyboardData[0] |= KEYBOARD_MODIFIER_RIGHT_SHIFT;
  1693	                  pChatpadStateData->rightShiftLatchActive = false;
  1694	               }
  1695	
  1696	               if(true == pChatpadStateData->rightAltLatchActive)
  1697	               {
  1698	                  keyboardData[0] |= KEYBOARD_MODIFIER_RIGHT_ALT;
  1699	                  pChatpadStateData->rightAltLatchActive = false;
  1700	               }
  1701	
  1702	               if(true == pChatpadStateData->rightGuiLatchActive)
  1703	               {
  1704	                  keyboardData[0] |= KEYBOARD_MODIFIER_RIGHT_GUI;
  1705	                  pChatpadStateData->rightGuiLatchActive = false;

  1709	               // it is in the binding itself.  This might change some day, but for
  1710	               // now it seems simplest to require a long configuration file that

  3345	                  else
  3346	                  {

  3348	                     nStatus = FAILURE;
  3349	                  }
  3350	               } // end if((WAIT_OBJECT_0 + controlsReadIndex) == ioctlWaitResult)
  3351	            } // end looping to find which controls read(s) completed
  3352	
  3353	            // If no controls reads seem to have finished, then an error presumably occurred.
  3354	            if(0 == numReadsFinished)
  3355	            {
  3356	               LogMessage("WaitForMultipleObjects failed with return value %u\n", ioctlWaitResult);
  3357	               LogMessage("Read from controls endpoint failed:  %lu\n", GetLastError());
  3358	               nStatus = FAILURE;
  3359	            }
  3360	         } // end handling if the worker thread shutdown event was not signaled
  3361	      } // end if(SUCCESS == nStatus)
  3362	   } // end while the worker thread is successful and has not been shutdown
  3363	
  3364	   // Cancel any I/O this worker thread has pending.
  3365	   CancelIo(hChatpadDevice);
  3366	
  3367	   // Exit this worker thread.
  3368	   LogMessage("Reading thread for controls data has ended, with nStatus == %d\n", nStatus);

  3711	   if(0xf0 != readBuffer[0])

  3716	      }

  3718	
  3719	   // Release the mutex if we obtained it.
  3720	   if(WAIT_OBJECT_0 == waitResult)
  3721	   {
  3722	      if(0 == ReleaseMutex(hStatesMutex))
  3723	      {
  3724	         LogMessage("States mutex release failed:  %lu\n", GetLastError());
  3725	         nRetValue = FAILURE;
  3726	      }

  3732	// Worker read code to continuously read chatpad data.
  3733	DWORD WINAPI ChatpadReadFunction(LPVOID lpParam)
  3734	{
  3735	   int         nStatus                                      = SUCCESS;
  3736	   HANDLE      hChatpadDevice                               = static_cast<HANDLE>(lpParam);
  3737	   BOOLEAN     workerThreadFinished                         = FALSE;
  3738	
  3739	   HANDLE      waitHandles[NUM_PENDING_READS_TO_USE + 1];
  3740	
  3741	   BOOL        readPending[NUM_PENDING_READS_TO_USE];
  3742	   DWORD       readBufferLength[NUM_PENDING_READS_TO_USE];
  3743	
  3744	   OVERLAPPED  overlappedStructure[NUM_PENDING_READS_TO_USE];
  3745	   UCHAR       readBuffer[NUM_PENDING_READS_TO_USE][32];
  3746	   DWORD       bytesRead[NUM_PENDING_READS_TO_USE];
  3747	
  3748	   LogMessage("Reading thread started for chatpad data.\n");
  3749	
  3750	   if(INVALID_HANDLE_VALUE == hChatpadDevice)
  3751	   {
  3752	      LogMessage("Error, invalid device handle provided to ChatpadReadFunction.\n");
  3753	      nStatus = FAILURE;
  3754	   }
  3755	
  3756	   if(SUCCESS == nStatus)
  3757	   {
  3758	      // Set up wait handles and mark the reads as not pending.
  3759	      for(int chatpadReadIndex = 0; chatpadReadIndex < NUM_PENDING_READS_TO_USE; chatpadReadIndex++)
  3760	      {
  3761	         waitHandles[chatpadReadIndex] = hChatpadReadEvents[chatpadReadIndex];
  3762	         readPending[chatpadReadIndex] = FALSE;
  3763	         readBufferLength[chatpadReadIndex] = 32;
  3764	      }
  3765	
  3766	      // Set up the wait handle for the worker thread shutdown event.
  3767	      waitHandles[NUM_PENDING_READS_TO_USE] = hWorkerThreadShutdownEvent;
  3768	   }
  3769	
  3770	   // Main loop for this worker thread.
  3771	   while((SUCCESS == nStatus) &&
  3772	         (FALSE == workerThreadFinished))
  3773	   {
  3774	      // Send any read requests that are not already pending.
  3775	      for(int chatpadReadIndex = 0;
  3776	          ((SUCCESS == nStatus) &&
  3777	           (chatpadReadIndex < NUM_PENDING_READS_TO_USE));
  3778	          chatpadReadIndex++)
  3779	      {
  3780	         if(FALSE == readPending[chatpadReadIndex])
  3781	         {
  3782	            // Mark the read as pending so it will not be sent again until it finishes.
  3783	            readPending[chatpadReadIndex] = TRUE;
  3784	
  3785	            // Initialize the OVERLAPPED structure.
  3786	            memset(&(overlappedStructure[chatpadReadIndex]), 0x00, sizeof(OVERLAPPED));
  3787	            overlappedStructure[chatpadReadIndex].hEvent = hChatpadReadEvents[chatpadReadIndex];
  3788	            // Reset the event in case it is already signaled.  If it is not
  3789	            // currently signaled, this call should simply have no effect.
  3790	            ResetEvent(hChatpadReadEvents[chatpadReadIndex]);
  3791	
  3792	            // Clear read buffer and the variable tracking the number of bytes read.
  3793	            memset(readBuffer[chatpadReadIndex],
  3794	                   0x00,
  3795	                   readBufferLength[chatpadReadIndex]);
  3796	            bytesRead[chatpadReadIndex] = 0;
  3797	
  3798	            // Call the IOCTL.
  3799	            BOOL operationComplete = DeviceIoControl(
  3800	              hChatpadDevice,
  3801	              IOCTL_CHATPAD_READ_FROM_CHATPAD_ENDPOINT,
  3802	              NULL,
  3803	              0,
  3804	              readBuffer[chatpadReadIndex],
  3805	              readBufferLength[chatpadReadIndex],
  3806	              &bytesRead[chatpadReadIndex],
  3807	              &overlappedStructure[chatpadReadIndex]);
  3808	            if(FALSE == operationComplete)
  3809	            {
  3810	               DWORD lastError = GetLastError();
  3811	
  3812	               // At this point, lastError should be ERROR_IO_PENDING.
  3813	               if(ERROR_IO_PENDING != lastError)
  3814	               {
  3815	                  LogMessage("DeviceIoControl failed:  %lu\n", lastError);
  3816	                  nStatus = FAILURE;
  3817	               }
  3818	            } // end if(FALSE == operationComplete)
  3819	         } // end if(FALSE == readPending[chatpadReadIndex])
  3820	      } // end looping to send read requests that need sent
  3821	
  3822	      // Now that all read requests should be pending, wait for any of them to finish, or
  3823	      // for worker thread shutdown to be signaled.
  3824	
  3825	      if(SUCCESS == nStatus)
  3826	      {
  3827	         DWORD ioctlWaitResult = 0;
  3828	
  3829	//         LogMessage("Waiting on chatpad input...\n");
  3830	         ioctlWaitResult = WaitForMultipleObjects(
  3831	           (NUM_PENDING_READS_TO_USE + 1),
  3832	           waitHandles,
  3833	           FALSE,       // Stop waiting after any of the specified events finishes.
  3834	           INFINITE);   // No timeout.
  3835	         if((WAIT_OBJECT_0 + NUM_PENDING_READS_TO_USE) == ioctlWaitResult)
  3836	         {
  3837	            // The worker thread shutdown event was signaled.
  3838	            LogMessage("Thread shutdown detected while in ChatpadReadFunction.\n");
  3839	            workerThreadFinished = TRUE;
  3840	            continue;
  3841	         }
  3842	         else
  3843	         {
  3844	            int numReadsFinished = 0;
  3845	
  3846	            // Find which chatpad read(s) completed.
  3847	            for(int chatpadReadIndex = 0;
  3848	                ((SUCCESS == nStatus) &&
  3849	                 (chatpadReadIndex < NUM_PENDING_READS_TO_USE));
  3850	                chatpadReadIndex++)
  3851	            {
  3852	               if((WAIT_OBJECT_0 + chatpadReadIndex) == ioctlWaitResult)
  3853	               {
  3854	                  numReadsFinished++;
  3855	
  3856	                  // Check if the IOCTL actually completed successfully.
  3857	                  BOOL operationComplete = GetOverlappedResult(
  3858	                    hChatpadDevice,
  3859	                    &overlappedStructure[chatpadReadIndex],
  3860	                    &(bytesRead[chatpadReadIndex]),
  3861	                    FALSE);
  3862	                  if(TRUE == operationComplete)
  3863	                  {
  3864	                     // Enable the chatpad keyboard backlight, which will activate when keys are pressed.
  3865	                     // This command seems to be required for chatpad key data to be received at all,
  3866	                     // so there may not even be a way for the backlight to be disabled but keys to
  3867	                     // still operate.
  3868	                     // Only sending this command once seems to work, but it needs to be after
  3869	                     // at least one chatpad data message has been received, and so it
  3870	                     // is done here.
  3871	                     // 
  3872	                     // No mutex protection is needed for backlightCommandSent since this is the only
  3873	                     // thread in which it will ever be read or set.
  3874	                     if(false == pChatpadStateData->backlightCommandSent)
  3875	                     {
  3876	                        pChatpadStateData->backlightCommandSent = true;
  3877	                        // TODO research at some point to see if the backlight can be disabled?
  3878	                        SendControlRequest(hChatpadDevice, CHATPAD_INTERFACE, 0x41, 0x00, 0x001b, 0x0002, 0x0000);
  3879	                     }
  3880	
  3881	/*
  3882	if(0xf0 != readBuffer[chatpadReadIndex][0])
  3883	{
  3884	   LogMessage(
  3885	     "[chatpad] Length %u:  [ 0x%02x 0x%02x 0x%02x 0x%02x 0x%02x ]\n",
  3886	     bytesRead[chatpadReadIndex],
  3887	     readBuffer[chatpadReadIndex][0],
  3888	     readBuffer[chatpadReadIndex][1],
  3889	     readBuffer[chatpadReadIndex][2],
  3890	     readBuffer[chatpadReadIndex][3],
  3891	     readBuffer[chatpadReadIndex][4]);
  3892	}
  3893	*/
  3894	                     // Process the chatpad data.
  3895	                     // TODO add some sort of state parameter, and pass a pointer here?  Probably
  3896	                     //   a struct.
  3897	                     int nChatpadReturnValue = HandleChatpadData(
  3898	                       hChatpadDevice, readBuffer[chatpadReadIndex], bytesRead[chatpadReadIndex]);
  3899	                     // I am anticipating that I may have some unexpected data-processing failures,
  3900	                     // so I am just printing an error if one occurs, but not stopping overall processing.
  3901	                     if(SUCCESS != nChatpadReturnValue)
  3902	                     {
  3903	                        LogMessage("Error, HandleChatpadData returned FAILURE.\n");
  3904	                     }
  3905	
  3906	                     // Mark the read as no longer pending.
  3907	                     readPending[chatpadReadIndex] = FALSE;
  3908	                  }
  3909	                  else
  3910	                  {
  3911	                     LogMessage("Chatpad data read completed but a failure occurred:  %lu\n", GetLastError());
  3912	                     nStatus = FAILURE;
  3913	                  }
  3914	               } // end if((WAIT_OBJECT_0 + chatpadReadIndex) == ioctlWaitResult)
  3915	            } // end looping to find which chatpad read(s) completed
  3916	
  3917	            // If no chatpad reads seem to have finished, then an error presumably occurred.
  3918	            if(0 == numReadsFinished)
  3919	            {
  3920	               LogMessage("WaitForMultipleObjects failed with return value %u\n", ioctlWaitResult);
  3921	               LogMessage("Read from chatpad endpoint failed:  %lu\n", GetLastError());
  3922	               nStatus = FAILURE;
  3923	            }
  3924	         } // end handling if the worker thread shutdown event was not signaled
  3925	      } // end if(SUCCESS == nStatus)
  3926	   } // end while the worker thread is successful and has not been shutdown
  3927	
  3928	   // Cancel any I/O this worker thread has pending.
  3929	   CancelIo(hChatpadDevice);
  3930	
  3931	   // Exit this worker thread.
  3932	   LogMessage("Reading thread for chatpad data has ended, with nStatus == %d\n", nStatus);
  3933	   return 0;
  3934	} // end ChatpadReadFunction

  4070	// Initializes the chatpad.
  4071	int InitChatpad(
  4072	  HANDLE hChatpadDevice)
  4073	{
  4074	   int         nRetValue            = SUCCESS;
  4075	   DWORD       waitResult           = 0;
  4076	   DWORD       bytesTransferred     = 0;
  4077	   OVERLAPPED  overlappedStructure;
  4078	
  4079	   // Get the control request mutex to ensure that only one control request is ongoing at a time,
  4080	   // and that two threads do not try to use the event at the same time.
  4081	   waitResult = WaitForSingleObject(hControlRequestMutex, INFINITE);
  4082	   // Make sure the wait actually succeeded.
  4083	   if(WAIT_OBJECT_0 != waitResult)
  4084	   {
  4085	      LogMessage("Wait for mutex failed:  %u\n", waitResult);
  4086	      nRetValue = FAILURE;
  4087	   }
  4088	
  4089	   if((SUCCESS == nRetValue) &&
  4090	      (NULL == hControlRequestEvent))
  4091	   {
  4092	      LogMessage("Error, control request event handle is NULL.\n");
  4093	      nRetValue = FAILURE;
  4094	   }
  4095	
  4096	   // Initialize the OVERLAPPED structure.
  4097	   memset(&overlappedStructure, 0x00, sizeof(OVERLAPPED));
  4098	   overlappedStructure.hEvent = hControlRequestEvent;
  4099	
  4100	   if(SUCCESS == nRetValue)
  4101	   {
  4102	      DWORD lastError = 0;
  4103	      // Note that bytesTransferred is needed according to
  4104	      // http://msdn.microsoft.com/en-us/library/aa363216%28VS.85%29.aspx even
  4105	      // though it is not used here.
  4106	      BOOL operationComplete = DeviceIoControl(
  4107	        hChatpadDevice,
  4108	        IOCTL_CHATPAD_IS_MS_INIT_DONE,
  4109	        NULL,
  4110	        0,
  4111	        NULL,
  4112	        0,
  4113	        &bytesTransferred,
  4114	        &overlappedStructure);
  4115	
  4116	      if(FALSE == operationComplete)
  4117	      {
  4118	         // If the request failed, it is hopefully pending.
  4119	         lastError = GetLastError();
  4120	
  4121	         if(ERROR_IO_PENDING == lastError)
  4122	         {
  4123	            // Wait for the request to finish.
  4124	            // TODO should I include timeout functionality?  Could these sorts of waits ever cause lockups?
  4125	            operationComplete = GetOverlappedResult(
  4126	              hChatpadDevice,
  4127	              &overlappedStructure,
  4128	              &bytesTransferred,
  4129	              TRUE);
  4130	         }
  4131	      }
  4132	
  4133	      if(TRUE == operationComplete)
  4134	      {
  4135	         LogMessage("Microsoft initialization has been finished, so we can init the chatpad now.\n");
  4136	      }
  4137	      else
  4138	      {
  4139	         LogMessage("Microsoft initialization is not finished, or detection has failed:  %d\n", lastError);
  4140	         LogMessage("Please contact GAFBlizzard to find a solution to this problem.\n");
  4141	         nRetValue = FAILURE;
  4142	      }
  4143	
  4144	      // Reset the event in case it was signaled.  If it was not signaled, this
  4145	      // call should simply have no effect.
  4146	      ResetEvent(hControlRequestEvent);
  4147	   } // end if(SUCCESS == nRetValue)
  4148	
  4149	   // Release the mutex if we obtained it.
  4150	   //
  4151	   // The mutex is released now because the rest of the initialization is not
  4152	   // raw DeviceIoControl calls, but instead calls to other functions which
  4153	   // themselves will try to obtain the mutex.
  4154	   if(WAIT_OBJECT_0 == waitResult)
  4155	   {
  4156	      if(0 == ReleaseMutex(hControlRequestMutex))
  4157	      {
  4158	         LogMessage("Mutex release failed:  %lu\n", GetLastError());
  4159	         nRetValue = FAILURE;
  4160	      }
  4161	   }
  4162	
  4163	   if(SUCCESS == nRetValue)
  4164	   {
  4165	      unsigned char sendMessageBuffer[8];
  4166	      unsigned char receiveMessageBuffer[32];
  4167	
  4168	      memset(sendMessageBuffer, 0x00, 8);
  4169	      memset(receiveMessageBuffer, 0x00, 32);
  4170	
  4171	      // Note that the XBox 360 controller security interface communication is
  4172	      // presumably not needed (the Mac driver doesn't do it as far as I can tell),
  4173	      // so it is not done here either.
  4174	
  4175	      // These three calls produce stalls/failures.  At any rate, if I do not send these
  4176	      // three, even with the failures they return, then the commands below to make chatpad
  4177	      // messages start will fail.
  4178	      // Apparently they are required and I do not know where they come from.  Maybe check the
  4179	      // code that I got the 0x09 0x00 data from.
  4180	      // TODO do I need to automatically reset/unstall?  Things seem to work without doing that.
  4181	      // TODO comment out the IOCTL failure message for release, since it is expected?  Or add
  4182	      //   an option to disable it for SendControlRequest?
  4183	      SendControlRequest(hChatpadDevice, MAIN_INTERFACE, 0x40, 0xa9, 0xa30c, 0x4423, 0x0000);
  4184	      SendControlRequest(hChatpadDevice, MAIN_INTERFACE, 0x40, 0xa9, 0x2344, 0x7f03, 0x0000);
  4185	      SendControlRequest(hChatpadDevice, MAIN_INTERFACE, 0x40, 0xa9, 0x5839, 0x6832, 0x0000);
  4186	
  4187	      // Send control codes to make the chatpad start sending data (or to make
  4188	      // the 360 controller let the chatpad data come through, who knows how
  4189	      // it works internally :P).
  4190	      if(SUCCESS == nRetValue)
  4191	      {
  4192	         nRetValue = SendControlRequest(
  4193	           hChatpadDevice, CHATPAD_INTERFACE, 0xc0, 0xa1, 0x0000, 0xe416, 0x0002, NULL, 0, receiveMessageBuffer, 32);
  4194	         LogMessage("Control response data received:  0x%02x 0x%02x\n", receiveMessageBuffer[0], receiveMessageBuffer[1]);
  4195	      }
  4196	
  4197	      if(SUCCESS == nRetValue)
  4198	      {
  4199	         // The working chatpad activation hex values are here thanks to
  4200	         // http://mbed.org/users/AjK/programs/SOWB/lg489e/docs/xbox360gamepad_8c_source.html.
  4201	         // The notes from the author there seem to imply that the security channel is required to handshake
  4202	         // with the controller in order for the chatpad to work.  That does not seem to be the case.
  4203	         // Neither this code nor the Mac driver uses the security channel to handshake, as far as I know,
  4204	         // unless the three mystery commands above (0x40, 0xa9, etc.) relate to security handshaking.
  4205	//sendMessageBuffer[0] = 0x01;
  4206	//sendMessageBuffer[1] = 0x02;
  4207	// TODO note that the above value may be necessary on some controllers

  4291	// Shuts down worker threads so that global handles can safely be deleted.
  4292	// IMPORTANT NOTE:  This function should only be called after InitializeGlobalVariables() has been called.
  4293	void ShutdownWorkerThreads(void)
  4294	{
  4295	   DWORD waitResult = 0;
  4296	
  4297	   if(NULL != hWorkerThreadShutdownEvent)
  4298	   {
  4299	      // Trigger the worker thread shutdown event to stop all child threads.
  4300	      if(0 == SetEvent(hWorkerThreadShutdownEvent))
  4301	      {
  4302	         // Hopefully this never happens.
  4303	         LogMessage("Error triggering shutdown event:  %lu\n", GetLastError());
  4304	         LogMessage("Forcing an exit since normal thread shutdown failed.\n");
  4305	         exit(-1);
  4306	      }
  4307	      // Set the special-case boolean to shut down the mouse worker thread.
  4308	      bShutdownMouseWorkerThread = true;
  4309	
  4310	      // Wait for the worker threads to shutdown.
  4311	      waitResult = WaitForMultipleObjects(
  4312	        NUM_WORKER_THREADS,
  4313	        hWorkerThreads,
  4314	        TRUE,     // Wait for all of the handles to be signaled.
  4315	        5000);    // Wait 5 seconds and then timeout.
  4316	
  4317	      if(WAIT_OBJECT_0 != waitResult)
  4318	      {
  4319	         LogMessage("Error while waiting on threads to finish, waitResult == 0x%x\n", waitResult);
  4320	         if(WAIT_FAILED == waitResult)
  4321	         {
  4322	            LogMessage("WaitForMultipleObjects failed.  GetLastError() returns %lu.\n", GetLastError());
  4323	         }
  4324	         LogMessage("Forcing an exit since normal thread shutdown failed.\n");
  4325	         exit(-1);
  4326	      }
  4327	   } // end if(NULL != hWorkerThreadShutdownEvent)
  4328	   else
  4329	   {
  4330	      LogMessage("Error while shutting down:  Shutdown event handle is NULL.\n");
  4331	      LogMessage("Forcing an exit since normal thread shutdown failed.\n");
  4332	      exit(-1);
  4333	   }
  4334	
  4335	   LogMessage("Worker threads successfully stopped.\n");
  4336	} // end ShutdownWorkerThreads

  4772	            continue;
  4773	         }
  4774	         Sleep(1000);
  4775	      }
  4776	   } // end if(SUCCESS == nRetValue)
  4777	
  4778	// TODO add Control-C handler, or a different method of running and quitting things?  Currently,
  4779	//   none of this cleanup code gets called if Control-C is used to just kill the control utility.
  4780	
  4781	   //***
  4782	   // Clean things up
  4783	   //***
  4784	
  4785	   // Make sure the driver stops intercepting or filtering controls data when this utility exits.
  4786	   if(INVALID_HANDLE_VALUE != hChatpadDevice)
  4787	   {
  4788	      SetFilterMode(hChatpadDevice, FILTER_MODE_UNFILTERED);
  4789	   }
  4790	
  4791	// TODO NEXT add code to set mapping options back to normal when the utility
  4792	//   exits, and make sure the equivalent code is also in the driver itself,
  4793	//   like the way the filtering stuff already works
  4794	
  4795	   // Shutdown the worker threads so that we can safely close handles.
  4796	   // Note that this will not get called if one of the threads is created
  4797	   // and the other one fails for some reason.  In that case, any remaining
  4798	   // threads will presumably just be terminated by Windows when the program
  4799	   // exits.
  4800	   bool bShutdownWorkerThreads = true;
  4801	   for(int i = 0; i < NUM_WORKER_THREADS; i++)
  4802	   {
  4803	      if(NULL == hWorkerThreads[i])
  4804	      {
  4805	         bShutdownWorkerThreads = false;
  4806	         break;
  4807	      }
  4808	   }
  4809	   if(true == bShutdownWorkerThreads)
  4810	   {
  4811	      ShutdownWorkerThreads();
  4812	   }
  4813	
  4814	   // Disable and close the virtual keyboard device.
  4815	   if(INVALID_HANDLE_VALUE != hVirtualKeyboardDevice)
  4816	   {
  4817	      // Disable the virtual keyboard device (even though this currently
  4818	      // happens automatically when the application disconnects).
  4819	      SetVirtualKeyboardDeviceEnabled(CHATPAD_KEYBOARD_KMDF_DISABLE);
  4820	
  4821	      LogMessage("Closing connection to virtual keyboard device.\n");
  4822	      CloseHandle(hVirtualKeyboardDevice);
  4823	      hVirtualKeyboardDevice = INVALID_HANDLE_VALUE;
  4824	   }
  4825	
  4826	   // Disable and close the virtual mouse device.
  4827	   if(INVALID_HANDLE_VALUE != hVirtualMouseDevice)
  4828	   {
  4829	      // Disable the virtual mouse device (even though this currently
  4830	      // happens automatically when the application disconnects).
  4831	      SetVirtualMouseDeviceEnabled(CHATPAD_MOUSE_KMDF_DISABLE);
  4832	
  4833	      LogMessage("Closing connection to virtual mouse device.\n");
  4834	      CloseHandle(hVirtualMouseDevice);
  4835	      hVirtualMouseDevice = INVALID_HANDLE_VALUE;
  4836	   }
  4837	
  4838	   // Close the chatpad control device.
  4839	   if(INVALID_HANDLE_VALUE != hChatpadDevice)
  4840	   {
  4841	      LogMessage("Closing chatpad filter driver.\n");
  4842	      CloseHandle(hChatpadDevice);
  4843	      hChatpadDevice = INVALID_HANDLE_VALUE;
  4844	   }
  4845	
  4846	   // Cleanup global objects.
  4847	   CleanupGlobalVariables();
  4848	
  4849	   // This message should show up in DebugView.
  4850	   OutputDebugString(("Chatpad control main loop finished.\n"));
  4851	
  4852		return nRetValue;
  4853	} // end ChatpadControlMainLoop
