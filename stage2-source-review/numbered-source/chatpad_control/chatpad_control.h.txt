   389	
   390	// State information used for handling chatpad data.
   391	typedef struct ChatpadStateData_t
   392	{
   393	   // Whether the 0x001b backlight command has been sent yet.  It must be sent
   394	   // in the proper sequence.  See chatpad_control_code.cpp in the
   395	   // HandleChatpadData function for details.
   396	   bool backlightCommandSent;
   397	
   398	   // Whether caps lock is currently active.  Caps lock is different from normal
   399	   // modifiers and latch keys, so it is its own boolean.  Caps lock does not
   400	   // provide modifiers to keys but instead just corresponds to the caps lock
   401	   // state that Windows tracks for keyboards.
   402	   bool capsLockActive;
   403	
   404	   // Currently active modifiers.  This value is set by using binary OR
   405	   // operations to combine CHATPAD_MODIFIER_ constants.
   406	   // If no modifiers are currently active, this value will be set to
   407	   // CHATPAD_MODIFIER_NONE.
   408	   unsigned int currentModifiers;
   409	
   410	   // Whether various special latch actions are active.  For each of these
   411	   // actions that are active, a chatpad or keyboard modifier will be applied
   412	   // to the next chatpad key press or simulated key event that occurs,
   413	   // unless the latch action triggers again, which will turn the latch back
   414	   // off.
   415	
   416	   // Chatpad event latches.
   417	   bool orangeLatchActive;
   418	   bool greenLatchActive;
   419	   bool peopleLatchActive;
   420	   // Simulated keyboard event latches.
   421	   bool leftControlLatchActive;
   422	   bool leftShiftLatchActive;
   423	   bool leftAltLatchActive;
   424	   bool leftGuiLatchActive;
   425	   bool rightControlLatchActive;
   426	   bool rightShiftLatchActive;
   427	   bool rightAltLatchActive;
   428	   bool rightGuiLatchActive;
   429	
   430	   // The previous raw message received from the chatpad, used for detecting duplicate messages.
   431	   // Note that this does NOT include messages starting with 0xf0 in the first byte.  Those
   432	   // messages are currently not used by the chatpad control code.
   433	   unsigned char previousChatpadRawData[5];
   434	
   435	   // Which keys are currently held down, identified by CHATPAD_KEY_ constants.
   436	   // If either of the slots does not correspond to a key, then it will be
   437	   // set to CHATPAD_KEY_NONE.
   438	   unsigned char currentKeysDown[2];
   439	   // Modifiers that were active for each of the keys that were pressed.
   440	   // These values are needed to detect when certain combinations were
   441	   // released.
   442	   unsigned int currentKeysDownModifiers[2];
   443	   // If this value is true, the next set of chatpad data that exactly matches
   444	   // the last set will be ignored.
   445	   bool ignoreNextRepeatedData;
   446	} ChatpadStateData;
