   306	// Converts a null-terminated string to uppercase.
   307	void StringToUpper(char* pString)
   308	{
   309	   if(NULL != pString)
   310	   {
   311	      char* pStringPtr = pString;
   312	      while('\0' != (*pStringPtr))
   313	      {
   314	         if(((*pStringPtr) >= 'a') &&
   315	            ((*pStringPtr) <= 'z'))
   316	         {
   317	            (*pStringPtr) = ((*pStringPtr) - 'a') + 'A';
   318	         }
   319	         pStringPtr++;
   320	      }
   321	   }
   322	} // end StringToUpper

   324	// Skips over whitespace pointed to by pSeekPtr and returns a pointer to a non-whitespace character.
   325	char* SkipWhiteSpace(char* pSeekPtr)
   326	{
   327	   char* pReturnPtr = pSeekPtr;
   328	
   329	   if(NULL != pReturnPtr)
   330	   {
   331	      while((' ' == (*pReturnPtr))    ||
   332	            ('\t' == (*pReturnPtr)))
   333	      {
   334	         pReturnPtr++;
   335	      }
   336	   }
   337	
   338	   return pReturnPtr;
   339	} // end SkipWhiteSpace

   341	// Skips backwards over whitespace pointed to by pSeekPtr and returns a pointer
   342	// to a non-whitespace character.  The pStringStart parameter must point to
   343	// the first character of the string to serve as a minimum value that pSeekPtr
   344	// can hold.  If the string is entirely whitespace then this function will
   345	// return a pointer to the first character in the string.
   346	char* SkipEndingWhiteSpace(char* pStringStart, char* pSeekPtr)
   347	{
   348	   char* pReturnPtr = pSeekPtr;
   349	
   350	   if((NULL != pStringStart) &&
   351	      (NULL != pReturnPtr))
   352	   {
   353	      if(pReturnPtr < pStringStart)
   354	      {
   355	         LogMessage("Error, seek pointer is less than start pointer in SkipWhiteSpace.\n");
   356	         pReturnPtr = pStringStart;
   357	      }
   358	
   359	      while((pReturnPtr > pStringStart) &&
   360	            ((' ' == (*pReturnPtr))             ||
   361	             ('\t' == (*pReturnPtr))))
   362	      {
   363	         pReturnPtr--;
   364	      }
   365	   }
   366	
   367	   return pReturnPtr;
   368	} // end SkipEndingWhiteSpace

   483	// This function examines the string pointed to by *pSeekPtr, skips
   484	// any initial whitespace, copies a token, skips any final whitespace,
   485	// skips over any '+' character, and leaves *pSeekPtr pointing to
   486	// a non-whitespace character that should be the start of the next token,
   487	// or '\0' if no more tokens are available.
   488	//
   489	// This function returns SUCCESS on success and FAILURE on failure.
   490	int CopyNextToken(
   491	  char** pSeekPtr,
   492	  char*  pTokenBuffer,
   493	  int    nTokenBufferLength)
   494	{
   495	   int   nRetValue            = SUCCESS;
   496	   // This pointer is used to copy data into pTokenBuffer.
   497	   char* pTokenBufferPtr      = NULL;
   498	   // This pointer points to the last valid location in pTokenBuffer to prevent overruns.
   499	   char* pTokenBufferEndPtr   = NULL;
   500	
   501	   if((NULL == pSeekPtr)         ||
   502	      (NULL == (*pSeekPtr))      ||
   503	      (NULL == pTokenBuffer)     ||
   504	      (0 == nTokenBufferLength))
   505	   {
   506	      LogMessage("Invalid parameters provided to CopyNextToken.\n");
   507	      nRetValue = FAILURE;
   508	   }
   509	
   510	   if(SUCCESS == nRetValue)
   511	   {
   512	      pTokenBufferPtr = pTokenBuffer;
   513	      pTokenBufferEndPtr = pTokenBufferPtr + (nTokenBufferLength - 1);
   514	
   515	      // Skip initial whitespace.
   516	      (*pSeekPtr) = SkipWhiteSpace(*pSeekPtr);
   517	
   518	      // Copy the token.
   519	      while((' ' != (**pSeekPtr))    &&
   520	            ('\t' != (**pSeekPtr))   &&
   521	            ('+' != (**pSeekPtr))    &&
   522	            ('\0' != (**pSeekPtr))   &&
   523	            (pTokenBufferPtr < pTokenBufferEndPtr))
   524	      {
   525	         (*pTokenBufferPtr) = (**pSeekPtr);
   526	         pTokenBufferPtr++;
   527	         (*pSeekPtr)++;
   528	      }
   529	
   530	      // Null-terminate the token.
   531	      (*pTokenBufferPtr) = '\0';
   532	
   533	      // Skip ending whitespace.
   534	      while((' ' == (**pSeekPtr))    ||
   535	            ('\t' == (**pSeekPtr)))
   536	      {
   537	         (*pSeekPtr)++;
   538	      }
   539	
   540	      // Make sure that the next character is a '+' or '\0', or else return an error.
   541	      if('+' == (**pSeekPtr))
   542	      {
   543	         (*pSeekPtr)++;
   544	      }
   545	      else
   546	      {
   547	         if('\0' != (**pSeekPtr))
   548	         {
   549	            nRetValue = FAILURE;
   550	         }
   551	      }
   552	   } // end if(SUCCESS == nRetValue)
   553	
   554	   return nRetValue;
   555	} // end CopyNextToken

   557	// This function looks up the string pTokenString in the configStringTable
   558	// structure.  If the string is found, then it writes the corresponding data to
   559	// the structure pointed to by pTokenMapping.
   560	//
   561	// IMPORTANT NOTE:  This function copies a string pointer into the structure
   562	// pointed to by pTokenMapping.  Since this pointer points to a constant
   563	// string in global storage, calling code should NOT attempt to free it.
   564	//
   565	// If the string is not found, or if an error occurs, then this function
   566	// returns FAILURE.  Otherwise, this function returns SUCCESS.
   567	int GetTokenMapping(
   568	  char*                             pTokenString,
   569	  ChatpadDriverConfigStringMapping* pTokenMapping)
   570	{
   571	   int   nRetValue   = SUCCESS;
   572	
   573	   if((NULL == pTokenString)     ||
   574	      (NULL == pTokenMapping))
   575	   {
   576	      LogMessage("Invalid parameters provided to GetTokenMapping.\n");
   577	      nRetValue = FAILURE;
   578	   }
   579	
   580	   if(SUCCESS == nRetValue)
   581	   {
   582	      bool bMatchFound = false;
   583	
   584	      for(unsigned int i = 0;
   585	          ((i < uNumConfigStringTableStrings) &&
   586	           (false == bMatchFound));
   587	          i++)
   588	      {
   589	         if(0 == strcmp(pTokenString, configStringTable[i].pString))
   590	         {
   591	            bMatchFound = true;
   592	            pTokenMapping->pString     = configStringTable[i].pString;
   593	            pTokenMapping->uConstant   = configStringTable[i].uConstant;
   594	            pTokenMapping->stringType  = configStringTable[i].stringType;
   595	         } // end if(0 == strcmp(pTokenString, configStringTable[i]))
   596	      } // end looping through configStringTable
   597	
   598	      if(false == bMatchFound)
   599	      {
   600	         nRetValue = FAILURE;
   601	      }
   602	   } // end if(SUCCESS == nRetValue)
   603	
   604	   return nRetValue;
   605	}

   607	// This function handles pConfigLineLeftSide and pConfigLineRightSide as a
   608	// config mapping line.  The line is parsed and data is written into the
   609	// structure pointed to by pConfigStructure.  This function then return
   610	// SUCCESS.
   611	//
   612	// IMPORTANT NOTE:  pConfigLineLeftSide and pConfigLineRightSide must
   613	// point to null-terminated strings since no string lengths are provided
   614	// to this function.
   615	//
   616	// If the left and right side strings do not represent a valid config
   617	// mapping line, or if another error occurs, then FAILURE is returned.
   618	int HandleConfigMappingLine(
   619	  ChatpadDriverConfigStructure*  pConfigStructure,
   620	  char*                          pConfigLineLeftSide,
   621	  char*                          pConfigLineRightSide)
   622	{
   623	   int   nRetValue   = SUCCESS;
   624	   char* pSeekPtr    = NULL;
   625	   ChatpadDriverConfigStringMapping sourceTokenMapping;
   626	   ChatpadDriverConfigStringMapping actionTokenMapping;
   627	   ChatpadDriverConfigStringMapping modifierTokenMapping;
   628	   ChatpadDriverActionBinding       actionBinding;
   629	
   630	   if((NULL == pConfigStructure)       ||
   631	      (NULL == pConfigLineLeftSide)    ||
   632	      (NULL == pConfigLineRightSide))
   633	   {
   634	      LogMessage("Invalid parameters provided to HandleConfigMappingLine.\n");
   635	      nRetValue = FAILURE;
   636	   }
   637	
   638	   if(SUCCESS == nRetValue)
   639	   {
   640	      // Parse tokens in the left side string first.
   641	      pSeekPtr = pConfigLineLeftSide;
   642	
   643	      // Copy the first token.
   644	      configFileTokenBuffer[0] = '\0';
   645	      nRetValue = CopyNextToken(&pSeekPtr, configFileTokenBuffer, CONFIG_FILE_TOKEN_BUFFER_LENGTH);
   646	   }
   647	
   648	   if(SUCCESS == nRetValue)
   649	   {
   650	      actionBinding.uSourceActionModifiers   = 0;
   651	      actionBinding.bindingType              = BINDING_NONE;
   652	      actionBinding.uActionModifiers         = 0;
   653	      actionBinding.uActionCode              = 0;
   654	      sourceTokenMapping.pString    = NULL;
   655	      sourceTokenMapping.uConstant  = 0;
   656	      sourceTokenMapping.stringType = INVALID_STRING;
   657	      nRetValue = GetTokenMapping(configFileTokenBuffer, &sourceTokenMapping);
   658	      if(SUCCESS != nRetValue)
   659	      {
   660	         LogMessage("Error, unknown config string \"%s\".\nSee README.TXT for valid strings.\n",
   661	                    configFileTokenBuffer);
   662	      }
   663	   }
   664	
   665	   if(SUCCESS == nRetValue)
   666	   {
   667	      // The first token must correspond to either a chatpad key or a controller button.
   668	      //
   669	      // Note that there is probably a more elegant way to do this than repeating so much
   670	      // code, but repeating the code is easier for now.
   671	      if(CHATPAD_KEY_STRING == sourceTokenMapping.stringType)
   672	      {
   673	         // Since this is a chatpad key binding, check for modifiers that are
   674	         // compatible with chatpad keys.
   675	
   676	         // Note that eventually, this code might be refactored to consolidate some of
   677	         // the similar code below that loops through modifiers.
   678	         while(('\0' != (*pSeekPtr))  &&
   679	               (SUCCESS == nRetValue))
   680	         {
   681	            configFileTokenBuffer[0] = '\0';
   682	            nRetValue = CopyNextToken(&pSeekPtr, configFileTokenBuffer, CONFIG_FILE_TOKEN_BUFFER_LENGTH);
   683	            if(SUCCESS == nRetValue)
   684	            {
   685	               modifierTokenMapping.pString    = NULL;
   686	               modifierTokenMapping.uConstant  = 0;
   687	               modifierTokenMapping.stringType = INVALID_STRING;
   688	               nRetValue = GetTokenMapping(configFileTokenBuffer, &modifierTokenMapping);
   689	               // Check modifier compatibility.
   690	               if(CHATPAD_MODIFIER_STRING == modifierTokenMapping.stringType)
   691	               {
   692	                  // Add the modifier to the current action binding.
   693	                  actionBinding.uSourceActionModifiers |= modifierTokenMapping.uConstant;
   694	               }
   695	               else
   696	               {
   697	                  LogMessage("Error, expected chatpad key modifier but found \"%s\".\n",
   698	                             configFileTokenBuffer);
   699	                  nRetValue = FAILURE;
   700	               }
   701	            } // end if(SUCCESS == nRetValue)
   702	         } // end while parsing through source action modifiers
   703	
   704	         if(SUCCESS == nRetValue)
   705	         {
   706	            bool                          bBindingAlreadyExists   = false;
   707	            ChatpadKeyMapping*            pKeyMapping             = NULL;
   708	            ChatpadDriverActionBinding*   pNewBinding             = NULL;
   709	
   710	            pKeyMapping = &(pConfigStructure->keyMappings[sourceTokenMapping.uConstant]);
   711	
   712	            if(NULL != pKeyMapping->pBindings)
   713	            {
   714	               // Check whether there is already a binding for the source action.
   715	               for(int nBindingIndex = 0; nBindingIndex < (pKeyMapping->nNumBindings); nBindingIndex++)
   716	               {
   717	                  if(pKeyMapping->pBindings[nBindingIndex].uSourceActionModifiers ==
   718	                     actionBinding.uSourceActionModifiers)
   719	                  {
   720	                     bBindingAlreadyExists = true;
   721	                     pNewBinding = &(pKeyMapping->pBindings[nBindingIndex]);
   722	                     break;
   723	                  }
   724	               }
   725	            }
   726	
   727	            if(false == bBindingAlreadyExists)
   728	            {
   729	               ChatpadDriverActionBinding* pReallocatedBindings = NULL;
   730	
   731	               // An intermediate pointer is used here since realloc could
   732	               // fail and cause the original memory in pKeyMapping->pBindings
   733	               // to leak.
   734	               pReallocatedBindings =
   735	                 static_cast<ChatpadDriverActionBinding*>(realloc(
   736	                   pKeyMapping->pBindings,
   737	                   ((pKeyMapping->nNumBindings + 1) * sizeof(ChatpadDriverActionBinding))));
   738	               if(NULL != pReallocatedBindings)
   739	               {
   740	                  pNewBinding = &(pReallocatedBindings[pKeyMapping->nNumBindings]);
   741	
   742	                  pKeyMapping->nNumBindings++;
   743	                  pKeyMapping->pBindings = pReallocatedBindings;
   744	               } // end if(NULL != pReallocatedBindings)
   745	               else
   746	               {
   747	                  LogMessage("Error allocating memory in HandleConfigMappingLine.\n");
   748	                  nRetValue = FAILURE;
   749	               }
   750	            } // end if(false == bBindingAlreadyExists)
   751	            else
   752	            {
   753	               LogMessage("Warning, binding already exists for \"%s\".\n", pConfigLineLeftSide);
   754	            }
   755	
   756	            // If successful thus far, parse the right side string.
   757	            if(SUCCESS == nRetValue)
   758	            {
   759	               pSeekPtr = pConfigLineRightSide;
   760	
   761	               configFileTokenBuffer[0] = '\0';
   762	               nRetValue = CopyNextToken(&pSeekPtr, configFileTokenBuffer, CONFIG_FILE_TOKEN_BUFFER_LENGTH);
   763	            } // end if(SUCCESS == nRetValue)
   764	
   765	            if(SUCCESS == nRetValue)
   766	            {
   767	               actionTokenMapping.pString    = NULL;
   768	               actionTokenMapping.uConstant  = 0;
   769	               actionTokenMapping.stringType = INVALID_STRING;
   770	               nRetValue = GetTokenMapping(configFileTokenBuffer, &actionTokenMapping);
   771	               if(SUCCESS != nRetValue)
   772	               {
   773	                  LogMessage("Error, unknown config string \"%s\".\nSee README.TXT for valid strings.\n",
   774	                             configFileTokenBuffer);
   775	               }
   776	            } // end if(SUCCESS == nRetValue)
   777	
   778	            if(SUCCESS == nRetValue)
   779	            {
   780	               if(KEYBOARD_KEY_STRING == actionTokenMapping.stringType)
   781	               {
   782	                  actionBinding.bindingType  = BINDING_SIMULATED_KEY;
   783	                  actionBinding.uActionCode  = actionTokenMapping.uConstant;
   784	
   785	                  // Since this is a binding to a simulated key, check for
   786	                  // modifiers that are compatible with simulated keys.
   787	
   788	                  while(('\0' != (*pSeekPtr))  &&
   789	                        (SUCCESS == nRetValue))
   790	                  {
   791	                     configFileTokenBuffer[0] = '\0';
   792	                     nRetValue = CopyNextToken(&pSeekPtr, configFileTokenBuffer, CONFIG_FILE_TOKEN_BUFFER_LENGTH);
   793	                     if(SUCCESS == nRetValue)
   794	                     {
   795	                        modifierTokenMapping.pString    = NULL;
   796	                        modifierTokenMapping.uConstant  = 0;
   797	                        modifierTokenMapping.stringType = INVALID_STRING;
   798	                        nRetValue = GetTokenMapping(configFileTokenBuffer, &modifierTokenMapping);
   799	                        // Check modifier compatibility.
   800	                        if(KEYBOARD_MODIFIER_STRING == modifierTokenMapping.stringType)
   801	                        {
   802	                           // Add the modifier to the current action binding.
   803	                           actionBinding.uActionModifiers |= modifierTokenMapping.uConstant;
   804	                        }
   805	                        else
   806	                        {
   807	                           LogMessage("Error, expected keyboard key modifier but found \"%s\".\n",
   808	                                      configFileTokenBuffer);
   809	                           nRetValue = FAILURE;
   810	                        }
   811	                     } // end if(SUCCESS == nRetValue)
   812	                  } // end while parsing through source action modifiers
   813	               } // end if(KEYBOARD_KEY_STRING == actionTokenMapping.stringType)
   814	               else if(MOUSE_ACTION_STRING == actionTokenMapping.stringType)
   815	               {
   816	                  actionBinding.bindingType  = BINDING_SIMULATED_MOUSE_ACTION;
   817	                  actionBinding.uActionCode  = actionTokenMapping.uConstant;
   818	
   819	                  if('\0' != (*pSeekPtr))
   820	                  {
   821	                     LogMessage("Error, no modifiers are supported for mouse actions.\n");
   822	                     nRetValue = FAILURE;
   823	                  }
   824	               } // end else if(MOUSE_ACTION_STRING == actionTokenMapping.stringType)
   825	               else if(SPECIAL_ACTION_STRING == actionTokenMapping.stringType)
   826	               {
   827	                  actionBinding.bindingType  = BINDING_SPECIAL;
   828	                  actionBinding.uActionCode  = actionTokenMapping.uConstant;
   829	
   830	                  if('\0' != (*pSeekPtr))
   831	                  {
   832	                     LogMessage("Error, no modifiers are supported for special actions.\n");
   833	                     nRetValue = FAILURE;
   834	                  }
   835	               } // end else if(SPECIAL_ACTION_STRING == actionTokenMapping.stringType)
   836	               else
   837	               {
   838	                  LogMessage("\"%s\" is not a keyboard key, mouse, or special binding.\n", configFileTokenBuffer);
   839	                  nRetValue = FAILURE;
   840	               }
   841	            } // end if(SUCCESS == nRetValue)
   842	
   843	            if((SUCCESS == nRetValue) &&
   844	               (NULL != pNewBinding))
   845	            {
   846	               // Copy the binding data.
   847	               pNewBinding->uSourceActionModifiers = actionBinding.uSourceActionModifiers;
   848	               pNewBinding->bindingType            = actionBinding.bindingType;
   849	               pNewBinding->uActionModifiers       = actionBinding.uActionModifiers;
   850	               pNewBinding->uActionCode            = actionBinding.uActionCode;
   851	/*
   852	               LogMessage("New binding:  0x%x, 0x%x, 0x%x, 0x%x\n",
   853	                          pNewBinding->uSourceActionModifiers,
   854	                          pNewBinding->bindingType,
   855	                          pNewBinding->uActionModifiers,
   856	                          pNewBinding->uActionCode);
   857	*/
   858	            }
   859	         } // end if(SUCCESS == nRetValue)
   860	      } // end if(CHATPAD_KEY_STRING == sourceTokenMapping.stringType)
   861	      else if(CONTROLLER_BUTTON_STRING == sourceTokenMapping.stringType)
   862	      {
   863	         // Since this is a controller button binding, check for modifiers that are
   864	         // compatible with controller buttons.
   865	
   866	         while(('\0' != (*pSeekPtr))  &&
   867	               (SUCCESS == nRetValue))
   868	         {
   869	            configFileTokenBuffer[0] = '\0';
   870	            nRetValue = CopyNextToken(&pSeekPtr, configFileTokenBuffer, CONFIG_FILE_TOKEN_BUFFER_LENGTH);
   871	            if(SUCCESS == nRetValue)
   872	            {
   873	               modifierTokenMapping.pString    = NULL;
   874	               modifierTokenMapping.uConstant  = 0;
   875	               modifierTokenMapping.stringType = INVALID_STRING;
   876	               nRetValue = GetTokenMapping(configFileTokenBuffer, &modifierTokenMapping);
   877	               // Check modifier compatibility.
   878	               if(CONTROLLER_MODIFIER_STRING == modifierTokenMapping.stringType)
   879	               {
   880	                  // Add the modifier to the current action binding.
   881	                  actionBinding.uSourceActionModifiers |= modifierTokenMapping.uConstant;
   882	               }
   883	               else
   884	               {
   885	                  LogMessage("Error, expected controller button modifier but found \"%s\".\n",
   886	                             configFileTokenBuffer);
   887	                  nRetValue = FAILURE;
   888	               }
   889	            } // end if(SUCCESS == nRetValue)
   890	         } // end while parsing through source action modifiers
   891	
   892	         if(SUCCESS == nRetValue)
   893	         {
   894	            bool                          bBindingAlreadyExists   = false;
   895	            ControllerButtonMapping*      pButtonMapping          = NULL;
   896	            ChatpadDriverActionBinding*   pNewBinding             = NULL;
   897	
   898	            pButtonMapping = &(pConfigStructure->buttonMappings[sourceTokenMapping.uConstant]);
   899	
   900	            if(NULL != pButtonMapping->pBindings)
   901	            {
   902	               // Check whether there is already a binding for the source action.
   903	               for(int nBindingIndex = 0; nBindingIndex < (pButtonMapping->nNumBindings); nBindingIndex++)
   904	               {
   905	                  if(pButtonMapping->pBindings[nBindingIndex].uSourceActionModifiers ==
   906	                     actionBinding.uSourceActionModifiers)
   907	                  {
   908	                     bBindingAlreadyExists = true;
   909	                     pNewBinding = &(pButtonMapping->pBindings[nBindingIndex]);
   910	                     break;
   911	                  }
   912	               }
   913	            }
   914	
   915	            if(false == bBindingAlreadyExists)
   916	            {
   917	               ChatpadDriverActionBinding* pReallocatedBindings = NULL;
   918	
   919	               // An intermediate pointer is used here since realloc could
   920	               // fail and cause the original memory in pButtonMapping->pBindings
   921	               // to leak.
   922	               pReallocatedBindings =
   923	                 static_cast<ChatpadDriverActionBinding*>(realloc(
   924	                   pButtonMapping->pBindings,
   925	                   ((pButtonMapping->nNumBindings + 1) * sizeof(ChatpadDriverActionBinding))));
   926	               if(NULL != pReallocatedBindings)
   927	               {
   928	                  pNewBinding = &(pReallocatedBindings[pButtonMapping->nNumBindings]);
   929	
   930	                  pButtonMapping->nNumBindings++;
   931	                  pButtonMapping->pBindings = pReallocatedBindings;
   932	               } // end if(NULL != pReallocatedBindings)
   933	               else
   934	               {
   935	                  LogMessage("Error allocating memory in HandleConfigMappingLine.\n");
   936	                  nRetValue = FAILURE;
   937	               }
   938	            } // end if(false == bBindingAlreadyExists)
   939	            else
   940	            {
   941	               LogMessage("Warning, binding already exists for \"%s\".\n", pConfigLineLeftSide);
   942	            }
   943	
   944	            // If successful thus far, parse the right side string.
   945	            if(SUCCESS == nRetValue)
   946	            {
   947	               pSeekPtr = pConfigLineRightSide;
   948	
   949	               configFileTokenBuffer[0] = '\0';
   950	               nRetValue = CopyNextToken(&pSeekPtr, configFileTokenBuffer, CONFIG_FILE_TOKEN_BUFFER_LENGTH);
   951	            } // end if(SUCCESS == nRetValue)
   952	
   953	            if(SUCCESS == nRetValue)
   954	            {
   955	               actionTokenMapping.pString    = NULL;
   956	               actionTokenMapping.uConstant  = 0;
   957	               actionTokenMapping.stringType = INVALID_STRING;
   958	               nRetValue = GetTokenMapping(configFileTokenBuffer, &actionTokenMapping);
   959	               if(SUCCESS != nRetValue)
   960	               {
   961	                  LogMessage("Error, unknown config string \"%s\".\nSee README.TXT for valid strings.\n",
   962	                             configFileTokenBuffer);
   963	               }
   964	            } // end if(SUCCESS == nRetValue)
   965	
   966	            if(SUCCESS == nRetValue)
   967	            {
   968	               if(CONTROLLER_BUTTON_STRING == actionTokenMapping.stringType)
   969	               {
   970	                  actionBinding.bindingType  = BINDING_CONTROLLER;
   971	                  actionBinding.uActionCode  = actionTokenMapping.uConstant;
   972	
   973	                  if('\0' != (*pSeekPtr))
   974	                  {
   975	                     LogMessage("Error, no modifiers are supported for output controller actions.\n");
   976	                     nRetValue = FAILURE;
   977	                  }
   978	               } // end if(CONTROLLER_BUTTON_STRING == actionTokenMapping.stringType)
   979	               else if(KEYBOARD_KEY_STRING == actionTokenMapping.stringType)
   980	               {
   981	                  actionBinding.bindingType  = BINDING_SIMULATED_KEY;
   982	                  actionBinding.uActionCode  = actionTokenMapping.uConstant;
   983	
   984	                  // Since this is a binding to a simulated key, check for
   985	                  // modifiers that are compatible with simulated keys.
   986	
   987	                  while(('\0' != (*pSeekPtr))  &&
   988	                        (SUCCESS == nRetValue))
   989	                  {
   990	                     configFileTokenBuffer[0] = '\0';
   991	                     nRetValue = CopyNextToken(&pSeekPtr, configFileTokenBuffer, CONFIG_FILE_TOKEN_BUFFER_LENGTH);
   992	                     if(SUCCESS == nRetValue)
   993	                     {
   994	                        modifierTokenMapping.pString    = NULL;
   995	                        modifierTokenMapping.uConstant  = 0;
   996	                        modifierTokenMapping.stringType = INVALID_STRING;
   997	                        nRetValue = GetTokenMapping(configFileTokenBuffer, &modifierTokenMapping);
   998	                        // Check modifier compatibility.
   999	                        if(KEYBOARD_MODIFIER_STRING == modifierTokenMapping.stringType)
  1000	                        {
  1001	                           // Add the modifier to the current action binding.
  1002	                           actionBinding.uActionModifiers |= modifierTokenMapping.uConstant;
  1003	                        }
  1004	                        else
  1005	                        {
  1006	                           LogMessage("Error, expected keyboard key modifier but found \"%s\".\n",
  1007	                                      configFileTokenBuffer);
  1008	                           nRetValue = FAILURE;
  1009	                        }
  1010	                     } // end if(SUCCESS == nRetValue)
  1011	                  } // end while parsing through source action modifiers
  1012	               } // end if(KEYBOARD_KEY_STRING == actionTokenMapping.stringType)
  1013	               else if(MOUSE_ACTION_STRING == actionTokenMapping.stringType)
  1014	               {
  1015	                  actionBinding.bindingType  = BINDING_SIMULATED_MOUSE_ACTION;
  1016	                  actionBinding.uActionCode  = actionTokenMapping.uConstant;
  1017	
  1018	                  if('\0' != (*pSeekPtr))
  1019	                  {
  1020	                     LogMessage("Error, no modifiers are supported for mouse actions.\n");
  1021	                     nRetValue = FAILURE;
  1022	                  }
  1023	               } // end else if(MOUSE_ACTION_STRING == actionTokenMapping.stringType)
  1024	               else if(SPECIAL_ACTION_STRING == actionTokenMapping.stringType)
  1025	               {
  1026	                  actionBinding.bindingType  = BINDING_SPECIAL;
  1027	                  actionBinding.uActionCode  = actionTokenMapping.uConstant;
  1028	
  1029	                  if('\0' != (*pSeekPtr))
  1030	                  {
  1031	                     LogMessage("Error, no modifiers are supported for special actions.\n");
  1032	                     nRetValue = FAILURE;
  1033	                  }
  1034	               } // end else if(SPECIAL_ACTION_STRING == actionTokenMapping.stringType)
  1035	               else
  1036	               {
  1037	                  LogMessage("\"%s\" is not a controller, keyboard key, mouse, or special binding.\n", configFileTokenBuffer);
  1038	                  nRetValue = FAILURE;
  1039	               }
  1040	            } // end if(SUCCESS == nRetValue)
  1041	
  1042	            if((SUCCESS == nRetValue) &&
  1043	               (NULL != pNewBinding))
  1044	            {
  1045	               // Copy the binding data.
  1046	               pNewBinding->uSourceActionModifiers = actionBinding.uSourceActionModifiers;
  1047	               pNewBinding->bindingType            = actionBinding.bindingType;
  1048	               pNewBinding->uActionModifiers       = actionBinding.uActionModifiers;
  1049	               pNewBinding->uActionCode            = actionBinding.uActionCode;
  1050	/*
  1051	               LogMessage("New binding:  0x%x, 0x%x, 0x%x, 0x%x\n",
  1052	                          pNewBinding->uSourceActionModifiers,
  1053	                          pNewBinding->bindingType,
  1054	                          pNewBinding->uActionModifiers,
  1055	                          pNewBinding->uActionCode);
  1056	*/
  1057	            }
  1058	         } // end if(SUCCESS == nRetValue)
  1059	      } // end else if(CONTROLLER_BUTTON_STRING == sourceTokenMapping.stringType)
  1060	      else
  1061	      {
  1062	         LogMessage("\"%s\" is neither a chatpad key nor a controller button.\n", configFileTokenBuffer);
  1063	         nRetValue = FAILURE;
  1064	      }
  1065	   } // end if(SUCCESS == nRetValue)
  1066	
  1067	   return nRetValue;
  1068	} // end HandleConfigMappingLine

  1103	// This function opens a config file, initializes pConfigStructure to a default
  1104	// state, parses the file, writes data into pConfigStructure as needed, and
  1105	// closes the file.
  1106	int LoadConfigFile(
  1107	  char*                          pFilename,
  1108	  ChatpadDriverConfigStructure*  pConfigStructure)
  1109	{
  1110	   int      nRetValue                  = SUCCESS;
  1111	   bool     bFinished                  = false;
  1112	   FILE*    pReadFile                  = NULL;
  1113	   // Allow for 2048 characters in a command string plus a null terminator.
  1114	   int      nConfigLineBufferSize      = 2048 + 1;
  1115	   char*    pConfigLine                = NULL;
  1116	   char*    pConfigLineLeftSide        = NULL;
  1117	   char*    pConfigLineRightSide       = NULL;
  1118	   char*    pReadResult                = NULL;
  1119	   char*    pSeekPtr                   = NULL;
  1120	   char*    pConfigLineLeftPtr         = NULL;
  1121	   char*    pConfigLineRightPtr        = NULL;
  1122	
  1123	   if((SUCCESS == nRetValue) &&
  1124	      ((NULL == pFilename)             ||
  1125	       (NULL == pConfigStructure)))
  1126	   {
  1127	      LogMessage("Invalid parameters provided to LoadConfigFile.\n");
  1128	      nRetValue = FAILURE;
  1129	   }
  1130	
  1131	   if(SUCCESS == nRetValue)
  1132	   {
  1133	      pReadFile = fopen(pFilename, "rb");
  1134	      if(NULL == pReadFile)
  1135	      {
  1136	         LogMessage("Failed to open config file \"%s\".\n", pFilename);
  1137	         nRetValue = FAILURE;
  1138	      }
  1139	   }
  1140	
  1141	   if(SUCCESS == nRetValue)
  1142	   {
  1143	      nRetValue = InitDefaultDriverConfigDataValues(pConfigStructure);
  1144	   }
  1145	
  1146	   if(SUCCESS == nRetValue)
  1147	   {
  1148	      pConfigLine = static_cast<char*>(malloc(nConfigLineBufferSize));
  1149	      pConfigLineLeftSide = static_cast<char*>(malloc(nConfigLineBufferSize));
  1150	      pConfigLineRightSide = static_cast<char*>(malloc(nConfigLineBufferSize));
  1151	      if((NULL == pConfigLine)            ||
  1152	         (NULL == pConfigLineLeftSide)    ||
  1153	         (NULL == pConfigLineRightSide))
  1154	      {
  1155	         LogMessage("Failed to allocate memory for config line buffers.\n");
  1156	         nRetValue = FAILURE;
  1157	      }
  1158	   }
  1159	
  1160	   while((SUCCESS == nRetValue) &&
  1161	         (false == bFinished))
  1162	   {
  1163	      pConfigLine[0] = '\0';
  1164	      pReadResult = fgets(pConfigLine, nConfigLineBufferSize, pReadFile);
  1165	
  1166	      if(NULL != pReadResult)
  1167	      {
  1168	         pSeekPtr = pConfigLine;
  1169	
  1170	         // Skip initial whitespace.
  1171	         pSeekPtr = SkipWhiteSpace(pSeekPtr);
  1172	
  1173	         // Ignore the line if it is empty or starts with '#'.
  1174	         if(('#' == (*pSeekPtr))    ||
  1175	            ('\n' == (*pSeekPtr))   ||
  1176	            ('\r' == (*pSeekPtr))   ||
  1177	            ('\0' == (*pSeekPtr)))
  1178	         {
  1179	            continue;
  1180	         }
  1181	
  1182	         // Copy the left side of the '=' sign that should exist into a buffer.
  1183	         pConfigLineLeftSide[0] = '\0';
  1184	         pConfigLineLeftPtr = pConfigLineLeftSide;
  1185	         while(('=' != (*pSeekPtr))    &&
  1186	               ('\0' != (*pSeekPtr)))
  1187	         {
  1188	            (*pConfigLineLeftPtr) = (*pSeekPtr);
  1189	            pConfigLineLeftPtr++;
  1190	            pSeekPtr++;
  1191	         }
  1192	
  1193	         if('=' == (*pSeekPtr))
  1194	         {
  1195	            // Skip backwards over any ending whitespace.
  1196	            if(pConfigLineLeftPtr > pConfigLineLeftSide)
  1197	            {
  1198	               pConfigLineLeftPtr--;
  1199	               pConfigLineLeftPtr = SkipEndingWhiteSpace(pConfigLineLeftSide, pConfigLineLeftPtr);
  1200	               pConfigLineLeftPtr++;
  1201	            }
  1202	            // Null-terminate the left-side string.
  1203	            (*pConfigLineLeftPtr) = '\0';
  1204	         }
  1205	         else
  1206	         {
  1207	            LogMessage("Error, missing '=' in the following line:\n%s", pConfigLine);
  1208	            nRetValue = FAILURE;
  1209	            continue;
  1210	         }
  1211	
  1212	         //***
  1213	         // Copy the right side of the '=' sign into a buffer.  If this point
  1214	         // in the code was reached, then pSeekPtr currently points to '='.
  1215	         //***
  1216	         pSeekPtr++;
  1217	
  1218	         // Now that we are past the '=' character, skip any initial whitespace.
  1219	         pSeekPtr = SkipWhiteSpace(pSeekPtr);
  1220	
  1221	         pConfigLineRightSide[0] = '\0';
  1222	         pConfigLineRightPtr = pConfigLineRightSide;
  1223	         while(('\n' != (*pSeekPtr))    &&
  1224	               ('\r' != (*pSeekPtr))    &&
  1225	               ('\0' != (*pSeekPtr)))
  1226	         {
  1227	            (*pConfigLineRightPtr) = (*pSeekPtr);
  1228	            pConfigLineRightPtr++;
  1229	            pSeekPtr++;
  1230	         }
  1231	
  1232	         // Skip backwards over any ending whitespace.
  1233	         if(pConfigLineRightPtr > pConfigLineRightSide)
  1234	         {
  1235	            pConfigLineRightPtr--;
  1236	            pConfigLineRightPtr = SkipEndingWhiteSpace(pConfigLineRightSide, pConfigLineRightPtr);
  1237	            pConfigLineRightPtr++;
  1238	         }
  1239	         // Null-terminate the right-side string.
  1240	         (*pConfigLineRightPtr) = '\0';
  1241	
  1242	         // Convert both left and right sides to all-uppercase.
  1243	         StringToUpper(pConfigLineLeftSide);
  1244	         StringToUpper(pConfigLineRightSide);
  1245	
  1246	         // Try to handle this as a config variable assignment.  If there is
  1247	         // a syntax error or the left side does not name a config variable,
  1248	         // then HandleConfigVariableAssignment will return FAILURE.
  1249	         // We can then try to handle it as a key/control mapping instead.
  1250	         if(FAILURE == HandleConfigVariableAssignment(pConfigStructure, pConfigLineLeftSide, pConfigLineRightSide))
  1251	         {
  1252	            nRetValue = HandleConfigMappingLine(pConfigStructure, pConfigLineLeftSide, pConfigLineRightSide);
  1253	            if(SUCCESS != nRetValue)
  1254	            {
  1255	               LogMessage("There was an error parsing the following config mapping line:\n%s", pConfigLine);
  1256	            }
  1257	         }
  1258	      } // end if(NULL != pReadResult)
  1259	      else
  1260	      {
  1261	         // This case indicates an error or the end of the file.
  1262	         bFinished = true;
  1263	         if(0 != ferror(pReadFile))
  1264	         {
  1265	            LogMessage("An error occurred while reading the config file.\n");
  1266	            nRetValue = FAILURE;
  1267	         }
  1268	      }
  1269	   } // end while successful and not finished
  1270	
  1271	   if(NULL == pConfigLine)
  1272	   {
  1273	      free(pConfigLine);
  1274	      pConfigLine = NULL;
  1275	   }
  1276	
  1277	   if(NULL == pConfigLineLeftSide)
  1278	   {
  1279	      free(pConfigLineLeftSide);
  1280	      pConfigLineLeftSide = NULL;
  1281	   }
  1282	
  1283	   if(NULL == pConfigLineRightSide)
  1284	   {
  1285	      free(pConfigLineRightSide);
  1286	      pConfigLineRightSide = NULL;
  1287	   }
  1288	
  1289	   if(NULL != pReadFile)
  1290	   {
  1291	      fclose(pReadFile);
  1292	      pReadFile = NULL;
  1293	   }
  1294	
  1295	   if(SUCCESS != nRetValue)
  1296	   {
  1297	      LogMessage("There was an error while attempting to load config file \"%s\".\n", pFilename);
  1298	   }
  1299	
  1300	   return nRetValue;
  1301	} // end LoadConfigFile
