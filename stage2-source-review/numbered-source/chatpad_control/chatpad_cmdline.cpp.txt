    12	#include <windows.h>
    13	#include <stdio.h>
    14	#include <stdlib.h>
    15	
    16	#include "chatpad_control.h"
    17	
    18	// TODO move elsewhere
    19	#define CHATPAD_CONTROL_UTILITY_VERSION      "0.1a"
    20	
    21	
    22	//***
    23	// Functions
    24	//***
    25	
    26	int __cdecl main(int argc, char* argv[])
    27	{
    28	   int nRetValue = SUCCESS;
    29	
    30	   printf("\n*****\n");
    31	   printf("Chatpad control utility version %s started.\n", CHATPAD_CONTROL_UTILITY_VERSION);
    32	   printf("*****\n");
    33	
    34	   if((2 == argc) && (0 == strcmp(argv[1], "--nodelay")))
    35	   {
    36	      printf("Control utility started with no delay.\n");
    37	   }
    38	   else
    39	   {
    40	      printf("Please wait...\n");
    41	      printf("\n");
    42	      Sleep(5000);
    43	      printf("Control utility started.\n");
    44	   }
    45	
    46	   nRetValue = ChatpadControlMainLoop();
    47	
    48	   printf("Done.  The exit status is %d.\n", nRetValue);
    49	   return nRetValue;
    50	}
