    85	
    86	// IMPORTANT NOTE:
    87	//
    88	// Last time I checked (11/25/2010), the below structure was less than 1 KB in
    89	// size.  If this structure gets much over a kilobyte in size, I may want to
    90	// refactor it so that it includes one or more pointers to allocated data
    91	// instead of putting so much data in this structure itself.
    92	//
    93	// See http://thedailyreviewer.com/windowsos/view/size-of-device-context-limit-107123784
    94	// for information that suggests there may be a device context limit that may not even
    95	// be documented, somewhere less than 64 KB.
    96	
    97	// Structure for a device context for each device.
    98	typedef struct _DEVICE_CONTEXT
    99	{
   100	   // TODO remove magicNumber and serialNumber if unused?
   101	   ULONG                         magicNumber;
   102	   ULONG                         serialNumber;
   103	   // TODO add descriptive comments for these two variables
   104	   WDFIOTARGET                   targetForRequests;
   105	   WDFUSBDEVICE                  usbDevice;
   106	   // Whether the USB device configuration has been selected yet.
   107	   BOOLEAN                       usbConfigSelected;
   108	   // Whether we think the Microsoft driver has finished initializing the 360 controller.
   109	   BOOLEAN                       msInitFinished;
   110	   // Whether we have finished initializing the chatpad successfully.
   111	   // TODO eventually remove chatpadInitFinished if is not used and needed
   112	   BOOLEAN                       chatpadInitFinished;
   113	   // The current filter mode.  This value must be one of the FILTER_MODE_ constants.
   114	   ULONG                         filterMode;
   115	   // Whether the controls continuous reader is started.
   116	   BOOLEAN                       controlsReaderStarted;
   117	
   118	   // Queue used to park controls data read requests from the top-level driver.
   119	   WDFQUEUE                      controlsParkedDriverReadRequestQueue;
   120	   // Queue used to park controls data read requests from the user.
   121	   WDFQUEUE                      controlsParkedUserReadRequestQueue;
   122	
   123	   // Queue used to park chatpad data read requests from the user.
   124	   WDFQUEUE                      chatpadParkedUserReadRequestQueue;
   125	
   126	   // Number of valid interface information structures in usbdInterfaceInformation.
   127	   BYTE                          numInterfaces;
   128	   // Array of dynamically-allocated structures containing saved information
   129	   // about interfaces so that URB_FUNCTION_SELECT_INTERFACE from the top-level
   130	   // driver can be handled (as long as only the default 0 alternate interface
   131	   // is selected).
   132	   PUSBD_INTERFACE_INFORMATION   usbdInterfaceInformation[CHATPAD_DRIVER_MAX_SUPPORTED_NUM_INTERFACES];
   133	
   134	   // Number of valid pipe information structures in wdfUsbPipeInfo and WDF pipe handles in wdfUsbPipe.
   135	   BYTE                          numPipes;
   136	   // Pipe information for each pipe.
   137	   WDF_USB_PIPE_INFORMATION      wdfUsbPipeInfo[CHATPAD_DRIVER_MAX_SUPPORTED_NUM_PIPES];
   138	   // Pipe handles for use in reading and writing data.
   139	   WDFUSBPIPE                    wdfUsbPipe[CHATPAD_DRIVER_MAX_SUPPORTED_NUM_PIPES];
   140	
   141	   // For the fields below, the "controls" endpoint refers to the endpoint that returns data
   142	   // about (potentially among other things) XBox 360 controller thumbstick and button data.
   143	
   144	   // TODO combine some of the below values for chatpad and top-level requests into
   145	   //   custom structures of their own to make things cleaner here?
   146	
   147	   // Memory containing a write buffer for writing data to the controls endpoint.
   148	   WDFMEMORY                     controlsWriteBufferMemory;
   149	   // Memory containing an URB for writing data to the controls endpoint.
   150	   WDFMEMORY                     controlsWriteUrbMemory;
   151	   // Request for writing data to the controls endpoint.
   152	   WDFREQUEST                    controlsWriteRequest;
   153	
   154	   // Memory containing read buffers for reading controls endpoint data from the XBox 360 controller.
   155	   WDFMEMORY                     controlsReadBufferMemory[CHATPAD_MAX_NUM_PENDING_USB_REQUESTS];
   156	   // Memory containing URBs for reading controls endpoint data from the XBox 360 controller.
   157	   WDFMEMORY                     controlsReadUrbMemory[CHATPAD_MAX_NUM_PENDING_USB_REQUESTS];
   158	   // Requests for reading controls endpoint data from the XBox 360 controller.
   159	   WDFREQUEST                    controlsReadRequests[CHATPAD_MAX_NUM_PENDING_USB_REQUESTS];
   160	   // Tracks whether each controls endpoint data read request is in use.  If this value is FALSE
   161	   // for a particular request, then that request can be formatted and sent.
   162	   // TODO do I need some sort of synchronization around this, or is the way I have callbacks
   163	   //   set up sufficient?
   164	   BOOLEAN                       controlsReadRequestsInUse[CHATPAD_MAX_NUM_PENDING_USB_REQUESTS];
   165	
   166	   // Memory containing read buffers for reading chatpad endpoint data from the XBox 360 controller.
   167	   WDFMEMORY                     chatpadReadBufferMemory[CHATPAD_MAX_NUM_PENDING_USB_REQUESTS];
   168	   // Memory containing URBs for reading chatpad endpoint data from the XBox 360 controller.
   169	   WDFMEMORY                     chatpadReadUrbMemory[CHATPAD_MAX_NUM_PENDING_USB_REQUESTS];
   170	   // Requests for reading chatpad endpoint data from the XBox 360 controller.
   171	   WDFREQUEST                    chatpadReadRequests[CHATPAD_MAX_NUM_PENDING_USB_REQUESTS];
   172	   // Tracks whether each chatpad endpoint data read request is in use.  If this value is FALSE
   173	   // for a particular request, then that request can be formatted and sent.
   174	   BOOLEAN                       chatpadReadRequestsInUse[CHATPAD_MAX_NUM_PENDING_USB_REQUESTS];
   175	} DEVICE_CONTEXT, *PDEVICE_CONTEXT;
   176
