    40	// Called when driver is first loaded.
    41	extern "C"
    42	NTSTATUS
    43	DriverEntry(
    44	  PDRIVER_OBJECT  driverObject,
    45	  PUNICODE_STRING registryPath)
    46	{
    47	   NTSTATUS                retStatus = STATUS_SUCCESS;
    48	   WDF_DRIVER_CONFIG       driverConfig;
    49	
    50	   ChatpadTrace(("XBox 360 Controller Chatpad Super Driver by GAFBlizzard.\n"));
    51	   ChatpadTrace(("Build %s %s\n", __DATE__, __TIME__));
    52	
    53	   // This is just an information printout so I can check the size of a DEVICE_CONTEXT structure
    54	   // (only the structure itself, not any additional allocated memory).
    55	   ChatpadTrace(("sizeof(DEVICE_CONTEXT) == %u\n", sizeof(DEVICE_CONTEXT)));
    56	
    57	   WDF_DRIVER_CONFIG_INIT(&driverConfig,
    58	                          ChatpadFilterEvtDeviceAdd);
    59	   // Set up the driver unload function.
    60	   driverConfig.EvtDriverUnload = DriverUnload;
    61	
    62	   retStatus = WdfDriverCreate(
    63	     driverObject,
    64	     registryPath,
    65	     WDF_NO_OBJECT_ATTRIBUTES,
    66	     &driverConfig,
    67	     NULL);
    68	   if(!NT_SUCCESS(retStatus))
    69	   {
    70	      // TODO why double parentheses?  Macro issues?
    71	      ChatpadTrace(("WdfDriverCreate failed:  0x%0x\n", retStatus));
    72	   }
    73	
    74	   // Create collection for different instances of the physical device.
    75	   if(NT_SUCCESS(retStatus))
    76	   {
    77	      retStatus = WdfCollectionCreate(
    78	        WDF_NO_OBJECT_ATTRIBUTES,
    79	        &chatpadDeviceCollection);
    80	      if(!NT_SUCCESS(retStatus))
    81	      {
    82	         ChatpadTrace(("WdfCollectionCreate failed:  0x%x\n", retStatus));
    83	      }
    84	   }
    85	
    86	   if(NT_SUCCESS(retStatus))
    87	   {
    88	      retStatus = WdfWaitLockCreate(
    89	        WDF_NO_OBJECT_ATTRIBUTES,
    90	        &chatpadDeviceCollectionLock);
    91	      if(!NT_SUCCESS(retStatus))
    92	      {
    93	         ChatpadTrace(("WdfWaitLockCreate failed:  0x%x\n", retStatus));
    94	      }
    95	   }
    96	
    97	   return retStatus;
    98	} // end DriverEntry

   113	// Utility method to free any allocated usbdInterfaceInformation pointers
   114	// and mark the number of interfaces as 0 in devContext.
   115	//
   116	// IMPORTANT NOTE:  Any needed synchronization must be done outside of
   117	//   this function.  Let the caller beware, and not call this in multiple
   118	//   threads simultaneously or anything like that.
   119	// TODO is there any risk of synchronization issues?  I assume not
   120	//   since other things should not be called once the device context
   121	//   cleanup function is called, but I may want to make sure.
   122	NTSTATUS
   123	ReleaseCachedInterfaceInformation(PDEVICE_CONTEXT devContext)
   124	{
   125	   NTSTATUS retStatus = STATUS_SUCCESS;
   126	
   127	   for(BYTE i = 0; i < devContext->numInterfaces; i++)
   128	   {
   129	      if(NULL != devContext->usbdInterfaceInformation[i])
   130	      {
   131	         ExFreePoolWithTag(devContext->usbdInterfaceInformation[i], 'zilB');
   132	         devContext->usbdInterfaceInformation[i] = NULL;
   133	      }
   134	   }
   135	
   136	   return retStatus;
   137	} // end ReleaseCachedInterfaceInformation

   192	// NOTE:  This function must not do anything that could cause a page fault, according to the osrusbfx2 example.
   193	NTSTATUS
   194	ChatpadFilterD0Entry(
   195	  IN WDFDEVICE                wdfDevice,
   196	  IN WDF_POWER_DEVICE_STATE   previousState)
   197	{
   198	   NTSTATUS          retStatus   = STATUS_SUCCESS;
   199	   PDEVICE_CONTEXT   devContext  = NULL;
   200	
   201	   devContext = ChatpadFilterGetDeviceContext(wdfDevice);
   202	   if(NULL != devContext)
   203	   {
   204	      ChatpadTrace(("D0 entry occurred.\n"));
   205	
   206	      // TODO is this safe?  Are there any synchronizations problem with using the below lines?
   207	      WdfWaitLockAcquire(chatpadDeviceCollectionLock, NULL);
   208	      chatpadControlDeviceAvailable = TRUE;
   209	      WdfWaitLockRelease(chatpadDeviceCollectionLock);
   210	   } // end if(NULL != devContext)
   211	   else
   212	   {
   213	      ChatpadTrace(("Unable to obtain device context.\n"));
   214	      retStatus = STATUS_INTERNAL_ERROR;
   215	   }
   216	
   217	   return retStatus;
   218	} // end ChatpadFilterD0Entry

   246	NTSTATUS
   247	ChatpadFilterEvtDeviceAdd(
   248	  IN WDFDRIVER       driver,
   249	  IN PWDFDEVICE_INIT deviceInit)
   250	{
   251	   NTSTATUS                      retStatus = STATUS_SUCCESS;
   252	   PDEVICE_CONTEXT               devContext = NULL;
   253	   //ULONG                         serialNumber = 0;
   254	   //ULONG                         returnSize = 0;
   255	   WDF_OBJECT_ATTRIBUTES         wdfObjectAttr;
   256	   WDFDEVICE                     wdfDevice;
   257	   WDF_IO_QUEUE_CONFIG           ioQueueConfig;
   258	   WDF_PNPPOWER_EVENT_CALLBACKS  powerCallbacks;
   259	
   260	   // TODO:  Find some way to get the serial number, or otherwise identify the device, since trying
   261	   //        to get the DevicePropertyUINumber fails (STATUS_OBJECT_NAME_NOT_FOUND) on my Vista 64 system, at least.
   262	   /*
   263	   // Get device serial number.
   264	   retStatus = WdfFdoInitQueryProperty(
   265	     deviceInit,
   266	     DevicePropertyUINumber,
   267	     sizeof(serialNumber),
   268	     &serialNumber,
   269	     &returnSize);
   270	   */
   271	
   272	   if(STATUS_SUCCESS == retStatus)
   273	   {
   274	      // Mark ourselves as a filter, causing requests we don't care about to be
   275	      // passed along etc.
   276	      // 
   277	      // By the way, apparently FDO means "Framework Device Object".
   278	      WdfFdoInitSetFilter(deviceInit);
   279	
   280	      // Mark our device type as unknown.
   281	      // TODO should this be set eventually since this driver is not generic, or is the
   282	      //      automatic stuff fine?
   283	      WdfDeviceInitSetDeviceType(deviceInit, FILE_DEVICE_UNKNOWN);
   284	
   285	      // Register a prepare hardware callback to create a WDF USB device object.
   286	      WDF_PNPPOWER_EVENT_CALLBACKS_INIT(&powerCallbacks);
   287	      powerCallbacks.EvtDevicePrepareHardware = ChatpadFilterEvtPrepareHardware;
   288	      powerCallbacks.EvtDeviceD0Entry = ChatpadFilterD0Entry;
   289	      powerCallbacks.EvtDeviceD0Exit = ChatpadFilterD0Exit;
   290	      powerCallbacks.EvtDeviceSurpriseRemoval = ChatpadFilterEvtSurpriseRemoval;
   291	      powerCallbacks.EvtDeviceQueryRemove = ChatpadFilterEvtQueryRemove;
   292	
   293	      WdfDeviceInitSetPnpPowerEventCallbacks(
   294	        deviceInit,
   295	        &powerCallbacks);
   296	
   297	      WDF_OBJECT_ATTRIBUTES_INIT_CONTEXT_TYPE(&wdfObjectAttr, DEVICE_CONTEXT);
   298	
   299	      // Register for cleanup notification.
   300	      wdfObjectAttr.EvtCleanupCallback = ChatpadFilterEvtDeviceContextCleanup;
   301	
   302	      // Create framework device object, attach to the lower stack etc.
   303	      retStatus = WdfDeviceCreate(
   304	        &deviceInit,
   305	        &wdfObjectAttr,
   306	        &wdfDevice);
   307	   }
   308	   /*
   309	   else
   310	   {
   311	      ChatpadTrace(("Failed to get serial number:  0x%x\n", retStatus));
   312	   }
   313	   */
   314	
   315	   if(STATUS_SUCCESS == retStatus)
   316	   {
   317	      devContext = ChatpadFilterGetDeviceContext(wdfDevice);
   318	      if(NULL != devContext)
   319	      {
   320	         // Initialize DEVICE_CONTEXT values.
   321	         ChatpadTrace(("Initializing device context.\n"));
   322	
   323	         devContext->magicNumber = DEVICE_CONTEXT_MAGIC_NUMBER;
   324	         // TODO if I ever support multiple devices, this probably needs to be unique per device!
   325	         //   It would be based on serial number or something similar.
   326	         //devContext->serialNumber = serialNumber;
   327	         devContext->serialNumber = 0x00000042;
   328	         devContext->targetForRequests = WdfDeviceGetIoTarget(wdfDevice);
   329	         devContext->usbDevice = NULL;
   330	         devContext->usbConfigSelected = FALSE;
   331	         devContext->msInitFinished = FALSE;
   332	         devContext->chatpadInitFinished = FALSE;
   333	         // Filter initially until the Microsoft driver controller init is finished.
   334	         devContext->filterMode = FILTER_MODE_FILTERED;
   335	         devContext->numInterfaces = 0;
   336	
   337	         for(BYTE i = 0; i < CHATPAD_DRIVER_MAX_SUPPORTED_NUM_INTERFACES; i++)
   338	         {
   339	            devContext->usbdInterfaceInformation[i] = NULL;
   340	         }
   341	
   342	         devContext->numPipes = 0;
   343	
   344	         for(BYTE i = 0; i < CHATPAD_DRIVER_MAX_SUPPORTED_NUM_PIPES; i++)
   345	         {
   346	            WDF_USB_PIPE_INFORMATION_INIT(&(devContext->wdfUsbPipeInfo[i]));
   347	            devContext->wdfUsbPipe[i] = NULL;
   348	         }
   349	
   350	         // Create requests and memory buffers for writing to and reading from endpoints.
   351	         // I am not sure if these have to be non-pageable, but I am making all request
   352	         // storage non-pageable to be on the safe side especially since the total amount
   353	         // of data I am allocating is presumably pretty small.
   354	
   355	         WDF_OBJECT_ATTRIBUTES requestAttributes;
   356	
   357	         WDF_OBJECT_ATTRIBUTES_INIT(&requestAttributes);
   358	
   359	         // Make the I/O target the parent object of these requests so that
   360	         // they will get deleted when the I/O target is deleted.
   361	         // TODO I may want to do more research and verify that these get
   362	         //   deleted properly, if possible.  Could they ever hang the
   363	         //   driver on shutdown or driver upgrade if they are not pending?
   364	         requestAttributes.ParentObject = devContext->targetForRequests;
   365	
   366	         if(STATUS_SUCCESS == retStatus)
   367	         {
   368	            retStatus = WdfMemoryCreate(
   369	              WDF_NO_OBJECT_ATTRIBUTES,
   370	              NonPagedPool,
   371	              0,
   372	              CONTROLS_WRITE_BUFFER_LENGTH,
   373	              &(devContext->controlsWriteBufferMemory),
   374	              NULL);
   375	            if(STATUS_SUCCESS != retStatus)
   376	            {
   377	               ChatpadTrace(("Allocation failed for controls write buffer memory.\n"));
   378	            }
   379	         }
   380	
   381	         if(STATUS_SUCCESS == retStatus)
   382	         {
   383	            retStatus = WdfMemoryCreate(
   384	              WDF_NO_OBJECT_ATTRIBUTES,
   385	              NonPagedPool,
   386	              0,
   387	              sizeof(struct _URB_BULK_OR_INTERRUPT_TRANSFER),
   388	              &(devContext->controlsWriteUrbMemory),
   389	              NULL);
   390	            if(STATUS_SUCCESS != retStatus)
   391	            {
   392	               ChatpadTrace(("Allocation failed for controls write URB memory.\n"));
   393	            }
   394	         }
   395	
   396	         if(STATUS_SUCCESS == retStatus)
   397	         {
   398	            retStatus = WdfRequestCreate(
   399	              &requestAttributes,
   400	              devContext->targetForRequests,
   401	              &(devContext->controlsWriteRequest));
   402	            if(STATUS_SUCCESS != retStatus)
   403	            {
   404	               ChatpadTrace(("WdfRequestCreate failed:  0x%x\n", retStatus));
   405	            }
   406	         }
   407	
   408	         if(STATUS_SUCCESS == retStatus)
   409	         {
   410	            PURB controlsWriteUrb =
   411	              static_cast<PURB>(WdfMemoryGetBuffer(devContext->controlsWriteUrbMemory, NULL));
   412	            if(NULL != controlsWriteUrb)
   413	            {
   414	               // TODO move this code somewhere common so it will not be redundant and prone to bugs if
   415	               //   one side changes?
   416	               controlsWriteUrb->UrbHeader.Function = URB_FUNCTION_BULK_OR_INTERRUPT_TRANSFER;
   417	               controlsWriteUrb->UrbHeader.Length = sizeof(struct _URB_BULK_OR_INTERRUPT_TRANSFER);
   418	               controlsWriteUrb->UrbBulkOrInterruptTransfer.PipeHandle = NULL;
   419	               controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferFlags =
   420	                 USBD_TRANSFER_DIRECTION_OUT;
   421	               controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferBufferLength =
   422	                 CONTROLS_WRITE_BUFFER_LENGTH;
   423	               controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferBufferMDL = NULL;
   424	               controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferBuffer =
   425	                 WdfMemoryGetBuffer(devContext->controlsWriteBufferMemory, NULL);
   426	               controlsWriteUrb->UrbBulkOrInterruptTransfer.UrbLink = NULL;
   427	            }
   428	            else
   429	            {
   430	               ChatpadTrace(("Failed to get controls write URB memory buffer.\n"));
   431	               retStatus = STATUS_INTERNAL_ERROR;
   432	            }
   433	         }
   434	
   435	         for(int requestIndex = 0;
   436	             ((STATUS_SUCCESS == retStatus) &&
   437	              (requestIndex < CHATPAD_MAX_NUM_PENDING_USB_REQUESTS));
   438	             requestIndex++)
   439	         {
   440	            if(STATUS_SUCCESS == retStatus)
   441	            {
   442	               retStatus = WdfMemoryCreate(
   443	                 WDF_NO_OBJECT_ATTRIBUTES,
   444	                 NonPagedPool,
   445	                 0,
   446	                 CONTROLS_READ_BUFFER_LENGTH,
   447	                 &(devContext->controlsReadBufferMemory[requestIndex]),
   448	                 NULL);
   449	               if(STATUS_SUCCESS != retStatus)
   450	               {
   451	                  ChatpadTrace(("Allocation failed for controls read buffer memory.\n"));
   452	               }
   453	            }
   454	
   455	            if(STATUS_SUCCESS == retStatus)
   456	            {
   457	               retStatus = WdfMemoryCreate(
   458	                 WDF_NO_OBJECT_ATTRIBUTES,
   459	                 NonPagedPool,
   460	                 0,
   461	                 sizeof(struct _URB_BULK_OR_INTERRUPT_TRANSFER),
   462	                 &(devContext->controlsReadUrbMemory[requestIndex]),
   463	                 NULL);
   464	               if(STATUS_SUCCESS != retStatus)
   465	               {
   466	                  ChatpadTrace(("Allocation failed for controls read URB memory.\n"));
   467	               }
   468	            }
   469	
   470	            if(STATUS_SUCCESS == retStatus)
   471	            {
   472	               PURB controlsReadUrb =
   473	                 static_cast<PURB>(WdfMemoryGetBuffer(devContext->controlsReadUrbMemory[requestIndex], NULL));
   474	               if(NULL != controlsReadUrb)
   475	               {
   476	                  // TODO move this code somewhere common so it will not be redundant and prone to bugs if
   477	                  //   one side changes?
   478	                  controlsReadUrb->UrbHeader.Function = URB_FUNCTION_BULK_OR_INTERRUPT_TRANSFER;
   479	                  controlsReadUrb->UrbHeader.Length = sizeof(struct _URB_BULK_OR_INTERRUPT_TRANSFER);
   480	                  controlsReadUrb->UrbBulkOrInterruptTransfer.PipeHandle = NULL;
   481	                  // TODO could the short transfer flag ever cause problems on some controllers and/or hubs?
   482	                  controlsReadUrb->UrbBulkOrInterruptTransfer.TransferFlags =
   483	                    USBD_TRANSFER_DIRECTION_IN | USBD_SHORT_TRANSFER_OK;
   484	                  controlsReadUrb->UrbBulkOrInterruptTransfer.TransferBufferLength =
   485	                    CONTROLS_READ_BUFFER_LENGTH;
   486	                  controlsReadUrb->UrbBulkOrInterruptTransfer.TransferBufferMDL = NULL;
   487	                  controlsReadUrb->UrbBulkOrInterruptTransfer.TransferBuffer =
   488	                    WdfMemoryGetBuffer(devContext->controlsReadBufferMemory[requestIndex], NULL);
   489	                  controlsReadUrb->UrbBulkOrInterruptTransfer.UrbLink = NULL;
   490	               }
   491	               else
   492	               {
   493	                  ChatpadTrace(("Failed to get controls read URB memory buffer.\n"));
   494	                  retStatus = STATUS_INTERNAL_ERROR;
   495	               }
   496	            }
   497	
   498	            if(STATUS_SUCCESS == retStatus)
   499	            {
   500	               devContext->controlsReadRequestsInUse[requestIndex] = FALSE;
   501	
   502	               retStatus = WdfRequestCreate(
   503	                 &requestAttributes,
   504	                 devContext->targetForRequests,
   505	                 &(devContext->controlsReadRequests[requestIndex]));
   506	               if(STATUS_SUCCESS != retStatus)
   507	               {
   508	                  ChatpadTrace(("WdfRequestCreate failed:  0x%x\n", retStatus));
   509	               }
   510	            }
   511	
   512	            if(STATUS_SUCCESS == retStatus)
   513	            {
   514	               retStatus = WdfMemoryCreate(
   515	                 WDF_NO_OBJECT_ATTRIBUTES,
   516	                 NonPagedPool,
   517	                 0,
   518	                 CHATPAD_READ_BUFFER_LENGTH,
   519	                 &(devContext->chatpadReadBufferMemory[requestIndex]),
   520	                 NULL);
   521	               if(STATUS_SUCCESS != retStatus)
   522	               {
   523	                  ChatpadTrace(("Allocation failed for chatpad read buffer memory.\n"));
   524	               }
   525	            }
   526	
   527	            if(STATUS_SUCCESS == retStatus)
   528	            {
   529	               retStatus = WdfMemoryCreate(
   530	                 WDF_NO_OBJECT_ATTRIBUTES,
   531	                 NonPagedPool,
   532	                 0,
   533	                 sizeof(struct _URB_BULK_OR_INTERRUPT_TRANSFER),
   534	                 &(devContext->chatpadReadUrbMemory[requestIndex]),
   535	                 NULL);
   536	               if(STATUS_SUCCESS != retStatus)
   537	               {
   538	                  ChatpadTrace(("Allocation failed for chatpad read URB memory.\n"));
   539	               }
   540	            }
   541	
   542	            if(STATUS_SUCCESS == retStatus)
   543	            {
   544	               PURB chatpadReadUrb =
   545	                 static_cast<PURB>(WdfMemoryGetBuffer(devContext->chatpadReadUrbMemory[requestIndex], NULL));
   546	               if(NULL != chatpadReadUrb)
   547	               {
   548	                  // TODO move this code somewhere common so it will not be redundant and prone to bugs if
   549	                  //   one side changes?
   550	                  chatpadReadUrb->UrbHeader.Function = URB_FUNCTION_BULK_OR_INTERRUPT_TRANSFER;
   551	                  chatpadReadUrb->UrbHeader.Length = sizeof(struct _URB_BULK_OR_INTERRUPT_TRANSFER);
   552	                  chatpadReadUrb->UrbBulkOrInterruptTransfer.PipeHandle = NULL;
   553	                  // TODO could the short transfer flag ever cause problems on some controllers and/or hubs?
   554	                  chatpadReadUrb->UrbBulkOrInterruptTransfer.TransferFlags =
   555	                    USBD_TRANSFER_DIRECTION_IN | USBD_SHORT_TRANSFER_OK;
   556	                  chatpadReadUrb->UrbBulkOrInterruptTransfer.TransferBufferLength =
   557	                    CHATPAD_READ_BUFFER_LENGTH;
   558	                  chatpadReadUrb->UrbBulkOrInterruptTransfer.TransferBufferMDL = NULL;
   559	                  chatpadReadUrb->UrbBulkOrInterruptTransfer.TransferBuffer =
   560	                    WdfMemoryGetBuffer(devContext->chatpadReadBufferMemory[requestIndex], NULL);
   561	                  chatpadReadUrb->UrbBulkOrInterruptTransfer.UrbLink = NULL;
   562	               }
   563	               else
   564	               {
   565	                  ChatpadTrace(("Failed to get chatpad read URB memory buffer.\n"));
   566	                  retStatus = STATUS_INTERNAL_ERROR;
   567	               }
   568	            }
   569	
   570	            if(STATUS_SUCCESS == retStatus)
   571	            {
   572	               devContext->chatpadReadRequestsInUse[requestIndex] = FALSE;
   573	
   574	               retStatus = WdfRequestCreate(
   575	                 &requestAttributes,
   576	                 devContext->targetForRequests,

   724	// This method sends the controls data write request specified by the given
   725	// device context and request index.
   726	NTSTATUS SendControlsWriteRequest(
   727	  IN PDEVICE_CONTEXT devContext,
   728	  PUCHAR             inputBuffer,
   729	  ULONG              bytesToWrite,
   730	  PULONG             bytesTransferred)
   731	{
   732	   NTSTATUS                   retStatus         = STATUS_SUCCESS;
   733	   PURB                       controlsWriteUrb  = NULL;
   734	   WDF_REQUEST_SEND_OPTIONS   sendOptions;
   735	
   736	   if((NULL == devContext)          ||
   737	      (NULL == inputBuffer)         ||
   738	      (0 == bytesToWrite)           ||
   739	      (NULL == bytesTransferred))
   740	   {
   741	      ChatpadTrace(("Invalid parameters to SendControlsWriteRequest.\n"));
   742	      retStatus = STATUS_INTERNAL_ERROR;
   743	   }
   744	
   745	   if(STATUS_SUCCESS == retStatus)
   746	   {
   747	      RtlFillMemory(WdfMemoryGetBuffer(devContext->controlsWriteBufferMemory, NULL),
   748	                    0x00,
   749	                    CONTROLS_WRITE_BUFFER_LENGTH);
   750	
   751	      // Reinitialize the URB since its contents may have been adjusted during
   752	      // the most recent transfer.
   753	      controlsWriteUrb =
   754	        static_cast<PURB>(WdfMemoryGetBuffer(devContext->controlsWriteUrbMemory, NULL));
   755	      if(NULL != controlsWriteUrb)
   756	      {
   757	         // TODO move this code somewhere common so it will not be redundant and prone to bugs if
   758	         //   one side changes?
   759	         controlsWriteUrb->UrbHeader.Function = URB_FUNCTION_BULK_OR_INTERRUPT_TRANSFER;
   760	         controlsWriteUrb->UrbHeader.Length = sizeof(struct _URB_BULK_OR_INTERRUPT_TRANSFER);
   761	         controlsWriteUrb->UrbBulkOrInterruptTransfer.PipeHandle =
   762	           devContext->usbdInterfaceInformation[0]->Pipes[1].PipeHandle;
   763	         controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferFlags =
   764	           USBD_TRANSFER_DIRECTION_OUT;
   765	         controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferBufferLength =
   766	           CONTROLS_WRITE_BUFFER_LENGTH;
   767	         controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferBufferMDL = NULL;
   768	         controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferBuffer =
   769	           WdfMemoryGetBuffer(devContext->controlsWriteBufferMemory, NULL);
   770	         controlsWriteUrb->UrbBulkOrInterruptTransfer.UrbLink = NULL;
   771	
   772	         // Copy the data to send and the data length into the URB.
   773	         RtlCopyMemory(
   774	           controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferBuffer,
   775	           inputBuffer,
   776	           bytesToWrite);
   777	         controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferBufferLength = bytesToWrite;
   778	      }
   779	      else
   780	      {
   781	         ChatpadTrace(("Error, failed to get controls write URB memory buffer.\n"));
   782	         retStatus = STATUS_INTERNAL_ERROR;
   783	      }
   784	   } // end if(STATUS_SUCCESS == retStatus)
   785	
   786	   if(STATUS_SUCCESS == retStatus)
   787	   {
   788	      // TODO make enums or some other system for pipe indices?
   789	      // Format the request for sending an URB.
   790	      retStatus = WdfUsbTargetPipeFormatRequestForUrb(
   791	        devContext->wdfUsbPipe[0],
   792	        devContext->controlsWriteRequest,
   793	        devContext->controlsWriteUrbMemory,
   794	        NULL);
   795	   }
   796	
   797	   if(STATUS_SUCCESS == retStatus)
   798	   {
   799	      // Set up a timeout.
   800	      WDF_REQUEST_SEND_OPTIONS_INIT(
   801	        &sendOptions,
   802	        WDF_REQUEST_SEND_OPTION_SYNCHRONOUS | WDF_REQUEST_SEND_OPTION_TIMEOUT);
   803	      sendOptions.Timeout = WDF_REL_TIMEOUT_IN_SEC(CHATPAD_DRIVER_SYNCHRONOUS_REQUEST_TIMEOUT_LENGTH);
   804	      retStatus = WdfRequestAllocateTimer(devContext->controlsWriteRequest);
   805	      if(STATUS_SUCCESS != retStatus)
   806	      {
   807	         ChatpadTrace(("WdfRequestAllocateTimer failed:  0x%x\n", retStatus));
   808	      }
   809	   }
   810	
   811	   if(STATUS_SUCCESS == retStatus)
   812	   {
   813	      // Send the request synchronously.
   814	      if(WdfRequestSend(devContext->controlsWriteRequest,
   815	                        devContext->targetForRequests,
   816	                        &sendOptions))
   817	      {
   818	//         ChatpadTrace(("*** Sent controls write request successfully. ***\n"));
   819	         retStatus = WdfRequestGetStatus(devContext->controlsWriteRequest);
   820	         // I presumably should never have a value other than STATUS_SUCCESS since
   821	         // I am sending the request synchronously, so all values are treated as errors.
   822	         if(STATUS_SUCCESS != retStatus)
   823	         {
   824	            ChatpadTrace(("(WdfRequestGetStatus returns 0x%x)\n", retStatus));
   825	         }
   826	      }
   827	      else
   828	      {
   829	         ChatpadTrace(("*** Failed to send write request. ***\n"));
   830	         retStatus = STATUS_INTERNAL_ERROR;
   831	      }
   832	
   833	      // Make sure that the IOCTL sees the proper number of bytes transferred.
   834	      *bytesTransferred = controlsWriteUrb->UrbBulkOrInterruptTransfer.TransferBufferLength;
   835	
   836	      WDF_REQUEST_REUSE_PARAMS reuseParams;
   837	
   838	      WDF_REQUEST_REUSE_PARAMS_INIT(
   839	        &reuseParams,
   840	        WDF_REQUEST_REUSE_NO_FLAGS,
   841	        STATUS_SUCCESS);
   842	      retStatus = WdfRequestReuse(
   843	        devContext->controlsWriteRequest,
   844	        &reuseParams);
   845	      if(STATUS_SUCCESS != retStatus)
   846	      {
   847	         ChatpadTrace(("WdfRequestReuse failed:  0x%x\n", retStatus));
   848	      }
   849	   } // end if(STATUS_SUCCESS == retStatus)
   850	   else
   851	   {
   852	      ChatpadTrace(("WdfUsbTargetPipeFormatRequestForUrb failed:  0x%x\n", retStatus));
   853	   }
   854	
   855	   return retStatus;
   856	} // end SendControlsWriteRequest

  1113	  WDFFILEOBJECT   fileObject)
  1114	{
  1115	   NTSTATUS apiStatus = STATUS_SUCCESS;
  1116	
  1117	   // Decrement the open count.
  1118	   WdfWaitLockAcquire(chatpadDeviceCollectionLock, NULL);
  1119	
  1120	   // Since I have a wait lock, can I safely just modify it like this?
  1121	   if(numOpenHandles > 0)
  1122	   {
  1123	      numOpenHandles--;
  1124	   }
  1125	   else
  1126	   {
  1127	      ChatpadTrace(("Error, there are supposedly 0 open handles but the close handler was called...\n"));
  1128	   }
  1129	
  1130	   ChatpadTrace(("There are now %lu open handles to the chatpad filter device.\n", numOpenHandles));
  1131	
  1132	   if(0 == numOpenHandles)
  1133	   {
  1134	      // TODO there is probably a cleaner way to do this than having one extension
  1135	      //   have a pointer to the other one etc...maybe?  I want to make sure
  1136	      //   that filter mode gets disabled if the application disconnects,
  1137	      //   though.
  1138	      PCHATPAD_CONTROL_DEVICE_EXTENSION   controlExtension  = NULL;
  1139	      PDEVICE_CONTEXT                     devContext        = NULL;
  1140	
  1141	      if(NULL == chatpadControlDevice)
  1142	      {
  1143	         ChatpadTrace(("Error, control device handle is NULL.\n"));
  1144	         apiStatus = STATUS_INTERNAL_ERROR;
  1145	      }
  1146	
  1147	      if(STATUS_SUCCESS == apiStatus)
  1148	      {
  1149	         controlExtension = ChatpadControlGetData(chatpadControlDevice);
  1150	
  1151	         if(NULL == controlExtension)
  1152	         {
  1153	            ChatpadTrace(("Failed to get control device extension in chatpad filter close handler.\n"));
  1154	            apiStatus = STATUS_INTERNAL_ERROR;
  1155	         }
  1156	      }
  1157	
  1158	      if(STATUS_SUCCESS == apiStatus)
  1159	      {
  1160	         devContext = controlExtension->chatpadFilterDeviceContext;
  1161	         if(NULL == devContext)
  1162	         {
  1163	            ChatpadTrace(("Failed to get device context in close handler.\n"));
  1164	            apiStatus = STATUS_INTERNAL_ERROR;
  1165	         }
  1166	      }
  1167	
  1168	      if(STATUS_SUCCESS == apiStatus)
  1169	      {
  1170	         // Set the filter mode to unfiltered when the application disconnects.
  1171	         // TODO could messing with the device context here cause synchronization issues?
  1172	         devContext->filterMode = FILTER_MODE_UNFILTERED;
  1173	         ChatpadTrace(("Filter mode set to unfiltered on application disconnect.\n"));
  1174	
  1175	         // If any available controls read requests are not already pending,
  1176	         // then send them.  This is done because the top-level driver
  1177	         // should always have read requests pending, so we can just
  1178	         // start completing them as soon as we get data.
  1179	         for(int requestIndex = 0;
  1180	             ((STATUS_SUCCESS == apiStatus) &&
  1181	              (requestIndex < CHATPAD_MAX_NUM_PENDING_USB_REQUESTS));
  1182	             requestIndex++)
  1183	         {
  1184	            // TODO do I need synchronization here?
  1185	            if(FALSE == devContext->controlsReadRequestsInUse[requestIndex])
  1186	            {
  1187	               devContext->controlsReadRequestsInUse[requestIndex] = TRUE;
  1188	               apiStatus = SendControlsReadRequest(devContext, requestIndex);
  1189	            } // if(FALSE == devContext->controlsReadRequestsInUse[requestIndex])
  1190	         } // end looping through chatpad filter driver request indices
  1191	      } // end if(STATUS_SUCCESS == apiStatus)
  1192	   } // end if(0 == numOpenHandles)
  1193	
  1194	   WdfWaitLockRelease(chatpadDeviceCollectionLock);
  1195	} // end ChatpadControlDeviceCloseHandler

  1197	VOID
  1198	ChatpadFilterEvtDeviceContextCleanup(
  1199	  IN WDFOBJECT       object)
  1200	{
  1201	   NTSTATUS apiStatus = STATUS_SUCCESS;
  1202	   // TODO is static_cast or reinterpret_cast preferred here?  I'm doing an
  1203	   //      oldstyle cast because that's what the suggestion on osronline
  1204	   //      said to do.  :)
  1205	   //      Normally, for void* to something* I'd do static_cast, and from
  1206	   //      thingA* to thingB* I'd do reinterpret_cast.
  1207	   WDFDEVICE device = (WDFDEVICE) object;
  1208	   PDEVICE_CONTEXT devContext = NULL;
  1209	   ULONG deviceCount;
  1210	
  1211	   ChatpadTrace(("Cleaning up device context.\n"));
  1212	
  1213	   devContext = ChatpadFilterGetDeviceContext(device);
  1214	   if((NULL != device) && (NULL != devContext))
  1215	   {
  1216	      WdfWaitLockAcquire(chatpadDeviceCollectionLock, NULL);
  1217	
  1218	      // Clean up device context resources.
  1219	      // TODO is this the appropriate place to do the cleanup, inside the lock
  1220	      //   and before the device gets deleted?
  1221	      ReleaseCachedInterfaceInformation(devContext);
  1222	      for(int requestIndex = 0;
  1223	          ((STATUS_SUCCESS == apiStatus) &&
  1224	           (requestIndex < CHATPAD_MAX_NUM_PENDING_USB_REQUESTS));
  1225	          requestIndex++)
  1226	      {
  1227	         // TODO is it safe to release request memory here, or is there any chance
  1228	         //   that it could still be in use by the system/requests?
  1229	
  1230	         if(NULL != devContext->controlsWriteBufferMemory)
  1231	         {
  1232	            WdfObjectDelete(devContext->controlsWriteBufferMemory);
  1233	            devContext->controlsWriteBufferMemory = NULL;
  1234	         }
  1235	
  1236	         if(NULL != devContext->controlsWriteUrbMemory)
  1237	         {
  1238	            WdfObjectDelete(devContext->controlsWriteUrbMemory);
  1239	            devContext->controlsWriteUrbMemory = NULL;
  1240	         }
  1241	
  1242	         if(NULL != devContext->controlsReadBufferMemory[requestIndex])
  1243	         {
  1244	            WdfObjectDelete(devContext->controlsReadBufferMemory[requestIndex]);
  1245	            devContext->controlsReadBufferMemory[requestIndex] = NULL;
  1246	         }
  1247	
  1248	         if(NULL != devContext->controlsReadUrbMemory[requestIndex])
  1249	         {
  1250	            WdfObjectDelete(devContext->controlsReadUrbMemory[requestIndex]);
  1251	            devContext->controlsReadUrbMemory[requestIndex] = NULL;
  1252	         }
  1253	
  1254	         if(NULL != devContext->chatpadReadBufferMemory[requestIndex])
  1255	         {
  1256	            WdfObjectDelete(devContext->chatpadReadBufferMemory[requestIndex]);
  1257	            devContext->chatpadReadBufferMemory[requestIndex] = NULL;
  1258	         }
  1259	
  1260	         if(NULL != devContext->chatpadReadUrbMemory[requestIndex])
  1261	         {
  1262	            WdfObjectDelete(devContext->chatpadReadUrbMemory[requestIndex]);
  1263	            devContext->chatpadReadUrbMemory[requestIndex] = NULL;
  1264	         }
  1265	      } // end looping through chatpad filter driver request indices
  1266	
  1267	      deviceCount = WdfCollectionGetCount(chatpadDeviceCollection);
  1268	      if(deviceCount == 1)
  1269	      {
  1270	         // If we are the last instance, then delete the control device so that
  1271	         // the driver can unload.  Apparently there was a bug in WDF 1.0, but 
  1272	         // apparently this sort of thing is fine in WDF 1.9 (and even earlier?).
  1273	         //
  1274	         // According to the toaster KMDF sideband filter example, the deletion
  1275	         // below has to be done while the lock on the collection is acquired
  1276	         // since it protects the global device variable, and another thread
  1277	         // cannot try to create while we are deleting the device.
  1278	
  1279	         // I added an open count, protected by chatpadDeviceCollectionLock,
  1280	         // which means that if the device is unplugged while the user still
  1281	         // has a handle open, the device should remain open until it is
  1282	         // replugged and unplugged after that point.
  1283	         // TODO make the above behavior nicer if possible?  Is it possible
  1284	         //   to delete the control device from the close handler?
  1285	         if(0 == numOpenHandles)
  1286	         {
  1287	            ChatpadDeleteControlDevice();
  1288	         }
  1289	         else
  1290	         {
  1291	            ChatpadTrace(("The device was not deleted since there are still %lu open handles to it.\n",
  1292	                          numOpenHandles));
  1293	         }
  1294	      }
  1295	
  1296	      // Even if the control device was not deleted, however, we mark it as unavailable since the cable
  1297	      // was pulled or whatnot.
  1298	      chatpadControlDeviceAvailable = FALSE;
  1299	
  1300	      WdfCollectionRemove(chatpadDeviceCollection, device);
  1301	
  1302	      WdfWaitLockRelease(chatpadDeviceCollectionLock);
  1303	   } // end if((NULL != device) && (NULL != devContext))
  1304	   else
  1305	   {
  1306	      // This should never happen.
  1307	      ChatpadTrace(("Error, device and/or devContext are NULL while cleaning up.\n"));
  1308	   }
  1309	} // end ChatpadFilterEvtDeviceContextCleanup

  1311	// Creates a control device object so that an application can talk to the filter driver.
  1312	// TODO how does this thing link to the regular filter device and its queues?
  1313	NTSTATUS
  1314	ChatpadCreateControlDevice(
  1315	  WDFDEVICE device)
  1316	{
  1317	   NTSTATUS                retStatus = STATUS_SUCCESS;
  1318	   BOOLEAN                 createDevice = FALSE;
  1319	   PWDFDEVICE_INIT         deviceInit = NULL;
  1320	   WDF_FILEOBJECT_CONFIG   fileObjectConfig;
  1321	   WDFDEVICE               newChatpadControlDevice = NULL;
  1322	   WDF_OBJECT_ATTRIBUTES   controlAttributes;
  1323	   DECLARE_CONST_UNICODE_STRING(ntDeviceName, CHATPAD_FILTER_CONTROL_NAME);
  1324	   DECLARE_CONST_UNICODE_STRING(symbolicLinkName, CHATPAD_FILTER_CONTROL_LINK);
  1325	   WDFQUEUE                newQueue;
  1326	   WDF_IO_QUEUE_CONFIG     ioQueueConfig;
  1327	
  1328	   // Determine whether we should create the device, or if it already exists.
  1329	   WdfWaitLockAcquire(chatpadDeviceCollectionLock, NULL);
  1330	
  1331	// TODO NEXT REMOVE after testing
  1332	ChatpadTrace(("Collection device count is %d.\n", WdfCollectionGetCount(chatpadDeviceCollection)));
  1333	   // TODO is this a safe way to synchronize things?
  1334	   if(NULL == chatpadControlDevice)
  1335	   {
  1336	      createDevice = TRUE;
  1337	   }
  1338	
  1339	   WdfWaitLockRelease(chatpadDeviceCollectionLock);
  1340	
  1341	   if(!createDevice)
  1342	   {
  1343	      ChatpadTrace(("Chatpad control device was already created.\n"));
  1344	   }
  1345	   else
  1346	   {
  1347	      ChatpadTrace(("Creating chatpad control device.\n"));
  1348	
  1349	      // Allocate and initialize a WDFDEVICE_INIT structure for the control device.
  1350	      deviceInit = WdfControlDeviceInitAllocate(
  1351	        WdfDeviceGetDriver(device),
  1352	        // TODO what does this ugly thing govern -- permissions?  Based off of toaster sideband filter example.
  1353	        &SDDL_DEVOBJ_SYS_ALL_ADM_RWX_WORLD_RW_RES_R);
  1354	      if(NULL == deviceInit)
  1355	      {
  1356	         retStatus = STATUS_INSUFFICIENT_RESOURCES;
  1357	      }
  1358	
  1359	      if(STATUS_SUCCESS == retStatus)
  1360	      {
  1361	         // Make the control device exclusive.  I only want one application talking to it at a time.
  1362	         WdfDeviceInitSetExclusive(deviceInit, TRUE);
  1363	         // Get device name.
  1364	         retStatus = WdfDeviceInitAssignName(deviceInit, &ntDeviceName);
  1365	      }
  1366	
  1367	      // Set up callbacks so we can keep track of whether user mode has the
  1368	      // control device open.
  1369	      WDF_FILEOBJECT_CONFIG_INIT(
  1370	        &fileObjectConfig,
  1371	        ChatpadControlDeviceCreateHandler,
  1372	        ChatpadControlDeviceCloseHandler,
  1373	        WDF_NO_EVENT_CALLBACK);
  1374	
  1375	      WdfDeviceInitSetFileObjectConfig(
  1376	        deviceInit,
  1377	        &fileObjectConfig,
  1378	        WDF_NO_OBJECT_ATTRIBUTES);
  1379	
  1380	      if(STATUS_SUCCESS == retStatus)
  1381	      {
  1382	         WDF_OBJECT_ATTRIBUTES_INIT_CONTEXT_TYPE(
  1383	           &controlAttributes,
  1384	           CHATPAD_CONTROL_DEVICE_EXTENSION);
  1385	         // Create the control device.
  1386	         retStatus = WdfDeviceCreate(
  1387	           &deviceInit,
  1388	           &controlAttributes,
  1389	           &newChatpadControlDevice);
  1390	      }
  1391	
  1392	      // Initialize control extension.
  1393	      PDEVICE_CONTEXT devContext = ChatpadFilterGetDeviceContext(device);
  1394	      if(NULL != devContext)
  1395	      {
  1396	         PCHATPAD_CONTROL_DEVICE_EXTENSION controlExtension =
  1397	           ChatpadControlGetData(newChatpadControlDevice);
  1398	
  1399	         if(NULL != controlExtension)
  1400	         {
  1401	            ChatpadTrace(("Initializing chatpad filter control device extension.\n"));
  1402	            controlExtension->chatpadFilterDeviceContext = devContext;
  1403	         }
  1404	         else
  1405	         {
  1406	            ChatpadTrace(("Error, could not initialize control device extension.\n"));
  1407	            retStatus = STATUS_INTERNAL_ERROR;
  1408	         }
  1409	      }
  1410	      else
  1411	      {
  1412	         retStatus = STATUS_INTERNAL_ERROR;
  1413	      }
  1414	
  1415	      if(STATUS_SUCCESS == retStatus)
  1416	      {
  1417	         // Create symbolic link so user mode can open the device.
  1418	         retStatus = WdfDeviceCreateSymbolicLink(
  1419	           newChatpadControlDevice,
  1420	           &symbolicLinkName);
  1421	      }
  1422	
  1423	      if(STATUS_SUCCESS == retStatus)
  1424	      {
  1425	         // Configure default queue so that requests to the control device will be parallel.
  1426	         // TODO is this okay?  The idea here is that I want to be able to do read and
  1427	         //   write requests at the same time depending on how blocking works.
  1428	         WDF_IO_QUEUE_CONFIG_INIT_DEFAULT_QUEUE(
  1429	           &ioQueueConfig,
  1430	           //WdfIoQueueDispatchSequential);
  1431	           WdfIoQueueDispatchParallel);
  1432	
  1433	         // Select what we want from the queue.
  1434	
  1435	         // This should catch application-command IOCTLs.
  1436	         ioQueueConfig.EvtIoDeviceControl =
  1437	           ChatpadFilterIOCTLHandler;
  1438	
  1439	         // Create the queue.
  1440	         retStatus = WdfIoQueueCreate(
  1441	           newChatpadControlDevice,
  1442	           &ioQueueConfig,
  1443	           WDF_NO_OBJECT_ATTRIBUTES, // TODO what does this parameter do?
  1444	           &newQueue);
  1445	      } // end if(STATUS_SUCCESS == retStatus)
  1446	
  1447	      if(STATUS_SUCCESS == retStatus)
  1448	      {
  1449	         WdfControlFinishInitializing(newChatpadControlDevice);
  1450	         WdfWaitLockAcquire(chatpadDeviceCollectionLock, NULL);
  1451	         chatpadControlDevice = newChatpadControlDevice;
  1452	         WdfWaitLockRelease(chatpadDeviceCollectionLock);
  1453	
  1454	         ChatpadTrace(("Chatpad control device created successfully.\n"));
  1455	      }
  1456	      else
  1457	      {
  1458	         ChatpadTrace(("WdfIoQueueCreate failed:  0x%x\n", retStatus));
  1459	      }
  1460	   } // end if creating the control device
  1461	
  1462	   // Handle error cases.
  1463	   if(STATUS_SUCCESS != retStatus)
  1464	   {
  1465	      ChatpadTrace(("Failed to create chatpad control device:  0x%x\n", retStatus));
  1466	
  1467	      if(NULL != deviceInit)
  1468	      {
  1469	         WdfDeviceInitFree(deviceInit);
  1470	         deviceInit = NULL;
  1471	      }
  1472	
  1473	      if(NULL != newChatpadControlDevice)
  1474	      {
  1475	         // Release the newly created, but uninitialized, object.
  1476	         WdfObjectDelete(newChatpadControlDevice);
  1477	         newChatpadControlDevice = NULL;
  1478	      }
  1479	   }
  1480	
  1481	   return retStatus;
  1482	} // end ChatpadCreateControlDevice

  1500	// This function normally runs at DISPATCH IRQL.  It might run at PASSIVE if
  1501	// something gets messed up with requests.
  1502	VOID
  1503	ControlsReadCompletion(
  1504	  IN WDFREQUEST                     request,
  1505	  IN WDFIOTARGET                    target,
  1506	  IN PWDF_REQUEST_COMPLETION_PARAMS params,
  1507	  IN WDFCONTEXT                     context)
  1508	{
  1509	   PDEVICE_CONTEXT   devContext           = reinterpret_cast<PDEVICE_CONTEXT>(context);
  1510	   NTSTATUS          apiStatus            = STATUS_SUCCESS;
  1511	   WDFMEMORY         urbMemory            = NULL;
  1512	   PUCHAR            transferBuffer       = NULL;
  1513	   ULONG             bytesTransferred     = 0;
  1514	   BOOLEAN           wasRequestCancelled  = FALSE;
  1515	
  1516	   // These variables are used for completing controls data read requests.
  1517	   WDFREQUEST        driverRequestFromAbove;
  1518	   WDFREQUEST        userRequestFromAbove;
  1519	   NTSTATUS          driverCompletionStatusForAboveCode  = STATUS_SUCCESS;
  1520	   NTSTATUS          userCompletionStatusForAboveCode    = STATUS_SUCCESS;
  1521	   BOOLEAN           driverCompleteRequestToAboveCode    = FALSE;
  1522	   BOOLEAN           userCompleteRequestToAboveCode      = FALSE;
  1523	
  1524	   ASSERT(devContext);
  1525	   ASSERT(params);
  1526	
  1527	//   ChatpadTrace(("Controls read completion callback called with IRQL %u\n", KeGetCurrentIrql()));
  1528	//   ChatpadTrace(("I/O status:  0x%x\n", params->IoStatus.Status));
  1529	
  1530	   // Get the status.  If the request was cancelled, then we want to be careful
  1531	   // not to complete requests from above.  Otherwise, there was apparently
  1532	   // some sort of infinite loop due to the top-level requests.
  1533	   // TODO why was this the case?
  1534	   if(STATUS_SUCCESS == apiStatus)
  1535	   {
  1536	      apiStatus = WdfRequestGetStatus(request);
  1537	//      ChatpadTrace(("WdfRequestGetStatus returns 0x%x\n", apiStatus));
  1538	
  1539	      if(STATUS_CANCELLED == apiStatus)
  1540	      {
  1541	         ChatpadTrace(("Controls read request was cancelled.\n"));
  1542	         wasRequestCancelled = TRUE;
  1543	      }
  1544	   }
  1545	
  1546	   // Only do this processing if the request was not cancelled.
  1547	   if(FALSE == wasRequestCancelled)
  1548	   {
  1549	//      ChatpadTrace(("USBD status 0x%x, request type %d\n",
  1550	//                    params->Parameters.Usb.Completion->UsbdStatus,
  1551	//                    params->Parameters.Usb.Completion->Type));
  1552	      PWDF_USB_REQUEST_COMPLETION_PARAMS completionParams =
  1553	        params->Parameters.Usb.Completion;
  1554	      if(NULL != completionParams)
  1555	      {
  1556	         if(WdfUsbRequestTypePipeUrb == completionParams->Type)
  1557	         {
  1558	            urbMemory = completionParams->Parameters.PipeUrb.Buffer;
  1559	//            ChatpadTrace(("WDFMEMORY handle is 0x%x\n", urbMemory));
  1560	            if(NULL != urbMemory)
  1561	            {
  1562	               PURB controlsReadUrb =
  1563	                 static_cast<PURB>(WdfMemoryGetBuffer(urbMemory, NULL));
  1564	               if(NULL != controlsReadUrb)
  1565	               {
  1566	                  transferBuffer =
  1567	                    static_cast<PUCHAR>(controlsReadUrb->UrbBulkOrInterruptTransfer.TransferBuffer);
  1568	                  bytesTransferred = controlsReadUrb->UrbBulkOrInterruptTransfer.TransferBufferLength;
  1569	//                  ChatpadTrace(("Buffer address is 0x%x with data length %lu\n", transferBuffer, bytesTransferred));
  1570	                  if(NULL != transferBuffer)
  1571	                  {
  1572	/*
  1573	                     ChatpadTrace(("### Buffer data (length %lu):  %02x %02x %02x %02x %02x %02x %02x %02x ... ###\n",
  1574	                                   bytesTransferred,
  1575	                                   transferBuffer[0],
  1576	                                   transferBuffer[1],
  1577	                                   transferBuffer[2],
  1578	                                   transferBuffer[3],
  1579	                                   transferBuffer[4],
  1580	                                   transferBuffer[5],
  1581	                                   transferBuffer[6],
  1582	                                   transferBuffer[7]));
  1583	*/
  1584	
  1585	                     // Only forward data to the top-level driver if we are not intercepting the data.
  1586	                     if(FILTER_MODE_INTERCEPTED != devContext->filterMode)
  1587	                     {
  1588	                        apiStatus = WdfIoQueueRetrieveNextRequest(
  1589	                          devContext->controlsParkedDriverReadRequestQueue,
  1590	                          &driverRequestFromAbove);
  1591	
  1592	                        if(STATUS_SUCCESS == apiStatus)
  1593	                        {
  1594	                           PURB topLevelUrb =
  1595	                             static_cast<PURB>(
  1596	                               URB_FROM_IRP(
  1597	                                 WdfRequestWdmGetIrp(
  1598	                                   driverRequestFromAbove)));
  1599	
  1600	                           if(NULL != topLevelUrb)
  1601	                           {
  1602	                              RtlCopyMemory(static_cast<unsigned char*>(
  1603	                                              topLevelUrb->UrbBulkOrInterruptTransfer.TransferBuffer),
  1604	                                            transferBuffer,
  1605	                                            bytesTransferred);
  1606	                              topLevelUrb->UrbBulkOrInterruptTransfer.TransferBufferLength = bytesTransferred;
  1607	
  1608	                              driverCompleteRequestToAboveCode       = TRUE;
  1609	                              driverCompletionStatusForAboveCode     = apiStatus;
  1610	                           }
  1611	                           else
  1612	                           {
  1613	                              ChatpadTrace(("Unable to get pointer to URB from top-level driver request.\n"));
  1614	                              apiStatus = STATUS_INTERNAL_ERROR;
  1615	                           }
  1616	                        } // end if(STATUS_SUCCESS == apiStatus)
  1617	                        else
  1618	                        {
  1619	                           ChatpadTrace(("No top-level controls read request available.\n"));
  1620	                           apiStatus = STATUS_INTERNAL_ERROR;
  1621	                        }
  1622	                     } // end if(FILTER_MODE_INTERCEPTED != devContext->filterMode)
  1623	
  1624	                     // Only send data to the user if we are filtering or intercepting the data.
  1625	                     if((FILTER_MODE_FILTERED      == devContext->filterMode)    ||
  1626	                        (FILTER_MODE_INTERCEPTED   == devContext->filterMode))
  1627	                     {
  1628	                        PVOID bufferToUserAsVoid   = NULL;
  1629	                        PUCHAR bufferToUser        = NULL;
  1630	
  1631	                        apiStatus = WdfIoQueueRetrieveNextRequest(
  1632	                          devContext->controlsParkedUserReadRequestQueue,
  1633	                          &userRequestFromAbove);
  1634	                        if(STATUS_SUCCESS != apiStatus)
  1635	                        {
  1636	                           ChatpadTrace(("No user-provided controls read request available.\n"));
  1637	                        }
  1638	
  1639	                        if(STATUS_SUCCESS == apiStatus)
  1640	                        {
  1641	//                           ChatpadTrace(("Got a user-provided read request from the queue.\n"));
  1642	
  1643	                           apiStatus = WdfRequestRetrieveOutputBuffer(
  1644	                             userRequestFromAbove,
  1645	                             bytesTransferred,
  1646	                             &bufferToUserAsVoid,
  1647	                             NULL);
  1648	                           // This ugly cast is probably avoidable, but it works for now.
  1649	                           bufferToUser = static_cast<PUCHAR>(bufferToUserAsVoid);
  1650	                           if(STATUS_SUCCESS != apiStatus)
  1651	                           {
  1652	                              ChatpadTrace(("WdfRequestRetrieveOutputBuffer failed:  0x%x\n", apiStatus));
  1653	                           }
  1654	
  1655	                           // Just in case, check the pointer.
  1656	                           if(NULL == bufferToUser)
