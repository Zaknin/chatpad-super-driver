    81	// Updates/installs the chatpad filter driver.
    82	int UpdateChatpadFilterDriver(void)
    83	{
    84	   int      nRetValue            = SUCCESS;
    85	
    86	   HMODULE  newDevModule         = NULL;
    87	   DWORD    requiredPathLength   = 0;
    88	   TCHAR    infPath[MAX_PATH];
    89	   UpdateDriverForPlugAndPlayDevicesProto UpdateFunction;
    90	
    91	   // Get a full INF path name.
    92	   requiredPathLength = GetFullPathName(CHATPAD_FILTER_DRIVER_INF, MAX_PATH, infPath, NULL);
    93	   if((requiredPathLength >= MAX_PATH) ||
    94	      (0 == requiredPathLength))
    95	   {
    96	      ShowMessage("Error finding chatpad filter INF file.\n");
    97	      nRetValue = FAILURE;
    98	   }
    99	
   100	   if(SUCCESS == nRetValue)
   101	   {
   102	      if(INVALID_FILE_ATTRIBUTES == GetFileAttributes(infPath))
   103	      {
   104	         nRetValue = FAILURE;
   105	         char errorString[80];
   106	         sprintf_s(errorString, 80, "Error getting chatpad filter INF file attributes:  0x%08x\n", GetLastError());
   107	         ShowMessage(errorString);
   108	
   109	         nRetValue = FAILURE;
   110	      }
   111	   }
   112	
   113	   if(SUCCESS == nRetValue)
   114	   {
   115	      // TODO why does devcon do things this way, i.e. loading the DLL instead
   116	      //   of linking to a library?  Does it reduce the number of different
   117	      //   executables required or something?
   118	      // Load newdev.dll so that UpdateDriverForPlugAndPlayDevices can be used.
   119	      newDevModule = LoadLibrary(TEXT("newdev.dll"));
   120	      if(NULL == newDevModule)
   121	      {
   122	         ShowMessage("Error loading newdev.dll.\n");
   123	         nRetValue = FAILURE;
   124	      }
   125	   }
   126	
   127	   if(SUCCESS == nRetValue)
   128	   {
   129	      // TODO use an explicit C++ cast instead?
   130	      UpdateFunction =
   131	        (UpdateDriverForPlugAndPlayDevicesProto)
   132	          GetProcAddress(newDevModule, UPDATEDRIVERFORPLUGANDPLAYDEVICES);
   133	      if(NULL == UpdateFunction)
   134	      {
   135	         ShowMessage("GetProcAddress returned NULL.\n");
   136	         nRetValue = FAILURE;
   137	      }
   138	   }
   139	
   140	   if(SUCCESS == nRetValue)
   141	   {
   142	      BOOL rebootRequired = FALSE;
   143	
   144	      if(FALSE == UpdateFunction(NULL, WIRED_CONTROLLER_HW_ID, infPath, INSTALLFLAG_FORCE, &rebootRequired))
   145	      {
   146	         char errorString[80];
   147	         sprintf_s(errorString, 80, "Failed to update chatpad filter device driver:  0x%08x\n", GetLastError());
   148	         ShowMessage(errorString);
   149	
   150	         nRetValue = FAILURE;
   151	      }
   152	
   153	      if(TRUE == rebootRequired)
   154	      {
   155	         ShowMessage(
   156	           "WARNING:  Windows thinks a reboot is required.\n"
   157	           "Reboot if you want, once the installer has finished running.\n");
   158	      }
   159	   }
   160	
   161	   if(NULL != newDevModule)
   162	   {
   163	      FreeLibrary(newDevModule);
   164	      newDevModule = NULL;
   165	   }
   166	
   167	   return nRetValue;
   168	} // end UpdateChatpadFilterDriver

   170	// Installs/updates the virtual keyboard driver.
   171	int UpdateVirtualKeyboardDriver(void)
   172	{
   173	   int      nRetValue            = SUCCESS;
   174	
   175	   HDEVINFO deviceInfoSet        = INVALID_HANDLE_VALUE;
   176	   SP_DEVINFO_DATA deviceInfoData;
   177	   GUID     classGUID;
   178	   TCHAR    className[MAX_CLASS_NAME_LEN];
   179	   // TODO what does LINE_LEN represent?  This is what devcon uses.  And why +4?  For two wide 0's?
   180	   TCHAR    hardwareIDList[LINE_LEN+4];
   181	
   182	   HMODULE  newDevModule         = NULL;
   183	   DWORD    requiredPathLength   = 0;
   184	   TCHAR    infPath[MAX_PATH];
   185	   UpdateDriverForPlugAndPlayDevicesProto UpdateFunction;
   186	
   187	   // Get a full INF path name.
   188	   requiredPathLength = GetFullPathName(CHATPAD_VIRTUAL_KEYBOARD_DRIVER_INF, MAX_PATH, infPath, NULL);
   189	   if((requiredPathLength >= MAX_PATH) ||
   190	      (0 == requiredPathLength))
   191	   {
   192	      ShowMessage("Error finding virtual keyboard INF file.\n");
   193	      nRetValue = FAILURE;
   194	   }
   195	
   196	   if(SUCCESS == nRetValue)
   197	   {
   198	      if(INVALID_FILE_ATTRIBUTES == GetFileAttributes(infPath))
   199	      {
   200	         ShowMessage("Error getting virtual keyboard INF file attributes.\n");
   201	         nRetValue = FAILURE;
   202	      }
   203	   }
   204	
   205	   //***
   206	   // First, create the device node.
   207	   //***
   208	
   209	   if(SUCCESS == nRetValue)
   210	   {
   211	      // List of hardware IDs must be double zero-terminated, according to devcon source.
   212	      ZeroMemory(hardwareIDList, sizeof(hardwareIDList));
   213	      if(S_OK != StringCchCopy(hardwareIDList, LINE_LEN, VIRTUAL_KEYBOARD_HW_ID))
   214	      {
   215	         ShowMessage("Error copying virtual keyboard driver hardware ID.\n");
   216	         nRetValue = FAILURE;
   217	      }
   218	   }
   219	
   220	   if(SUCCESS == nRetValue)
   221	   {
   222	      // Extract the class GUID.
   223	      if(FALSE == SetupDiGetINFClass(infPath, &classGUID, className, MAX_CLASS_NAME_LEN, NULL))
   224	      {
   225	         ShowMessage("Error extracting virtual keyboard class GUID.\n");
   226	         nRetValue = FAILURE;
   227	      }
   228	   }
   229	
   230	   if(SUCCESS == nRetValue)
   231	   {
   232	      // Create a container for a device information element. 
   233	      // TODO read up on how this all works...for now, I am just following the devcon example flow
   234	      deviceInfoSet = SetupDiCreateDeviceInfoList(&classGUID, NULL);
   235	      if(INVALID_HANDLE_VALUE == deviceInfoSet)
   236	      {
   237	         ShowMessage("Error creating device info list for virtual keyboard device.\n");
   238	         nRetValue = FAILURE;
   239	      }
   240	   }
   241	
   242	   if(SUCCESS == nRetValue)
   243	   {
   244	      // Create the device information element.
   245	      deviceInfoData.cbSize = sizeof(SP_DEVINFO_DATA);
   246	      if(FALSE == SetupDiCreateDeviceInfo(
   247	        deviceInfoSet,
   248	        className,
   249	        &classGUID,
   250	        _T("Chatpad Virtual Keyboard Device"),
   251	        NULL,
   252	        DICD_GENERATE_ID,
   253	        &deviceInfoData))
   254	      {
   255	         char errorString[80];
   256	         sprintf_s(errorString, 80, "Failed to create virtual keyboard device:  0x%08x\n", GetLastError());
   257	         ShowMessage(errorString);
   258	
   259	         nRetValue = FAILURE;
   260	      }
   261	   } // end if(SUCCESS == nRetValue)
   262	
   263	   if(SUCCESS == nRetValue)
   264	   {
   265	      // Add the hardware ID to the device's HardwareID property.
   266	      if(FALSE == SetupDiSetDeviceRegistryProperty(
   267	        deviceInfoSet,
   268	        &deviceInfoData,
   269	        SPDRP_HARDWAREID,
   270	        reinterpret_cast<BYTE*>(hardwareIDList),
   271	        (lstrlen(hardwareIDList) + 1 + 1) * sizeof(TCHAR)))
   272	      {
   273	         ShowMessage("Error setting HardwareID property for virtual keyboard device.\n");
   274	         nRetValue = FAILURE;
   275	      }
   276	   } // end if(SUCCESS == nRetValue)
   277	
   278	   if(SUCCESS == nRetValue)
   279	   {
   280	      // Transform the registry element into a device node in the plug-and-play hardware tree.
   281	      if(FALSE == SetupDiCallClassInstaller(
   282	        DIF_REGISTERDEVICE,
   283	        deviceInfoSet,
   284	        &deviceInfoData))
   285	      {
   286	         char errorString[80];
   287	         sprintf_s(errorString, 80, "Failed to create virtual keyboard device node:  0x%08x\n", GetLastError());
   288	         ShowMessage(errorString);
   289	
   290	         nRetValue = FAILURE;
   291	      }
   292	   } // end if(SUCCESS == nRetValue)
   293	
   294	   //***
   295	   // Now, "update" the driver.
   296	   //***
   297	
   298	   if(SUCCESS == nRetValue)
   299	   {
   300	      // Load newdev.dll so that UpdateDriverForPlugAndPlayDevices can be used.
   301	      newDevModule = LoadLibrary(TEXT("newdev.dll"));
   302	      if(NULL == newDevModule)
   303	      {
   304	         ShowMessage("Error loading newdev.dll.\n");
   305	         nRetValue = FAILURE;
   306	      }
   307	   }
   308	
   309	   if(SUCCESS == nRetValue)
   310	   {
   311	      // TODO use an explicit C++ cast instead?
   312	      UpdateFunction =
   313	        (UpdateDriverForPlugAndPlayDevicesProto)
   314	          GetProcAddress(newDevModule, UPDATEDRIVERFORPLUGANDPLAYDEVICES);
   315	      if(NULL == UpdateFunction)
   316	      {
   317	         ShowMessage("GetProcAddress returned NULL.\n");
   318	         nRetValue = FAILURE;
   319	      }
   320	   }
   321	
   322	   if(SUCCESS == nRetValue)
   323	   {
   324	      BOOL rebootRequired = FALSE;
   325	
   326	      if(FALSE == UpdateFunction(NULL, VIRTUAL_KEYBOARD_HW_ID, infPath, INSTALLFLAG_FORCE, &rebootRequired))
   327	      {
   328	         char errorString[80];
   329	         sprintf_s(errorString, 80, "Failed to update virtual keyboard device driver:  0x%08x\n", GetLastError());
   330	         ShowMessage(errorString);
   331	
   332	         nRetValue = FAILURE;
   333	      }
   334	
   335	      if(TRUE == rebootRequired)
   336	      {
   337	         ShowMessage(
   338	           "WARNING:  Windows thinks a reboot is required.\n"
   339	           "Reboot if you want, once the installer has finished running.\n");
   340	      }
   341	   }
   342	
   343	   if(NULL != newDevModule)
   344	   {
   345	      FreeLibrary(newDevModule);
   346	      newDevModule = NULL;
   347	   }
   348	
   349	   if(INVALID_HANDLE_VALUE != deviceInfoSet)
   350	   {
   351	      SetupDiDestroyDeviceInfoList(deviceInfoSet);
   352	      deviceInfoSet = INVALID_HANDLE_VALUE;
   353	   }
   354	
   355	   return nRetValue;
   356	} // end UpdateVirtualKeyboardDriver

   546	// Uninstalls the chatpad filter driver.
   547	// TODO is there eventually a better way to uninstall, i.e. by deleting the driver store entry or
   548	//   removing the lower filters etc.?  This way requires the user to go into device manager and
   549	//   manually update to get the old driver back.
   550	int UninstallChatpadFilterDriver(void)
   551	{
   552	   int      nRetValue               = SUCCESS;
   553	   UINT     requiredPathLength      = 0;
   554	   // TODO:  Depending on the build system, these might be wide characters, and it might
   555	   //   break the functionality of some code below.  I need to go through the code
   556	   //   eventually and make sure I'm handling strings properly.
   557	   TCHAR    windowsPath[MAX_PATH];
   558	   TCHAR    pathToDelete[MAX_PATH];
   559	
   560	   // Get the Windows directory path.  Make sure there is enough space to append the driver path.
   561	   requiredPathLength = GetSystemWindowsDirectory(windowsPath, MAX_PATH);
   562	   // TODO eventually don't use an arbitrary constant that's long enough here?  At least make
   563	   //   it configurable via a #define or something?
   564	   if((requiredPathLength >= (MAX_PATH - 50)) ||
   565	      (0 == requiredPathLength))
   566	   {
   567	      ShowMessage("Error getting Windows directory.\n");
   568	      nRetValue = FAILURE;
   569	   }
   570	
   571	   if(SUCCESS == nRetValue)
   572	   {
   573	      // Add a '\' to the path if necessary.
   574	      if('\\' != windowsPath[requiredPathLength])
   575	      {
   576	         windowsPath[requiredPathLength]     = '\\';
   577	         windowsPath[requiredPathLength+1]   = '\0';
   578	      }
   579	   }
   580	
   581	   if(SUCCESS == nRetValue)
   582	   {
   583	      // Delete the main chatpad filter driver.
   584	      _stprintf_s(pathToDelete, MAX_PATH, "%sSystem32\\Drivers\\chatpad_filter.sys", windowsPath);
   585	      if(0 == DeleteFile(pathToDelete))
   586	      {
   587	         char errorString[80];
   588	         sprintf_s(errorString, 80, "Failed to delete filter driver:  0x%08x\n", GetLastError());
   589	         ShowMessage(errorString);
   590	
   591	         nRetValue = FAILURE;
   592	      }
   593	   }
   594	
   595	   return nRetValue;
   596	} // end UninstallChatpadFilterDriver

   598	// Installs or upgrades the chatpad drivers.
   599	int DoInstallOrUpgrade(void)
   600	{
   601	   int nRetValue = SUCCESS;
   602	
   603	   ShowMessage(
   604	     "Preparing to install chatpad driver.\n"
   605	     "WARNING:  Make sure the controller is plugged in before clicking OK!");
   606	
   607	   if(SUCCESS == nRetValue)
   608	   {
   609	      SetStatus("Installing controller filter driver...");
   610	      nRetValue = UpdateChatpadFilterDriver();
   611	   }
   612	
   613	   if(SUCCESS == nRetValue)
   614	   {
   615	      SetStatus("Uninstalling virtual keyboard device drivers...");
   616	      // Try to uninstall the virtual keyboard driver in case it was installed already.
   617	      // TODO eventually make update actually update instead of requiring the uninstall first?
   618	      nRetValue = UninstallVirtualKeyboardDevice();
   619	   }
   620	
   621	   if(SUCCESS == nRetValue)
   622	   {
   623	      SetStatus("Updating virtual keyboard device drivers...");
   624	      nRetValue = UpdateVirtualKeyboardDriver();
   625	   }
   626	
   627	   if(SUCCESS == nRetValue)
   628	   {
   629	      SetStatus("Uninstalling virtual mouse device drivers...");
   630	      // Try to uninstall the virtual mouse driver in case it was installed already.
   631	      // TODO eventually make update actually update instead of requiring the uninstall first?
   632	      nRetValue = UninstallVirtualMouseDevice();
   633	   }
   634	
   635	   if(SUCCESS == nRetValue)
   636	   {
   637	      SetStatus("Updating virtual mouse device drivers...");
   638	      nRetValue = UpdateVirtualMouseDriver();
   639	   }
   640	
   641	   if(SUCCESS == nRetValue)
   642	   {
   643	      SetStatus("Install/upgrade succeeded.");
   644	   }
   645	   else
   646	   {
   647	      SetStatus("Install/upgrade failed.");
   648	   }
   649	
   650	   return nRetValue;
   651	} // end DoInstallOrUpgrade

   694	// Dialog box callback.
   695	LRESULT CALLBACK DlgProc(HWND hwnd, UINT msg, WPARAM wParam, LPARAM lParam)
   696	{
   697	   LRESULT retValue = FALSE;
   698	
   699	   switch(msg)
   700	   {
   701	      case WM_INITDIALOG:
   702	      {
   703	         hwndInstallDialog = hwnd;
   704	         SetStatus("Ready.");
   705	         retValue = TRUE;
   706	      }
   707	
   708	      case WM_COMMAND:
   709	      {
   710	         switch(wParam)
   711	         {
   712	            case IDC_INSTALL_BUTTON:
   713	            {
   714	               if(IDYES == MessageBox(hwnd, "Install/upgrade chatpad drivers?", "Install", MB_YESNO))
   715	               {
   716	                  HWND dialogButtons[NUM_DIALOG_BUTTONS];
   717	                  dialogButtons[0] = GetDlgItem(hwnd, IDC_INSTALL_BUTTON);
   718	                  dialogButtons[1] = GetDlgItem(hwnd, IDC_UNINSTALL_BUTTON);
   719	                  dialogButtons[2] = GetDlgItem(hwnd, IDC_EXIT_BUTTON);
   720	
   721	                  // Disable the buttons.
   722	                  for(int i = 0; i < NUM_DIALOG_BUTTONS; i++)
   723	                  {
   724	                     if(NULL != dialogButtons[i])
   725	                     {
   726	                        EnableWindow(dialogButtons[i], FALSE);
   727	                     }
   728	                  }
   729	
   730	                  // Do the install/upgrade.
   731	                  DoInstallOrUpgrade();
   732	
   733	                  // Enable the buttons.
   734	                  for(int i = 0; i < NUM_DIALOG_BUTTONS; i++)
   735	                  {
   736	                     if(NULL != dialogButtons[i])
   737	                     {
   738	                        EnableWindow(dialogButtons[i], TRUE);
   739	                     }
   740	                  }
   741	               } // end if user confirmed install/upgrade
   742	               retValue = TRUE;
   743	               break;
   744	            } // end case IDC_INSTALL_BUTTON
   745	
   746	            case IDC_UNINSTALL_BUTTON:
   747	            {
   748	               if(IDYES == MessageBox(hwnd, "Uninstall chatpad drivers?", "Uninstall", MB_YESNO))
   749	               {
   750	                  HWND dialogButtons[NUM_DIALOG_BUTTONS];
   751	                  dialogButtons[0] = GetDlgItem(hwnd, IDC_INSTALL_BUTTON);
   752	                  dialogButtons[1] = GetDlgItem(hwnd, IDC_UNINSTALL_BUTTON);
   753	                  dialogButtons[2] = GetDlgItem(hwnd, IDC_EXIT_BUTTON);
   754	
   755	                  // Disable the buttons.
   756	                  for(int i = 0; i < NUM_DIALOG_BUTTONS; i++)
   757	                  {
   758	                     if(NULL != dialogButtons[i])
   759	                     {
   760	                        EnableWindow(dialogButtons[i], FALSE);
   761	                     }
   762	                  }
   763	
   764	                  // Do the uninstall.
   765	                  DoUninstall();
   766	
   767	                  // Enable the buttons.
   768	                  for(int i = 0; i < NUM_DIALOG_BUTTONS; i++)
   769	                  {
   770	                     if(NULL != dialogButtons[i])
   771	                     {
   772	                        EnableWindow(dialogButtons[i], TRUE);
   773	                     }
   774	                  }
   775	               } // end if user confirmed uninstall
   776	               retValue = TRUE;
   777	               break;
   778	            } // end case IDC_UNINSTALL_BUTTON
   779	
   780	            case IDC_EXIT_BUTTON:   // fall-through
   781	            case IDCANCEL:
   782	            {
   783	               EndDialog(hwnd, 0);
   784	               retValue = TRUE;
   785	               break;
   786	            }
   787	
   788	            default:
   789	            {
   790	               break;
   791	            }
   792	         }
   793	      }
   794	      default:
   795	      {
   796	         break;
   797	      }
   798	   } // end switch(msg)
   799	
   800	   return FALSE;
   801	}
