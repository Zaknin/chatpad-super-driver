    38	// The Device Class Definition for Human Interface Devices (HID) document, version 1.11,

    41	typedef UCHAR HID_REPORT_DESCRIPTOR, *PHID_REPORT_DESCRIPTOR;

    43	// TODO do I need to concern myself with the boot protocol?
    44	//   (see http://msdn.microsoft.com/en-us/library/ff543350%28v=VS.85%29.aspx)
    45	
    46	// Make sure this structure is byte-aligned and not padded to match word boundaries.
    47	#pragma pack(1)
    48	
    49	// Note that this structure must exactly match the report described by the
    50	// descriptor below.
    51	typedef struct _CHATPAD_MOUSE_HID_INPUT_REPORT
    52	{
    53	   // The rightmost bit (least significant, 0x01) of this byte corresponds to a left click.
    54	   // Then, the next bit to the left (0x02) corresponds to mouse button a right click.
    55	   // Then, the next bit to the left (0x04) corresponds to mouse button a middle click.
    56	   BYTE     buttonStateByte;
    57	
    58	   // These values can range from -127 to 127 inclusive and represent a relative mouse movement.
    59	   // Note that at small offsets, these are roughly equivalent to pixels, but
    60	   // at larger offsets (more than 6-8 pixels) they start varying somewhat and
    61	   // becoming affected by mouse acceleration, apparently.
    62	   //
    63	   // See this page for additional information, and note that the algorithm was
    64	   // apparently "completely wrong" for Windows XP according to justchat_1 on
    65	   // mp3car.com:  http://www.microsoft.com/whdc/archive/pointer-bal.mspx
    66	   //
    67	   // The poster justchat_1 also said that it is fixed in Vista but he could not find any
    68	   // documentation on the details.
    69	   BYTE     offsetX;
    70	   BYTE     offsetY;
    71	
    72	   // This value can range from -127 to 127 inclusive and represents a relative vertical scroll wheel movement.
    73	   // Note that currently, any negative value seems to represent only one
    74	   // "tick", and any positive value only one "tick" in the opposite direction.
    75	   // TODO is there a way to make the wheel go more than one "tick" at a time?
    76	   BYTE     offsetVerticalScrollWheel;
    77	} CHATPAD_MOUSE_HID_INPUT_REPORT, *PCHATPAD_MOUSE_HID_INPUT_REPORT;

    81	// HID report descriptor (mouse), based off an example in HID1_11.PDF and the HID report descriptor in vmulti.
    82	const HID_REPORT_DESCRIPTOR chatpadMouseHidReportDescriptor[] =
    83	{
    84	   0x05, 0x01,    // Usage page (generic desktop).
    85	   0x09, 0x02,    // Usage (mouse).
    86	   0xA1, 0x01,    // Collection (application).
    87	
    88	   0x09, 0x01,    // Usage (pointer).
    89	   0xA1, 0x00,    // Collection (physical)
    90	
    91	   0x05, 0x09,    // Usage page (buttons)
    92	   0x19, 0x01,    // Usage minimum (1).
    93	   0x29, 0x03,    // Usage maximum (3).
    94	   0x15, 0x00,    // Logical minimum (0).
    95	   0x25, 0x01,    // Logical maximum (1).
    96	   0x95, 0x03,    // Report count (3).
    97	   0x75, 0x01,    // Report size (1).
    98	   0x81, 0x02,    // Input (data, variable, absolute):   3 mouse-button bits
    99	   0x95, 0x01,    // Report count (1).
   100	   0x75, 0x05,    // Report size (5).
   101	   0x81, 0x01,    // Input (data, constant):             5 padding bits
   102	
   103	   // TODO should I make the X and Y report values more complex, like in the vmulti HID report descriptor,
   104	   //   to simulate a more complex (higher DPI?) mouse or whatever?
   105	   0x05, 0x01,    // Usage page (generic desktop)
   106	   0x09, 0x30,    // Usage (X).
   107	   0x09, 0x31,    // Usage (Y).
   108	   0x15, 0x81,    // Logical minimum (-127).
   109	   0x25, 0x7F,    // Logical maximum (127).
   110	   0x95, 0x02,    // Report count (2).
   111	   0x75, 0x08,    // Report size (8).
   112	   0x81, 0x06,    // Input (data, variable, relative ):  2 position bytes (X and Y)
   113	
   114	   0x05, 0x01,    // Usage page (generic desktop)
   115	   0x09, 0x38,    // Usage (wheel).
   116	   0x15, 0x81,    // Logical minimum (-127).
   117	   0x25, 0x7F,    // Logical maximum (127).
   118	   0x95, 0x01,    // Report count (1).
   119	   0x75, 0x08,    // Report size (8).
   120	   0x81, 0x06,    // Input (data, variable, relative ):  relative mouse wheel value
   121	
   122	   0xC0,          // End (physical) collection
   123	   0xC0           // End (application) collection
   124	}; // end chatpadMouseHidReportDescriptor

   168	// Called when driver is first loaded.
   169	extern "C"
   170	NTSTATUS
   171	DriverEntry(
   172	  __in PDRIVER_OBJECT  driverObject,
   173	  __in PUNICODE_STRING registryPath)
   174	{
   175	   NTSTATUS                retStatus = STATUS_SUCCESS;
   176	   WDF_DRIVER_CONFIG       driverConfig;
   177	   WDF_OBJECT_ATTRIBUTES   objectAttributes;
   178	
   179	   ChatpadTrace(("XBox 360 Controller Chatpad Virtual Mouse KMDF Layer by GAFBlizzard.\n"));
   180	   ChatpadTrace(("Build %s %s\n", __DATE__, __TIME__));
   181	
   182	   // This is just an information printout so I can check the size of a DEVICE_CONTEXT structure
   183	   // (only the structure itself, not any additional allocated memory).
   184	   ChatpadTrace(("sizeof(DEVICE_CONTEXT) == %u\n", sizeof(DEVICE_CONTEXT)));
   185	   ChatpadTrace(("sizeof(CHATPAD_MOUSE_HID_INPUT_REPORT)  == %u\n", sizeof(CHATPAD_MOUSE_HID_INPUT_REPORT)));
   186	// TODO NEXT REMOVE BELOW
   187	ChatpadTrace(("report descriptor size is %u\n", sizeof(chatpadMouseHidReportDescriptor))); 
   188	
   189	   WDF_DRIVER_CONFIG_INIT(&driverConfig,
   190	                          ChatpadMouseKMDFEvtDeviceAdd);
   191	
   192	   // Set up the driver unload function.
   193	   driverConfig.EvtDriverUnload = DriverUnload;
   194	
   195	   // Register a cleanup callback.
   196	   WDF_OBJECT_ATTRIBUTES_INIT(&objectAttributes);
   197	   objectAttributes.EvtCleanupCallback = ChatpadMouseKMDFDriverContextCleanup;
   198	
   199	   // Create framework driver object (FDO).
   200	   retStatus = WdfDriverCreate(
   201	     driverObject,
   202	     registryPath,
   203	     &objectAttributes,
   204	     &driverConfig,
   205	     WDF_NO_HANDLE);
   206	   if(STATUS_SUCCESS != retStatus)
   207	   {
   208	      ChatpadTrace(("WdfDriverCreate failed:  0x%0x\n", retStatus));
   209	   }
   210	
   211	   // Create collection for different instances of the top-level (non-control) device.
   212	   if(STATUS_SUCCESS == retStatus)
   213	   {
   214	      retStatus = WdfCollectionCreate(
   215	        WDF_NO_OBJECT_ATTRIBUTES,
   216	        &chatpadMouseKMDFDeviceCollection);
   217	      if(STATUS_SUCCESS != retStatus)
   218	      {
   219	         ChatpadTrace(("WdfCollectionCreate failed:  0x%x\n", retStatus));
   220	      }
   221	   }
   222	
   223	   if(STATUS_SUCCESS == retStatus)
   224	   {
   225	      retStatus = WdfWaitLockCreate(
   226	        WDF_NO_OBJECT_ATTRIBUTES,
   227	        &chatpadMouseKMDFDeviceCollectionLock);
   228	      if(STATUS_SUCCESS != retStatus)
   229	      {
   230	         ChatpadTrace(("WdfWaitLockCreate failed:  0x%x\n", retStatus));
   231	      }
   232	   }
   233	
   234	   return retStatus;
   235	} // end DriverEntry

   323	NTSTATUS
   324	ChatpadMouseKMDFEvtDeviceAdd(
   325	  IN WDFDRIVER       driver,
   326	  IN PWDFDEVICE_INIT deviceInit)
   327	{
   328	   NTSTATUS                      retStatus = STATUS_SUCCESS;
   329	   WDF_IO_QUEUE_CONFIG           ioQueueConfig;
   330	   WDF_OBJECT_ATTRIBUTES         objectAttributes;
   331	   WDFDEVICE                     wdfDevice;
   332	   PDEVICE_CONTEXT               devContext = NULL;
   333	   WDFQUEUE                      queue;
   334	   WDF_PNPPOWER_EVENT_CALLBACKS  powerCallbacks;
   335	   UCHAR                         minorFunction;
   336	
   337	   if(STATUS_SUCCESS == retStatus)
   338	   {
   339	      // Mark ourselves as a filter, causing requests we don't care about to be
   340	      // passed along etc.
   341	      WdfFdoInitSetFilter(deviceInit);
   342	
   343	      // Mark our device type as unknown.
   344	      // TODO this was from another example, I think, but the hidusbfx2 example does not have it.
   345	      //   Is there something I should understand about what it does and whether I need it?
   346	      // TODO should this be set eventually since this driver is not generic, or is the
   347	      //      automatic stuff fine?
   348	//      WdfDeviceInitSetDeviceType(deviceInit, FILE_DEVICE_UNKNOWN);
   349	
   350	      // Override IRP_MN_QUERY_ID, like vmulti does.
   351	      // TODO is overriding this IRP necessary?  It may only be necessary for multifunction drivers.
   352	      minorFunction = IRP_MN_QUERY_ID;
   353	      retStatus = WdfDeviceInitAssignWdmIrpPreprocessCallback(
   354	        deviceInit,
   355	        ChatpadMouseKMDFEvtWdmPreprocessMnQueryId,
   356	        IRP_MJ_PNP,        // Major function.
   357	        &minorFunction,    // Array of minor functions.
   358	        1);                // Number of minor functions.
   359	      if(STATUS_SUCCESS != retStatus)
   360	      {
   361	         ChatpadTrace(("WdfDeviceInitAssignWdmIrpPreprocessCallback failed:  0x%x\n", retStatus));
   362	      }
   363	   }
   364	
   365	   if(STATUS_SUCCESS == retStatus)
   366	   {
   367	      // Register a prepare hardware callback to create a WDF USB device object.
   368	      WDF_PNPPOWER_EVENT_CALLBACKS_INIT(&powerCallbacks);
   369	      powerCallbacks.EvtDevicePrepareHardware = ChatpadMouseKMDFEvtPrepareHardware;
   370	      powerCallbacks.EvtDeviceD0Entry = ChatpadMouseKMDFD0Entry;
   371	      powerCallbacks.EvtDeviceD0Exit = ChatpadMouseKMDFD0Exit;
   372	
   373	      WdfDeviceInitSetPnpPowerEventCallbacks(deviceInit, &powerCallbacks);
   374	
   375	      WDF_OBJECT_ATTRIBUTES_INIT_CONTEXT_TYPE(&objectAttributes, DEVICE_CONTEXT);
   376	
   377	      // Register for cleanup notification.
   378	      objectAttributes.EvtCleanupCallback = ChatpadMouseKMDFContextCleanup;
   379	
   380	      // Create framework device object, attach to the lower stack etc.
   381	      retStatus = WdfDeviceCreate(&deviceInit, &objectAttributes, &wdfDevice);
   382	      if(STATUS_SUCCESS != retStatus)
   383	      {
   384	         ChatpadTrace(("WdfDeviceCreate failed:  0x%x\n", retStatus));
   385	      }
   386	   } // end if(STATUS_SUCCESS == retStatus)
   387	
   388	   if(STATUS_SUCCESS == retStatus)
   389	   {
   390	      devContext = ChatpadMouseKMDFGetDeviceContext(wdfDevice);
   391	      if(NULL != devContext)
   392	      {
   393	         // Initialize DEVICE_CONTEXT values.
   394	         ChatpadTrace(("Initializing device context.\n"));
   395	
   396	         // TODO is it possible for a driver to read from the registry and detect whether to start up enabled?
   397	         //   (apparently so, see http://flylib.com/books/en/3.141.1.102/1/)
   398	         devContext->mouseEnabled = FALSE;
   399	
   400	         if(STATUS_SUCCESS == retStatus)
   401	         {
   402	            // Initialize default queue config structure.
   403	            WDF_IO_QUEUE_CONFIG_INIT_DEFAULT_QUEUE(&ioQueueConfig, WdfIoQueueDispatchParallel);
   404	
   405	            // Select what we want from the queue.
   406	            // This should catch IOCTL_INTERNAL_USB_SUBMIT_URB.
   407	            ioQueueConfig.EvtIoInternalDeviceControl =
   408	              ChatpadMouseKMDFInternalIOCTLHandler;
   409	
   410	            // Create the default queue.
   411	            retStatus = WdfIoQueueCreate(
   412	              wdfDevice,
   413	              &ioQueueConfig,
   414	              WDF_NO_OBJECT_ATTRIBUTES,
   415	              NULL);
   416	            if(STATUS_SUCCESS == retStatus)
   417	            {
   418	               ChatpadTrace(("Created chatpad mouse KMDF layer default queue successfully.\n"));
   419	            }
   420	            else
   421	            {
   422	               ChatpadTrace(("WdfIoQueueCreate failed:  0x%x\n", retStatus));
   423	            }
   424	         } // end if(STATUS_SUCCESS == retStatus)
   425	
   426	         if(STATUS_SUCCESS == retStatus)
   427	         {
   428	            // Create manual I/O queue for handling HID read requests from
   429	            // above us.
   430	            WDF_IO_QUEUE_CONFIG hidReadRequestQueueConfig;
   431	
   432	            WDF_IO_QUEUE_CONFIG_INIT(&hidReadRequestQueueConfig, WdfIoQueueDispatchManual);
   433	            hidReadRequestQueueConfig.PowerManaged = WdfFalse;
   434	
   435	            retStatus =
   436	              WdfIoQueueCreate(
   437	                wdfDevice,
   438	                &hidReadRequestQueueConfig,
   439	                WDF_NO_OBJECT_ATTRIBUTES,
   440	                &(devContext->hidReadRequestQueue));
   441	            if(STATUS_SUCCESS == retStatus)
   442	            {
   443	               ChatpadTrace(("Created HID read request queue successfully.\n"));
   444	            }
   445	            else
   446	            {
   447	               ChatpadTrace(("WdfIoQueueCreate failed:  0x%x\n", retStatus));
   448	            }
   449	         } // end if(STATUS_SUCCESS == retStatus)
   450	      } // end if(NULL != devContext)
   451	      else
   452	      {
   453	         retStatus = STATUS_INTERNAL_ERROR;
   454	      }
   455	   } // end if(STATUS_SUCCESS == retStatus)
   456	
   457	   if(STATUS_SUCCESS == retStatus)
   458	   {
   459	      // Add the device to the collection.
   460	      WdfWaitLockAcquire(chatpadMouseKMDFDeviceCollectionLock, NULL);
   461	
   462	      retStatus = WdfCollectionAdd(
   463	        chatpadMouseKMDFDeviceCollection,
   464	        wdfDevice);
   465	      if(STATUS_SUCCESS != retStatus)
   466	      {
   467	         ChatpadTrace(("WdfCollectionAdd failed:  0x%x\n", retStatus));
   468	      }
   469	
   470	      WdfWaitLockRelease(chatpadMouseKMDFDeviceCollectionLock);
   471	   }
   472	
   473	   // Create the control device.
   474	   if(STATUS_SUCCESS == retStatus)
   475	   {
   476	      retStatus = ChatpadMouseKMDFCreateControlDevice(wdfDevice);
   477	      // ChatpadMouseKMDFCreateControlDevice should print an error if it fails.
   478	   }
   479	
   480	   return retStatus;
   481	} // end ChatpadMouseKMDFEvtDeviceAdd

   496	// This code is hand-written but is patterned after VMultiEvtWdmPreprocessMnQueryID
   497	// from vmulti (http://code.google.com/p/vmulti/).
   498	NTSTATUS
   499	ChatpadMouseKMDFEvtWdmPreprocessMnQueryId(
   500	  WDFDEVICE       device,
   501	  PIRP            irp)
   502	{
   503	   NTSTATUS             retStatus               = STATUS_SUCCESS;
   504	   PIO_STACK_LOCATION   currentStackLocation    = NULL;
   505	   PIO_STACK_LOCATION   previousStackLocation   = NULL;
   506	   PDEVICE_OBJECT       deviceObject            = NULL;
   507	   PWCHAR               buffer                  = NULL;
   508	
   509	   currentStackLocation = IoGetCurrentIrpStackLocation(irp);
   510	   deviceObject = WdfDeviceWdmGetDeviceObject(device);
   511	
   512	   // According to vmulti, this check is required to filter out QUERY_IDs
   513	   // forwarded by the HIDCLASS for the parent framework device object.
   514	   // These IDs, says vmulti, are sent by the PNP manager for the parent
   515	   // framework device object if you root-enumerate this driver.
   516	   // TODO study the documentation to fully understand how all this works!
   517	   // TODO also study how stack locations work to fully understand the pointer arithmetic.
   518	   previousStackLocation =
   519	     (reinterpret_cast<PIO_STACK_LOCATION>(
   520	        reinterpret_cast<PUCHAR>(currentStackLocation) +
   521	        sizeof(IO_STACK_LOCATION)));
   522	   if(NULL == previousStackLocation)
   523	   {
   524	      ChatpadTrace(("Error trying to get previous stack pointer.\n"));
   525	      retStatus = STATUS_INTERNAL_ERROR;
   526	   }
   527	
   528	   if(STATUS_SUCCESS == retStatus)
   529	   {
   530	      if(previousStackLocation->DeviceObject == deviceObject)
   531	      {
   532	         // The vmulti code says that filtering out this basically prevents the
   533	         // found new hardware popup for a root-enumerated device (on reboot
   534	         // only?).
   535	         // TODO study how this works
   536	         retStatus = irp->IoStatus.Status;
   537	      }
   538	      else
   539	      {
   540	         if((BusQueryDeviceID == currentStackLocation->Parameters.QueryId.IdType) ||
   541	            (BusQueryHardwareIDs == currentStackLocation->Parameters.QueryId.IdType))
   542	         {
   543	            // In this case, according to vmulti, HIDClass is asking for a child
   544	            // device ID and hardware IDs.
   545	            // TODO study how this works regarding the device/hardware IDs
   546	
   547	            // TODO NEXT do we ever need to free this, or is that handled
   548	            //   automatically by the system, or when the driver is unloaded?
   549	            buffer = static_cast<PWCHAR>(ExAllocatePoolWithTagWrapper(
   550	              NonPagedPool,
   551	              CHATPADMOUSEKMDF_HARDWARE_IDS_LENGTH,
   552	              CHATPADMOUSEKMDF_POOL_TAG));
   553	
   554	            if(NULL != buffer)
   555	            {
   556	               RtlCopyMemory(
   557	                 buffer,
   558	                 CHATPADMOUSEKMDF_HARDWARE_IDS,
   559	                 CHATPADMOUSEKMDF_HARDWARE_IDS_LENGTH);
   560	               // TODO why do we have to cast to a ULONG_PTR?  And why ULONG_PTR, not PULONG?
   561	               irp->IoStatus.Information = reinterpret_cast<ULONG_PTR>(buffer);
   562	            }
   563	            else
   564	            {
   565	               ChatpadTrace(("Failed to allocate memory for hardware IDs string.\n"));
   566	               retStatus = STATUS_INSUFFICIENT_RESOURCES;
   567	            }
   568	            irp->IoStatus.Status = retStatus;
   569	         }
   570	         else
   571	         {
   572	            retStatus = irp->IoStatus.Status;
   573	         }
   574	      } // end case where previous stack location's device object is not the current device object
   575	
   576	      IoCompleteRequest(irp, IO_NO_INCREMENT);
   577	   } // end if(STATUS_SUCCESS == retStatus)
   578	
   579	   ChatpadTrace(("ChatpadMouseKMDFEvtWdmPreprocessMnQueryId returning status 0x%x\n", retStatus));
   580	
   581	   return retStatus;
   582	} // end ChatpadMouseKMDFEvtWdmPreprocessMnQueryId

   650	VOID
   651	ChatpadMouseKMDFControlDeviceCloseHandler(
   652	  WDFFILEOBJECT   fileObject)
   653	{
   654	   NTSTATUS apiStatus = STATUS_SUCCESS;
   655	
   656	   ChatpadTrace(("The IRQL in the mouse close handler is %u.\n", KeGetCurrentIrql()));
   657	
   658	   // Decrement the open count.
   659	   WdfWaitLockAcquire(chatpadMouseKMDFDeviceCollectionLock, NULL);
   660	
   661	   // Since I have a wait lock, can I safely just modify it like this?
   662	   if(numOpenHandles > 0)
   663	   {
   664	      numOpenHandles--;
   665	   }
   666	   else
   667	   {
   668	      ChatpadTrace(("Error, there are supposedly 0 open handles but the close handler was called...\n"));
   669	   }
   670	
   671	   ChatpadTrace(("There are now %lu open handles to the chatpad mouse KMDF control device.\n", numOpenHandles));
   672	
   673	   if(0 == numOpenHandles)
   674	   {
   675	      // TODO there is probably a cleaner way to do this than having one extension
   676	      //   have a pointer to the other one etc...maybe?
   677	      //   I want to make sure the no-mouse-events-active HID response gets sent, though.
   678	      PCHATPAD_MOUSE_KMDF_CONTROL_DEVICE_EXTENSION    controlExtension  = NULL;
   679	      PDEVICE_CONTEXT                                 devContext        = NULL;
   680	      WDFREQUEST                                      requestFromAbove;
   681	
   682	      if(NULL == chatpadMouseKMDFControlDevice)
   683	      {
   684	         ChatpadTrace(("Error, control device handle is NULL.\n"));
   685	         apiStatus = STATUS_INTERNAL_ERROR;
   686	      }
   687	
   688	      if(STATUS_SUCCESS == apiStatus)
   689	      {
   690	         controlExtension = ChatpadMouseKMDFControlGetData(chatpadMouseKMDFControlDevice);
   691	
   692	         if(NULL == controlExtension)
   693	         {
   694	            ChatpadTrace(("Failed to get control device extension in close handler.\n"));
   695	            apiStatus = STATUS_INTERNAL_ERROR;
   696	         }
   697	      }
   698	
   699	      if(STATUS_SUCCESS == apiStatus)
   700	      {
   701	         devContext = controlExtension->mouseDeviceContext;
   702	         if(NULL == devContext)
   703	         {
   704	            ChatpadTrace(("Failed to get device context in close handler.\n"));
   705	            apiStatus = STATUS_INTERNAL_ERROR;
   706	         }
   707	      }
   708	
   709	      if(STATUS_SUCCESS == apiStatus)
   710	      {
   711	         // Mark the virtual mouse as disabled when the application disconnects.
   712	         // TODO could messing with the device context here cause synchronization issues?
   713	         devContext->mouseEnabled = FALSE;
   714	         ChatpadTrace(("Virtual mouse disabled on application disconnect.\n"));
   715	
   716	         apiStatus = WdfIoQueueRetrieveNextRequest(
   717	           devContext->hidReadRequestQueue,
   718	           &requestFromAbove);
   719	         if(STATUS_SUCCESS != apiStatus)
   720	         {
   721	            ChatpadTrace(("No top-level HID read report request available.\n"));
   722	            apiStatus = STATUS_INTERNAL_ERROR;
   723	         }
   724	      }
   725	
   726	      if(STATUS_SUCCESS == apiStatus)
   727	      {
   728	         PCHATPAD_MOUSE_HID_INPUT_REPORT     inputReport                   = NULL;
   729	         ULONG                               bytesTransferred              = 0;
   730	         size_t                              bytesToCopy                   = 0;
   731	         size_t                              requestFromAboveBufferLength  = 0;
   732	
   733	         bytesToCopy = sizeof(CHATPAD_MOUSE_HID_INPUT_REPORT);
   734	         apiStatus = WdfRequestRetrieveOutputBuffer(
   735	           requestFromAbove,
   736	           bytesToCopy,
   737	           reinterpret_cast<PVOID*>(&inputReport),
   738	           &requestFromAboveBufferLength);
   739	         if((STATUS_SUCCESS == apiStatus)                   &&
   740	            (NULL != inputReport)                           &&
   741	            (requestFromAboveBufferLength >= bytesToCopy))
   742	         {
   743	            // Fill in the report fields to indicate that no mouse actions are
   744	            // active, so the virtual mouse does not end up in a state where
   745	            // the mouse is moving or buttons are held down when the control
   746	            // application disconnects.
   747	            inputReport->buttonStateByte           = 0x00;
   748	            inputReport->offsetX                   = 0x00;
   749	            inputReport->offsetY                   = 0x00;
   750	            inputReport->offsetVerticalScrollWheel = 0x00;
   751	
   752	            // Indicate to the requester how many bytes were transferred.
   753	            bytesTransferred = bytesToCopy;
   754	         }
   755	         else
   756	         {
   757	            ChatpadTrace(("Unable to get valid buffer from top-level HID read report request:  0x%x\n",
   758	                          apiStatus));
   759	            apiStatus = STATUS_INTERNAL_ERROR;
   760	         }
   761	
   762	         // Return the report to the requester.
   763	         WdfRequestCompleteWithInformation(requestFromAbove, apiStatus, bytesTransferred);
   764	
   765	/*
   766	         if(STATUS_SUCCESS == apiStatus)
   767	         {
   768	            ChatpadTrace(("Successfully sent HID response indicating no mouse action is active.\n"));
   769	         }
   770	*/
   771	      } // end if(STATUS_SUCCESS == apiStatus)
   772	   } // end if(0 == numOpenHandles)
   773	
   774	   WdfWaitLockRelease(chatpadMouseKMDFDeviceCollectionLock);
   775	} // end ChatpadMouseKMDFControlDeviceCloseHandler

   787	VOID
   788	ChatpadMouseKMDFContextCleanup(
   789	  IN WDFOBJECT       object)
   790	{
   791	   NTSTATUS apiStatus = STATUS_SUCCESS;
   792	   // TODO is static_cast or reinterpret_cast preferred here?  I'm doing an
   793	   //      oldstyle cast because that's what the suggestion on osronline
   794	   //      said to do.  :)
   795	   //      Normally, for void* to something* I'd do static_cast, and from
   796	   //      thingA* to thingB* I'd do reinterpret_cast.
   797	   WDFDEVICE device = (WDFDEVICE) object;
   798	   PDEVICE_CONTEXT devContext = NULL;
   799	   ULONG deviceCount;
   800	
   801	   ChatpadTrace(("Normal context cleanup handler called.\n"));
   802	
   803	   devContext = ChatpadMouseKMDFGetDeviceContext(device);
   804	   if((NULL != device) && (NULL != devContext))
   805	   {
   806	      WdfWaitLockAcquire(chatpadMouseKMDFDeviceCollectionLock, NULL);
   807	
   808	      // Clean up device context resources.
   809	
   810	      deviceCount = WdfCollectionGetCount(chatpadMouseKMDFDeviceCollection);
   811	      if(deviceCount == 1)
   812	      {
   813	         // If we are the last instance, then delete the control device so that
   814	         // the driver can unload.  Apparently there was a bug in WDF 1.0, but 
   815	         // apparently this sort of thing is fine in WDF 1.9 (and even earlier?).
   816	         //
   817	         // According to the toaster KMDF sideband filter example, the deletion
   818	         // below has to be done while the lock on the collection is acquired
   819	         // since it protects the global device variable, and another thread
   820	         // cannot try to create while we are deleting the device.
   821	
   822	         if(0 == numOpenHandles)
   823	         {
   824	            ChatpadMouseKMDFDeleteControlDevice();
   825	         }
   826	         else
   827	         {
   828	            ChatpadTrace(("The device was not deleted since there are still %lu open handles to it.\n",
   829	                          numOpenHandles));
   830	         }
   831	      } // end if(deviceCount == 1)
   832	
   833	      // Even if the control device was not deleted, however, we mark it as unavailable.
   834	      chatpadMouseKMDFControlDeviceAvailable = FALSE;
   835	
   836	      WdfCollectionRemove(chatpadMouseKMDFDeviceCollection, device);
   837	
   838	      WdfWaitLockRelease(chatpadMouseKMDFDeviceCollectionLock);
   839	   } // end if((NULL != device) && (NULL != devContext))
   840	   else
   841	   {
   842	      // This should never happen.
   843	      ChatpadTrace(("Error, device and/or devContext are NULL while cleaning up.\n"));
   844	   }
   845	}

   847	// Creates a control device object so that an application can talk to the filter driver.
   848	NTSTATUS
   849	ChatpadMouseKMDFCreateControlDevice(
   850	  WDFDEVICE device)
   851	{
   852	   NTSTATUS                retStatus = STATUS_SUCCESS;
   853	   BOOLEAN                 createDevice = FALSE;
   854	   PWDFDEVICE_INIT         deviceInit = NULL;
   855	   WDF_FILEOBJECT_CONFIG   fileObjectConfig;
   856	   WDFDEVICE               newChatpadMouseKMDFControlDevice = NULL;
   857	   WDF_OBJECT_ATTRIBUTES   controlAttributes;
   858	   DECLARE_CONST_UNICODE_STRING(ntDeviceName, CHATPAD_MOUSE_KMDF_CONTROL_NAME);
   859	   DECLARE_CONST_UNICODE_STRING(symbolicLinkName, CHATPAD_MOUSE_KMDF_CONTROL_LINK);
   860	   WDFQUEUE                newQueue;
   861	   WDF_IO_QUEUE_CONFIG     ioQueueConfig;
   862	
   863	   // Determine whether we should create the device, or if it already exists.
   864	   WdfWaitLockAcquire(chatpadMouseKMDFDeviceCollectionLock, NULL);
   865	
   866	   // TODO REMOVE after testing
   867	   ChatpadTrace(("Collection device count is %d.\n", WdfCollectionGetCount(chatpadMouseKMDFDeviceCollection)));
   868	   // TODO is this a safe way to synchronize things?
   869	   if(NULL == chatpadMouseKMDFControlDevice)
   870	   {
   871	      createDevice = TRUE;
   872	   }
   873	
   874	   WdfWaitLockRelease(chatpadMouseKMDFDeviceCollectionLock);
   875	
   876	   if(!createDevice)
   877	   {
   878	      ChatpadTrace(("Chatpad mouse KMDF control device was already created.\n"));
   879	   }
   880	   else
   881	   {
   882	      ChatpadTrace(("Creating chatpad mouse KMDF control device.\n"));
   883	
   884	      // Allocate and initialize a WDFDEVICE_INIT structure for the control device.
   885	      deviceInit = WdfControlDeviceInitAllocate(
   886	        WdfDeviceGetDriver(device),
   887	        // Permissions:  Allow kernel, system, and admin control over the device,
   888	        //   but do not allow any other users to access it.
   889	        // TODO does this work with XP?
   890	        &SDDL_DEVOBJ_SYS_ALL_ADM_ALL);
   891	      if(NULL == deviceInit)
   892	      {
   893	         retStatus = STATUS_INSUFFICIENT_RESOURCES;
   894	      }
   895	
   896	      if(STATUS_SUCCESS == retStatus)
   897	      {
   898	         // Make the control device exclusive.  I only want one application talking to it at a time.
   899	         WdfDeviceInitSetExclusive(deviceInit, TRUE);
   900	         // Get device name.
   901	         retStatus = WdfDeviceInitAssignName(deviceInit, &ntDeviceName);
   902	      }
   903	
   904	      // Set up callbacks so we can keep track of whether user mode has the
   905	      // control device open.
   906	      WDF_FILEOBJECT_CONFIG_INIT(
   907	        &fileObjectConfig,
   908	        ChatpadMouseKMDFControlDeviceCreateHandler,
   909	        ChatpadMouseKMDFControlDeviceCloseHandler,
   910	        WDF_NO_EVENT_CALLBACK);
   911	
   912	      WdfDeviceInitSetFileObjectConfig(
   913	        deviceInit,
   914	        &fileObjectConfig,
   915	        WDF_NO_OBJECT_ATTRIBUTES);
   916	
   917	      if(STATUS_SUCCESS == retStatus)
   918	      {
   919	         WDF_OBJECT_ATTRIBUTES_INIT_CONTEXT_TYPE(
   920	           &controlAttributes,
   921	           CHATPAD_MOUSE_KMDF_CONTROL_DEVICE_EXTENSION);
   922	         // Create the control device.
   923	         retStatus = WdfDeviceCreate(
   924	           &deviceInit,
   925	           &controlAttributes,
   926	           &newChatpadMouseKMDFControlDevice);
   927	      }
   928	
   929	      // Initialize control extension.
   930	      PDEVICE_CONTEXT devContext = ChatpadMouseKMDFGetDeviceContext(device);
   931	      if(NULL != devContext)
   932	      {
   933	         PCHATPAD_MOUSE_KMDF_CONTROL_DEVICE_EXTENSION controlExtension =
   934	           ChatpadMouseKMDFControlGetData(newChatpadMouseKMDFControlDevice);
   935	
   936	         if(NULL != controlExtension)
   937	         {
   938	            ChatpadTrace(("Initializing control device extension.\n"));
   939	            controlExtension->mouseDeviceContext = devContext;
   940	         }
   941	         else
   942	         {
   943	            ChatpadTrace(("Error, could not initialize control device extension.\n"));
   944	            retStatus = STATUS_INTERNAL_ERROR;
   945	         }
   946	      }
   947	      else
   948	      {
   949	         retStatus = STATUS_INTERNAL_ERROR;
   950	      }
   951	
   952	      if(STATUS_SUCCESS == retStatus)
   953	      {
   954	         // Create symbolic link so user mode can open the device.
   955	         retStatus = WdfDeviceCreateSymbolicLink(
   956	           newChatpadMouseKMDFControlDevice,
   957	           &symbolicLinkName);
   958	      }
   959	
   960	      if(STATUS_SUCCESS == retStatus)
   961	      {
   962	         // Configure default queue so that requests to the control device will be parallel.
   963	         WDF_IO_QUEUE_CONFIG_INIT_DEFAULT_QUEUE(
   964	           &ioQueueConfig,
   965	//           WdfIoQueueDispatchSequential);
   966	           WdfIoQueueDispatchParallel);
   967	
   968	         // Select what we want from the queue.
   969	
   970	         // This should catch application-command IOCTLs.
   971	         ioQueueConfig.EvtIoDeviceControl =
   972	           ChatpadMouseKMDFIOCTLHandler;
   973	
   974	         // Create the queue.
   975	         retStatus = WdfIoQueueCreate(
   976	           newChatpadMouseKMDFControlDevice,
   977	           &ioQueueConfig,
   978	           WDF_NO_OBJECT_ATTRIBUTES, // TODO what does this parameter do?
   979	           &newQueue);
   980	      } // end if(STATUS_SUCCESS == retStatus)
   981	
   982	      if(STATUS_SUCCESS == retStatus)
   983	      {
   984	         WdfControlFinishInitializing(newChatpadMouseKMDFControlDevice);
   985	         WdfWaitLockAcquire(chatpadMouseKMDFDeviceCollectionLock, NULL);
   986	         chatpadMouseKMDFControlDevice = newChatpadMouseKMDFControlDevice;
   987	         WdfWaitLockRelease(chatpadMouseKMDFDeviceCollectionLock);
   988	
   989	         ChatpadTrace(("Chatpad mouse KMDF control device created successfully.\n"));
   990	      }
   991	      else
   992	      {
   993	         ChatpadTrace(("WdfIoQueueCreate failed:  0x%x\n", retStatus));
   994	      }
   995	   } // end if creating the control device
   996	
   997	   // Handle error cases.
   998	   if(STATUS_SUCCESS != retStatus)
   999	   {
  1000	      ChatpadTrace(("Failed to create chatpad mouse KMDF control device:  0x%x\n", retStatus));
  1001	
  1002	      if(NULL != deviceInit)
  1003	      {
  1004	         WdfDeviceInitFree(deviceInit);
  1005	         deviceInit = NULL;
  1006	      }
  1007	
  1008	      if(NULL != newChatpadMouseKMDFControlDevice)
  1009	      {
  1010	         // Release the newly created, but uninitialized, object.
  1011	         WdfObjectDelete(newChatpadMouseKMDFControlDevice);
  1012	         newChatpadMouseKMDFControlDevice = NULL;
  1013	      }
  1014	   }
  1015	
  1016	   return retStatus;
  1017	} // end ChatpadMouseKMDFCreateControlDevice

  1035	VOID
  1036	ChatpadMouseKMDFInternalIOCTLHandler(
  1037	  IN WDFQUEUE     queue,
  1038	  IN WDFREQUEST   request,
  1039	  IN size_t       outputBufferLength,
  1040	  IN size_t       inputBufferLength,
  1041	  IN ULONG        ioControlCode)
  1042	{
  1043	   NTSTATUS                   apiStatus               = STATUS_SUCCESS;
  1044	   WDFDEVICE                  wdfDevice;
  1045	   PDEVICE_CONTEXT            devContext              = NULL;
  1046	   // Whether this handler should complete the request.
  1047	   BOOLEAN                    completeRequestNow      = TRUE;
  1048	
  1049	//   ChatpadTrace(("Chatpad mouse KMDF layer internal IOCTL handler called with ioControlCode == %x\n",
  1050	//                 ioControlCode));
  1051	
  1052	   wdfDevice = WdfIoQueueGetDevice(queue);
  1053	
  1054	   // Get device context so we know where to forward the request.
  1055	   devContext = ChatpadMouseKMDFGetDeviceContext(wdfDevice);
  1056	
  1057	// TODO what control codes do I need to support for a mouse?
  1058	   if(IOCTL_HID_GET_DEVICE_DESCRIPTOR == ioControlCode)
  1059	   {
  1060	      apiStatus = ChatpadMouseKMDFGetHidDescriptor(wdfDevice, request);
  1061	   }
  1062	   else if(IOCTL_HID_GET_DEVICE_ATTRIBUTES == ioControlCode)
  1063	   {
  1064	// TODO NEXT do I need to support this?  If so I guess I have to provide placeholder vendor/product IDs.
  1065	      apiStatus = ChatpadMouseKMDFGetHidDeviceAttributes(request);
  1066	   }
  1067	   else if(IOCTL_HID_GET_REPORT_DESCRIPTOR == ioControlCode)
  1068	   {
  1069	      apiStatus = ChatpadMouseKMDFGetHidReportDescriptor(wdfDevice, request);
  1070	   }
  1071	   else if(IOCTL_HID_GET_STRING == ioControlCode)
  1072	   {
  1073	      apiStatus = ChatpadMouseKMDFGetHidString(request);
  1074	   }
  1075	   else if(IOCTL_HID_SET_FEATURE == ioControlCode)
  1076	   {
  1077	ChatpadTrace(("Received HID set feature request.\n"));
  1078	// TODO NEXT do I need to support this request?
  1079	apiStatus = STATUS_NOT_SUPPORTED; // TODO NEXT REMOVE
  1080	   }
  1081	   else if(IOCTL_HID_GET_FEATURE == ioControlCode)
  1082	   {
  1083	ChatpadTrace(("Received HID get feature request.\n"));
  1084	// TODO NEXT do I need to support this request?
  1085	apiStatus = STATUS_NOT_SUPPORTED; // TODO NEXT REMOVE
  1086	   }
  1087	   else if(IOCTL_HID_WRITE_REPORT == ioControlCode)
  1088	   {
  1089	ChatpadTrace(("Received HID write report request.\n"));
  1090	// TODO NEXT do I need to support this request?  Apparently so, if I want to track the system caps lock state.
  1091	apiStatus = STATUS_NOT_SUPPORTED; // TODO NEXT REMOVE
  1092	   }
  1093	   else if(IOCTL_HID_READ_REPORT == ioControlCode)
  1094	   {
  1095	      // Queue the request to a manual queue so that it can be handled when we
  1096	      // receive data from the control device.
  1097	      // TODO NEXT do I need to handle IOCTL_HID_GET_INPUT_REPORT as well?
  1098	      //   If only handling IOCTL_HID_READ_REPORT works on XP, Vista, and Windows 7, then apparently I don't.

  1290	// This function retrieves a mouse HID string.

  1293	ChatpadMouseKMDFGetHidString(
  1294	  IN WDFREQUEST   request)
  1295	{
  1296	   NTSTATUS                retStatus                  = STATUS_SUCCESS;
  1297	   PWSTR                   hidString                  = NULL;
  1298	   size_t                  lengthToReturn             = 0;
  1299	   PVOID                   outputStringBuffer         = NULL;
  1300	   size_t                  outputStringBufferLength   = 0;
  1301	   WDF_REQUEST_PARAMETERS  requestParameters;
  1302	   ULONG_PTR               whichStringRequested;
  1303	
  1304	   ChatpadTrace(("Received HID get string request.\n"));

  1306	   WDF_REQUEST_PARAMETERS_INIT(&requestParameters);

  1311	   //   or at LEAST review the rules on whether such constants are always available and thus it is

  1315	     ((reinterpret_cast<ULONG_PTR>(requestParameters.Parameters.DeviceIoControl.Type3InputBuffer)) &

  1384	// This function runs at PASSIVE IRQL.
  1385	VOID
  1386	ChatpadMouseKMDFIOCTLHandler(
  1387	  IN WDFQUEUE     queue,
  1388	  IN WDFREQUEST   request,
  1389	  IN size_t       outputBufferLength,
  1390	  IN size_t       inputBufferLength,
  1391	  IN ULONG        ioControlCode)
  1392	{
  1393	   NTSTATUS                   apiStatus                           = STATUS_SUCCESS;
  1394	   ULONG                      numDriverInstances                  = 0;
  1395	   // Device context for a particular filter driver instance.
  1396	   PDEVICE_CONTEXT            devContext                          = NULL;
  1397	   PVOID                      dataFromUserAsVoid                  = NULL;
  1398	   PUCHAR                     dataFromUser                        = NULL;
  1399	   // The following pointer can point at either a receive buffer or a send buffer.
  1400	   ULONG                      bytesToCopy                         = 0;
  1401	   ULONG                      bytesTransferred                    = 0;
  1402	   BOOLEAN                    immediatelyFailRequest              = FALSE;
  1403	   WDFREQUEST                 requestFromAbove;
  1404	
  1405	//   ChatpadTrace(("The IRQL in the chatpad mouse KMDF IOCTL handler is %u.\n", KeGetCurrentIrql()));
  1406	
  1407	   WdfWaitLockAcquire(chatpadMouseKMDFDeviceCollectionLock, NULL);
  1408	
  1409	   if(TRUE == chatpadMouseKMDFControlDeviceAvailable)
  1410	   {
  1411	      // Get the number of filter driver instances.
  1412	      numDriverInstances = WdfCollectionGetCount(chatpadMouseKMDFDeviceCollection);
  1413	      // Get device context.
  1414	      if(numDriverInstances >= 1)
  1415	      {
  1416	         devContext = ChatpadMouseKMDFGetDeviceContext(WdfCollectionGetItem(chatpadMouseKMDFDeviceCollection, 0));
  1417	      }
  1418	   }
  1419	   else
  1420	   {
  1421	      // If there is no control device available, then we want to return immediately to avoid
  1422	      // crashing.  This situation can happen if a user program has the driver open
  1423	      // and the device is suddenly unplugged.
  1424	      immediatelyFailRequest = TRUE;
  1425	   }
  1426	
  1427	   WdfWaitLockRelease(chatpadMouseKMDFDeviceCollectionLock);
  1428	
  1429	   if(TRUE == immediatelyFailRequest)
  1430	   {
  1431	      ChatpadTrace(("Ignoring control device IOCTL since the device is unavailable.\n"));
  1432	      WdfRequestCompleteWithInformation(request, STATUS_INTERNAL_ERROR, 0);
  1433	      return;
  1434	   }
  1435	
  1436	   if(IOCTL_CHATPAD_MOUSE_KMDF_SET_ENABLED == ioControlCode)
  1437	   {
  1438	      // TODO add constant for how long the buffer should be at minimum
  1439	      if(inputBufferLength != 1)
  1440	      {
  1441	         ChatpadTrace(("Error, input buffer length is %u.\n", inputBufferLength));
  1442	         apiStatus = STATUS_INTERNAL_ERROR;
  1443	      }
  1444	
  1445	      if(STATUS_SUCCESS == apiStatus)
  1446	      {
  1447	         // Get input buffer data.
  1448	         apiStatus = WdfRequestRetrieveInputBuffer(request, inputBufferLength, &dataFromUserAsVoid, NULL);
  1449	         // This ugly cast is probably avoidable, but it works for now.
  1450	         dataFromUser = static_cast<PUCHAR>(dataFromUserAsVoid);
  1451	         if((STATUS_SUCCESS != apiStatus) || (NULL == dataFromUser))
  1452	         {
  1453	            ChatpadTrace(("WdfRequestRetrieveInputBuffer failed:  0x%x\n", apiStatus));
  1454	         }
  1455	      }
  1456	
  1457	      if(STATUS_SUCCESS == apiStatus)
  1458	      {
  1459	         if(CHATPAD_MOUSE_KMDF_ENABLE == dataFromUser[0])
  1460	         {
  1461	            devContext->mouseEnabled = TRUE;
  1462	            ChatpadTrace(("Virtual mouse enabled due to IOCTL command.\n"));
  1463	         }
  1464	         else if(CHATPAD_MOUSE_KMDF_DISABLE == dataFromUser[0])
  1465	         {
  1466	            devContext->mouseEnabled = FALSE;
  1467	            ChatpadTrace(("Virtual mouse disabled due to IOCTL command.\n"));
  1468	         }
  1469	         else
  1470	         {
  1471	            ChatpadTrace(("Unknown mouse enable code %d.\n", dataFromUser[0]));
  1472	         }
  1473	      } // end if(STATUS_SUCCESS == apiStatus)
  1474	   } // end if(IOCTL_CHATPAD_MOUSE_KMDF_SET_ENABLED == ioControlCode)
  1475	   else if(IOCTL_CHATPAD_MOUSE_KMDF_SEND_DATA == ioControlCode)
  1476	   {
  1477	      if(inputBufferLength < 4)
  1478	      {
  1479	         ChatpadTrace(("Error, input buffer length is %u.\n", inputBufferLength));
  1480	         apiStatus = STATUS_INTERNAL_ERROR;
  1481	      }
  1482	
  1483	      if(STATUS_SUCCESS == apiStatus)
  1484	      {
  1485	         // Get input buffer data.
  1486	         apiStatus = WdfRequestRetrieveInputBuffer(request, inputBufferLength, &dataFromUserAsVoid, NULL);
  1487	         // This ugly cast is probably avoidable, but it works for now.
  1488	         dataFromUser = static_cast<PUCHAR>(dataFromUserAsVoid);
  1489	         if((STATUS_SUCCESS != apiStatus) || (NULL == dataFromUser))
  1490	         {
  1491	            ChatpadTrace(("WdfRequestRetrieveInputBuffer failed:  0x%x\n", apiStatus));
  1492	         }
  1493	      }
  1494	
  1495	      // Do not send data out from the virtual mouse device if the device is disabled.
  1496	      if((STATUS_SUCCESS == apiStatus)             &&
  1497	         (TRUE == devContext->mouseEnabled))
  1498	      {
  1499	/*
  1500	         ChatpadTrace((
  1501	           "Virtual mouse device received bytes 0x%02x, 0x%02x, 0x%02x, 0x%02x\n",
  1502	           dataFromUser[0],
  1503	           dataFromUser[1],
  1504	           dataFromUser[2],
  1505	           dataFromUser[3]));
  1506	*/
  1507	
  1508	         apiStatus = WdfIoQueueRetrieveNextRequest(
  1509	           devContext->hidReadRequestQueue,
  1510	           &requestFromAbove);
  1511	
  1512	         if(STATUS_SUCCESS == apiStatus)
  1513	         {
  1514	            PCHATPAD_MOUSE_HID_INPUT_REPORT     inputReport                   = NULL;
  1515	            size_t                              requestFromAboveBufferLength  = 0;
  1516	
  1517	            bytesToCopy = sizeof(CHATPAD_MOUSE_HID_INPUT_REPORT);
  1518	            apiStatus = WdfRequestRetrieveOutputBuffer(
  1519	              requestFromAbove,
  1520	              bytesToCopy,
  1521	              reinterpret_cast<PVOID*>(&inputReport),
  1522	              &requestFromAboveBufferLength);
  1523	            if((STATUS_SUCCESS == apiStatus)                   &&
  1524	               (NULL != inputReport)                           &&
  1525	               (requestFromAboveBufferLength >= bytesToCopy))
  1526	            {
  1527	               // Fill in the report fields.
  1528	               inputReport->buttonStateByte           = dataFromUser[0];
  1529	               inputReport->offsetX                   = dataFromUser[1];
  1530	               inputReport->offsetY                   = dataFromUser[2];
  1531	               inputReport->offsetVerticalScrollWheel = dataFromUser[3];
  1532	
  1533	               // Indicate to the requester how many bytes were transferred.
  1534	               bytesTransferred = bytesToCopy;
  1535	            }
  1536	            else
  1537	            {
  1538	               ChatpadTrace(("Unable to get valid buffer from top-level HID read report request:  0x%x\n",
  1539	                             apiStatus));
  1540	               apiStatus = STATUS_INTERNAL_ERROR;
  1541	            }
  1542	
  1543	            // Return the report to the requester.
  1544	            WdfRequestCompleteWithInformation(requestFromAbove, apiStatus, bytesTransferred);
  1545	         } // end if(STATUS_SUCCESS == apiStatus)
  1546	         else
  1547	         {
  1548	            ChatpadTrace(("No top-level HID read report request available.\n"));
  1549	            apiStatus = STATUS_INTERNAL_ERROR;
  1550	         }
  1551	      } // end if successful and the virtual mouse device is enabled
  1552	      else if((STATUS_SUCCESS == apiStatus) &&
  1553	              (FALSE == devContext->mouseEnabled))
  1554	      {
  1555	         ChatpadTrace(("Ignoring mouse data since virtual mouse device is disabled.\n"));
  1556	      }
  1557	   } // end if(IOCTL_CHATPAD_MOUSE_KMDF_SEND_DATA == ioControlCode)
  1558	   else
  1559	   {
  1560	      ChatpadTrace(("Unknown ioctl control code 0x%x.\n", ioControlCode));
  1561	      // TODO does this status make the driver/device die, or just return an error
  1562	      //   to DeviceIoControl?
  1563	      apiStatus = STATUS_INTERNAL_ERROR;
  1564	   }
  1565	
  1566	//   ChatpadTrace(("Completing ChatpadMouseKMDFIOCTLHandler with bytesTransferred == %lu\n", bytesTransferred));
  1567	
  1568	   // Completing with the length as the information here seems to work, but I cannot seem to find
  1569	   // in the API or an example where that is specified as actually always happening for IOCTLs.
  1570	   // TODO find that information at some point?
  1571	   WdfRequestCompleteWithInformation(request, apiStatus, bytesTransferred);
  1572	
  1573	   return;
  1574	} // end ChatpadMouseKMDFIOCTLHandler
