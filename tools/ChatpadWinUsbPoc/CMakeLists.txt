cmake_minimum_required(VERSION 3.25)
project(ChatpadWinUsbPoc LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "Native Windows target required")
endif()
add_executable(ChatpadTopologyTests Topology.cpp topology-tests.cpp)
target_compile_options(ChatpadTopologyTests PRIVATE /W4 /WX)
enable_testing()
add_test(NAME topology COMMAND ChatpadTopologyTests)
set(PROTOCOL "${CMAKE_CURRENT_SOURCE_DIR}/../../src/protocol/ChatpadProtocol")
add_library(ChatpadBridgeCore STATIC BridgeCore.cpp KeyboardOutput.cpp
  "${PROTOCOL}/ChatpadActivationRequests.c" "${PROTOCOL}/ChatpadActivationSequence.c"
  "${PROTOCOL}/ChatpadLiveTransferPolicy.c" "${PROTOCOL}/ChatpadConfiguration.c"
  "${PROTOCOL}/ChatpadKeyboardHid.c" "${PROTOCOL}/ChatpadKeyboardParser.c")
target_include_directories(ChatpadBridgeCore PUBLIC "${PROTOCOL}")
target_compile_options(ChatpadBridgeCore PRIVATE /W4 /WX)
target_link_libraries(ChatpadBridgeCore PUBLIC user32)
add_executable(ChatpadBridgeTests tests.cpp)
target_link_libraries(ChatpadBridgeTests PRIVATE ChatpadBridgeCore)
target_compile_options(ChatpadBridgeTests PRIVATE /W4 /WX)
add_test(NAME bridge COMMAND ChatpadBridgeTests)
add_executable(ChatpadNativePolicyTests native-policy-tests.cpp)
target_link_libraries(ChatpadNativePolicyTests PRIVATE ChatpadBridgeCore)
target_compile_options(ChatpadNativePolicyTests PRIVATE /W4 /WX)
add_test(NAME native-policy COMMAND ChatpadNativePolicyTests)
add_executable(ChatpadSessionTests session-tests.cpp SessionPolicy.cpp)
target_compile_options(ChatpadSessionTests PRIVATE /W4 /WX)
add_test(NAME session COMMAND ChatpadSessionTests)
add_executable(ChatpadWinUsbPoc main.cpp WinUsbTransport.cpp Topology.cpp VirtualHelper.cpp CompletionPolicy.cpp C3Session.cpp SessionPolicy.cpp)
target_link_libraries(ChatpadWinUsbPoc PRIVATE ChatpadBridgeCore winusb setupapi)
target_compile_options(ChatpadWinUsbPoc PRIVATE /W4 /WX /sdl)
target_compile_definitions(ChatpadWinUsbPoc PRIVATE UNICODE _UNICODE)
target_link_options(ChatpadWinUsbPoc PRIVATE /DYNAMICBASE /NXCOMPAT)
add_executable(ChatpadHelperTests helper-tests.cpp VirtualHelper.cpp)
target_compile_options(ChatpadHelperTests PRIVATE /W4 /WX)
set(CHATPAD_VIRTUAL_HELPER_EXE "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/task-8lc2/virtual/bin/ChatpadVirtualXbox/release/ChatpadVirtualXbox.exe" CACHE FILEPATH "Offline managed helper")
add_test(NAME helper COMMAND ChatpadHelperTests "${CHATPAD_VIRTUAL_HELPER_EXE}")
add_executable(ChatpadCompletionTests CompletionPolicy.cpp completion-tests.cpp)
target_compile_options(ChatpadCompletionTests PRIVATE /W4 /WX)
add_test(NAME completion COMMAND ChatpadCompletionTests)
